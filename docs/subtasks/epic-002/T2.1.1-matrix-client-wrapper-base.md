# T2.1.1 — Create Matrix Client Wrapper Base Class

## Цель
Создать базовый класс `MatrixClientWrapper` с инициализацией и аутентификацией.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| homeserver, user, password | Инициализация + login | Авторизованный Client |
| Авторизованный Client | disconnect() | Завершённая сессия |

## Пошаговый план

1. Создать `bot/matrix/client.py`:

```python
"""Matrix client wrapper for briefer bot."""

import logging
from typing import Optional

from matrix_nio import AsyncClient

logger = logging.getLogger(__name__)


class MatrixClientError(Exception):
    """Base exception for Matrix client errors."""


class MatrixAuthError(MatrixClientError):
    """Authentication failed."""


class MatrixConnectionError(MatrixClientError):
    """Connection to Matrix server failed."""


class MatrixClientWrapper:
    """Wrapper around matrix-nio AsyncClient."""

    def __init__(
        self,
        homeserver: str,
        user_id: str,
        password: str,
        device_name: str = "briefer-bot",
    ):
        """Initialize Matrix client wrapper.

        Args:
            homeserver: Matrix server URL.
            user_id: Full user ID (e.g. @user:server.com).
            password: User password.
            device_name: Device name for login.
        """
        self.homeserver = homeserver.rstrip("/")
        self.user_id = user_id
        self.password = password
        self.device_name = device_name
        self._client: Optional[AsyncClient] = None

    @property
    def is_authenticated(self) -> bool:
        """Check if client is authenticated."""
        return (
            self._client is not None
            and self._client.user_id is not None
            and self._client.access_token is not None
        )

    @property
    def client(self) -> AsyncClient:
        """Get the underlying AsyncClient.

        Raises:
            MatrixClientError: If not initialized.
        """
        if self._client is None:
            raise MatrixClientError("Client not initialized. Call connect() first.")
        return self._client

    async def connect(self) -> None:
        """Initialize and connect to Matrix server.

        Raises:
            MatrixConnectionError: If connection fails.
        """
        try:
            self._client = AsyncClient(
                self.homeserver,
                self.user_id,
                self.password,
            )
            self._client.store_name = f"briefer_{self.user_id.split(':')[1]}"
            logger.info(f"Connected to {self.homeserver} as {self.user_id}")
        except Exception as e:
            self._client = None
            raise MatrixConnectionError(f"Failed to connect: {e}") from e

    async def login(self) -> None:
        """Authenticate with Matrix server.

        Raises:
            MatrixAuthError: If authentication fails.
        """
        if self._client is None:
            await self.connect()

        try:
            await self._client.login(self.password, device_name=self.device_name)
            logger.info(f"Logged in as {self.user_id}")
        except Exception as e:
            raise MatrixAuthError(f"Authentication failed: {e}") from e

    async def disconnect(self) -> None:
        """Disconnect from Matrix server."""
        if self._client is not None:
            try:
                await self._client.sync_stop()
                await self._client.close()
                logger.info(f"Disconnected from {self.homeserver}")
            except Exception as e:
                logger.warning(f"Error during disconnect: {e}")
            finally:
                self._client = None
```

2. Создать `tests/unit/test_matrix_client.py` для проверки.

## Граничные случаи

- homeserver с trailing slash — strip() убирает
- homeserver без trailing slash — не меняет
- user_id без `:` — остаётся как есть
- device_name по умолчанию — "briefer-bot"
- login без connect — вызывает connect() автоматически
- disconnect без connect — ничего не делает
- disconnect с ошибкой — очищает self._client
- is_authenticated до connect — False
- is_authenticated после connect но без login — зависит от access_token
