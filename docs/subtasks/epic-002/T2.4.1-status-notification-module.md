# T2.4.1 — Create Status Notification Module

## Цель
Реализовать отправку статусных сообщений в Matrix комнату.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| status, message | Форматирование → Matrix send | Message ID |
| task_id, progress | Форматирование → Matrix send | Message ID |
| task_id, error | Форматирование → Matrix send | Message ID |

## Пошаговый план

1. Создать `bot/notifications.py`:

```python
"""Status notifications for Matrix bot."""

import logging
from typing import Optional

logger = logging.getLogger(__name__)


class NotificationError(Exception):
    """Notification error."""


class StatusNotifier:
    """Send status notifications to Matrix room."""

    def __init__(self, matrix_client, room_id: str):
        """Initialize notifier.

        Args:
            matrix_client: Matrix client instance.
            room_id: Matrix room ID.
        """
        self.matrix_client = matrix_client
        self.room_id = room_id
        self._message_ids: dict = {}  # task_id -> latest message id

    async def send_status(self, status: str, message: str) -> Optional[str]:
        """Send a status message.

        Args:
            status: Status label (e.g., "Processing", "Ready").
            message: Status message.

        Returns:
            Matrix message ID or None on failure.
        """
        text = f"**{status}**: {message}"
        return await self._send_message(text)

    async def send_progress(self, task_id: str, progress: float, message: str = "") -> Optional[str]:
        """Send a progress update.

        Args:
            task_id: Task ID.
            progress: Progress percentage (0.0 - 1.0).
            message: Optional progress message.

        Returns:
            Matrix message ID or None on failure.
        """
        if progress < 0.0 or progress > 1.0:
            raise NotificationError(f"Invalid progress: {progress}")

        bar_length = 20
        filled = int(bar_length * progress)
        bar = "█" * filled + "░" * (bar_length - filled)
        pct = f"{progress * 100:.0f}%"

        text = f"[{task_id}] **Progress**: {bar} {pct}"
        if message:
            text += f"\n{message}"

        return await self._send_message(text, task_id=task_id)

    async def send_error(self, task_id: str, error: str) -> Optional[str]:
        """Send an error notification.

        Args:
            task_id: Task ID.
            error: Error message.

        Returns:
            Matrix message ID or None on failure.
        """
        text = f"❌ **Error** [{task_id}]: {error}"
        return await self._send_message(text)

    async def send_success(self, task_id: str, message: str = "") -> Optional[str]:
        """Send a success notification.

        Args:
            task_id: Task ID.
            message: Optional success message.

        Returns:
            Matrix message ID or None on failure.
        """
        text = f"✅ **Success** [{task_id}]: {message}"
        return await self._send_message(text)

    async def _send_message(self, text: str, task_id: Optional[str] = None) -> Optional[str]:
        """Send a message to the room.

        Args:
            text: Message text.
            task_id: Optional task ID for tracking.

        Returns:
            Matrix message ID or None on failure.
        """
        try:
            msg_id = await self.matrix_client.send_text(
                room_id=self.room_id,
                text=text,
                html_text=self._to_html(text),
            )
            if task_id:
                self._message_ids[task_id] = msg_id
            logger.info(f"Notification sent: {msg_id}")
            return msg_id
        except Exception as e:
            logger.error(f"Failed to send notification: {e}")
            return None

    @staticmethod
    def _to_html(text: str) -> str:
        """Convert plain text to HTML.

        Args:
            text: Plain text.

        Returns:
            HTML text.
        """
        html = text.replace("**", "", 1) if text.startswith("**") else text
        # Simple markdown → HTML conversion
        import re
        html = re.sub(r"\*\*(.+?)\*\*", r"<b>\1</b>", html)
        html = re.sub(r"\n", "<br>", html)
        return html
```

## Граничные случаи

- progress = 0.0 — "░░░░░░░░░░░░░░░░░░░░" 0%
- progress = 1.0 — "████████████████████" 100%
- progress = 0.5 — "█████████████░░░░░░░" 50%
- progress < 0.0 — NotificationError
- progress > 1.0 — NotificationError
- progress = -0.01 — NotificationError
- progress = 1.01 — NotificationError
- matrix_client.send_text raises — None returned, error logged
- room_id пустой — ValueError
- task_id пустой — OK (message sent without task_id tracking)
- send_status без message — "" → "**Status**: "
- send_error без error — "" → "❌ Error []: "
- _to_html с markdown — converts **bold** to <b>bold</b>
- _to_html с newline — converts \n to <br>
