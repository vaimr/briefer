# T2.2.1 — Create Audio Download Module

## Цель
Реализовать скачивание аудиофайлов из Matrix с валидацией MIME и magic bytes.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| URL аудиофайла | Скачивание + валидация | Путь к файлу |
| URL без аудио | Проверка MIME/magic | ValueError |

## Пошаговый план

1. Создать `worker/audio.py`:

```python
"""Audio file download from URLs."""

import logging
import mimetypes
import os
import tempfile
from pathlib import Path
from typing import Optional

import aiohttp

logger = logging.getLogger(__name__)

ALLOWED_MIME_TYPES = {
    "audio/mpeg", "audio/ogg", "audio/wav", "audio/webm",
    "audio/mp4", "audio/x-m4a", "audio/flac", "audio/aac",
}

ALLOWED_EXTENSIONS = {".mp3", ".ogg", ".wav", ".webm", ".m4a", ".flac", ".aac", ".opus"}


class AudioDownloadError(Exception):
    """Audio download failed."""


class AudioDownloader:
    """Download audio files from URLs."""

    MAX_FILE_SIZE = 100 * 1024 * 1024  # 100 MB
    DOWNLOAD_TIMEOUT = 300  # 5 minutes

    def __init__(self, download_dir: Optional[str] = None):
        self.download_dir = download_dir or tempfile.gettempdir()
        Path(self.download_dir).mkdir(parents=True, exist_ok=True)

    async def download(
        self,
        url: str,
        filename: Optional[str] = None,
        expected_mime: Optional[str] = None,
    ) -> str:
        """Download audio file from URL.

        Args:
            url: Audio file URL.
            filename: Optional output filename.
            expected_mime: Expected MIME type.

        Returns:
            Path to downloaded file.

        Raises:
            AudioDownloadError: On download failure.
        """
        if not url or not url.strip():
            raise AudioDownloadError("Empty URL")

        if not filename:
            filename = self._extract_filename(url)

        output_path = os.path.join(self.download_dir, filename)

        async with aiohttp.ClientSession() as session:
            async with session.get(url, timeout=aiohttp.ClientTimeout(total=self.DOWNLOAD_TIMEOUT)) as resp:
                if resp.status != 200:
                    raise AudioDownloadError(f"HTTP {resp.status}: {url}")

                content_type = resp.headers.get("Content-Type", "")
                if expected_mime and expected_mime not in content_type:
                    raise AudioDownloadError(
                        f"Unexpected MIME type: {content_type} (expected {expected_mime})"
                    )

                total_size = int(resp.headers.get("Content-Length", 0))
                if total_size > self.MAX_FILE_SIZE:
                    raise AudioDownloadError(f"File too large: {total_size} bytes")

                content = await resp.read()

                if not self._is_valid_audio(content, content_type):
                    raise AudioDownloadError(f"Invalid audio content: {content_type}")

                with open(output_path, "wb") as f:
                    f.write(content)

                logger.info(f"Downloaded {url} -> {output_path} ({len(content)} bytes)")
                return output_path

    def _extract_filename(self, url: str) -> str:
        """Extract filename from URL."""
        from urllib.parse import urlparse

        parsed = urlparse(url)
        path = parsed.path
        filename = os.path.basename(path)

        if not os.path.splitext(filename)[1]:
            guessed = mimetypes.guess_extension(
                mimetypes.guess_type(url)[0]
            )
            if guessed:
                filename += guessed
            else:
                filename += ".mp3"

        return filename

    def _is_valid_audio(self, content: bytes, content_type: str) -> bool:
        """Validate that content is audio."""
        if content_type.split(";")[0] in ALLOWED_MIME_TYPES:
            return True

        if len(content) >= 3 and content[:3] == b"ID3":  # MP3 with ID3 tag
            return True
        if len(content) >= 4 and content[:4] == b"OggS":  # OGG
            return True
        if len(content) >= 12 and content[:4] == b"RIFF" and content[8:12] == b"WAVE":  # WAV
            return True

        return False
```

## Граничные случаи

- Пустой URL — AudioDownloadError("Empty URL")
- URL только из пробелов — AudioDownloadError("Empty URL")
- URL без расширения — добавляет .mp3
- URL с query params — basename из path
- HTTP 404 — AudioDownloadError("HTTP 404")
- HTTP 200, wrong MIME — AudioDownloadError
- File > 100MB — AudioDownloadError("File too large")
- MP3 с ID3 tag — magic bytes валидация
- OGG — magic bytes валидация
- WAV — magic bytes валидация
- Non-audio content — AudioDownloadError
- expected_mime mismatch — AudioDownloadError
- download_dir не существует — создаётся
