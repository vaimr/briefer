```python
"""Audio file format validation."""

import logging
import os
from dataclasses import dataclass
from typing import Optional

logger = logging.getLogger(__name__)


@dataclass
class AudioMetadata:
    """Audio file metadata."""
    file_path: str
    file_size: int = 0
    mime_type: str = ""
    extension: str = ""
    is_valid: bool = True
    error: str = ""


class AudioValidationError(Exception):
    """Audio validation failed."""


ALLOWED_EXTENSIONS = {".mp3", ".ogg", ".wav", ".webm", ".m4a", ".flac", ".aac", ".opus"}
MIN_FILE_SIZE = 100  # bytes
MAX_FILE_SIZE = 100 * 1024 * 1024  # 100 MB


class AudioValidator:
    """Validate audio file format."""

    def __init__(self, allowed_extensions: Optional[set] = None):
        """Initialize validator.

        Args:
            allowed_extensions: Set of allowed file extensions.
        """
        self.allowed_extensions = allowed_extensions or ALLOWED_EXTENSIONS

    def validate(self, file_path: str) -> AudioMetadata:
        """Validate audio file.

        Args:
            file_path: Path to audio file.

        Returns:
            AudioMetadata with validation results.

        Raises:
            AudioValidationError: If file is invalid.
        """
        metadata = AudioMetadata(file_path=file_path)

        # Check file exists
        if not os.path.exists(file_path):
            metadata.is_valid = False
            metadata.error = f"File not found: {file_path}"
            raise AudioValidationError(metadata.error)

        # Check file size
        file_size = os.path.getsize(file_path)
        metadata.file_size = file_size

        if file_size < MIN_FILE_SIZE:
            metadata.is_valid = False
            metadata.error = f"File too small: {file_size} bytes"
            raise AudioValidationError(metadata.error)

        if file_size > MAX_FILE_SIZE:
            metadata.is_valid = False
            metadata.error = f"File too large: {file_size} bytes"
            raise AudioValidationError(metadata.error)

        # Check extension
        _, ext = os.path.splitext(file_path)
        ext = ext.lower()
        metadata.extension = ext

        if ext not in self.allowed_extensions:
            metadata.is_valid = False
            metadata.error = f"Unsupported extension: {ext}"
            raise AudioValidationError(metadata.error)

        # Check magic bytes
        if not self._check_magic_bytes(file_path):
            metadata.is_valid = False
            metadata.error = "Invalid audio magic bytes"
            raise AudioValidationError(metadata.error)

        metadata.is_valid = True
        return metadata

    def is_valid(self, file_path: str) -> bool:
        """Quick validation check.

        Args:
            file_path: Path to audio file.

        Returns:
            True if valid audio file.
        """
        try:
            self.validate(file_path)
            return True
        except AudioValidationError:
            return False

    def _check_magic_bytes(self, file_path: str) -> bool:
        """Check file magic bytes.

        Args:
            file_path: Path to file.

        Returns:
            True if magic bytes match audio format.
        """
        with open(file_path, "rb") as f:
            header = f.read(12)

        if len(header) < 4:
            return False

        # MP3: ID3 tag or MPEG frame sync
        if header[:3] == b"ID3":
            return True
        if header[:2] == b"\xff\xfb" or header[:2] == b"\xff\xf3":
            return True

        # OGG
        if header[:4] == b"OggS":
            return True

        # WAV
        if header[:4] == b"RIFF" and len(header) >= 12 and header[8:12] == b"WAVE":
            return True

        # FLAC
        if header[:4] == b"fLaC":
            return True

        # AAC/MP4: check for 'ftyp' box
        if header[4:8] == b"ftyp":
            return True

        return False
```

## Граничные случаи

- Файл не существует — AudioValidationError("File not found")
- Пустой файл (0 байт) — AudioValidationError("File too small")
- Файл 50 байт — AudioValidationError("File too small")
- Файл 100 байт — OK (MIN_FILE_SIZE)
- Файл 101 байт — OK (если расширение и magic bytes верны)
- Файл 100MB+1 — AudioValidationError("File too large")
- Неизвестное расширение — AudioValidationError
- .mp3 без ID3 — magic bytes MPEG sync
- .ogg с OggS — OK
- .wav с RIFF+WAVE — OK
- .flac с fLaC — OK
- .m4a с ftyp — OK
- Путь с пробелами — os.path работает
- Путь с unicode — os.path работает
