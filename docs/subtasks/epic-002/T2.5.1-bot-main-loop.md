# T2.5.1 — Create Bot Main Loop

## Цель
Реализовать основной цикл бота с message listener и обработкой запросов.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| Matrix event | Dispatch → handler | Task submitted |
| /transcribe command | Audio download → queue | Task ID |
| /summary command | Audio download → queue | Task ID |

## Пошаговый план

1. Создать `bot/bot.py`:

```python
"""Main bot loop for Matrix transcription bot."""

import asyncio
import logging
import signal
import sys
from typing import Optional

logger = logging.getLogger(__name__)


class BotError(Exception):
    """Bot error."""


class Bot:
    """Main bot class."""

    def __init__(
        self,
        matrix_client,
        queue_producer,
        notifier,
        room_ids: Optional[list] = None,
    ):
        """Initialize bot.

        Args:
            matrix_client: Matrix client instance.
            queue_producer: QueueProducer instance.
            notifier: StatusNotifier instance.
            room_ids: List of room IDs to listen to.
        """
        self.matrix_client = matrix_client
        self.queue_producer = queue_producer
        self.notifier = notifier
        self.room_ids = room_ids or []
        self._running = False
        self._task_id = None

    async def start(self) -> None:
        """Start the bot."""
        self._running = True
        logger.info("Starting bot...")

        # Setup signal handlers
        loop = asyncio.get_running_loop()
        for sig in (signal.SIGINT, signal.SIGTERM):
            loop.add_signal_handler(sig, self._shutdown)

        try:
            await self.matrix_client.sync()
            logger.info("Bot started successfully")
        except Exception as e:
            logger.error(f"Failed to start bot: {e}")
            raise BotError(f"Bot start failed: {e}") from e

    async def stop(self) -> None:
        """Stop the bot."""
        logger.info("Stopping bot...")
        self._running = False
        await self.matrix_client.stop()

    def _shutdown(self) -> None:
        """Signal shutdown."""
        logger.info("Shutdown signal received")
        self._running = False

    async def on_message(self, event: dict) -> None:
        """Handle incoming message event.

        Args:
            event: Matrix event dict.
        """
        if not self._running:
            return

        msg_type = event.get("type", "")
        if msg_type != "m.room.message":
            return

        room_id = event.get("room_id", "")
        sender = event.get("sender", "")
        content = event.get("content", {})
        msg_body = content.get("body", "")

        logger.info(f"Message from {sender} in {room_id}: {msg_body[:50]}")

        # Handle commands
        if msg_body.startswith("/transcribe"):
            await self.handle_transcription_request(event)
        elif msg_body.startswith("/summary"):
            await self.handle_summary_request(event)
        else:
            logger.info(f"Unknown command: {msg_body}")

    async def handle_transcription_request(self, event: dict) -> None:
        """Handle /transcribe command.

        Args:
            event: Matrix event dict.
        """
        room_id = event.get("room_id", "")
        task_id = self.queue_producer.submit_task({
            "type": "transcription",
            "room_id": room_id,
            "sender": event.get("sender", ""),
            "body": event.get("content", {}).get("body", ""),
        })

        await self.notifier.send_status("Transcription", f"Task {task_id} created")
        logger.info(f"Transcription task {task_id} submitted")

    async def handle_summary_request(self, event: dict) -> None:
        """Handle /summary command.

        Args:
            event: Matrix event dict.
        """
        room_id = event.get("room_id", "")
        task_id = self.queue_producer.submit_task({
            "type": "summary",
            "room_id": room_id,
            "sender": event.get("sender", ""),
            "body": event.get("content", {}).get("body", ""),
        })

        await self.notifier.send_status("Summary", f"Task {task_id} created")
        logger.info(f"Summary task {task_id} submitted")

    async def run(self) -> None:
        """Run the bot until shutdown."""
        await self.start()
        try:
            while self._running:
                await asyncio.sleep(0.1)
        except asyncio.CancelledError:
            logger.info("Bot cancelled")
        finally:
            await self.stop()
```

## Граничные случаи

- _running = False при on_message — return без обработки
- event type != m.room.message — return без обработки
- event без room_id — default ""
- event без sender — default ""
- event без content — content.get("body", "") → ""
- msg_body без /transcribe или /summary — "Unknown command" log
- /transcribe с body — transcription task в queue
- /summary с body — summary task в queue
- SIGINT во время run() — _running = False, loop exits
- SIGTERM во время run() — _running = False, loop exits
- start() без matrix_client.sync() — BotError
- stop() без start() — matrix_client.stop() вызывается
- run() → CancelledError — finally: stop()
