```python
"""Matrix room management."""

import logging
from dataclasses import dataclass, field
from typing import List, Optional

from matrix_nio import AsyncClient

from bot.matrix.client import MatrixClientError, MatrixClientWrapper

logger = logging.getLogger(__name__)


@dataclass
class RoomInfo:
    """Information about a Matrix room."""
    room_id: str
    name: str = ""
    canonical_alias: Optional[str] = None
    topic: str = ""
    member_count: int = 0
    is_direct: bool = False
    tags: List[str] = field(default_factory=list)

    def __str__(self) -> str:
        alias = self.canonical_alias or self.room_id
        return f"RoomInfo(name='{self.name}', alias='{alias}', members={self.member_count})"


class RoomNotFoundError(MatrixClientError):
    """Room not found."""


class RoomManagementError(MatrixClientError):
    """Room management operation failed."""


class RoomManager:
    """Manage Matrix rooms."""

    def __init__(self, client_wrapper: MatrixClientWrapper):
        """Initialize room manager.

        Args:
            client_wrapper: MatrixClientWrapper instance.
        """
        self._wrapper = client_wrapper

    def _get_client(self) -> AsyncClient:
        """Get the AsyncClient instance.

        Raises:
            MatrixClientError: If client not available.
        """
        return self._wrapper.client

    async def list_rooms(self, filter_name: Optional[str] = None) -> List[RoomInfo]:
        """List all rooms the bot is in.

        Args:
            filter_name: Optional name filter (case-insensitive substring match).

        Returns:
            List of RoomInfo.
        """
        client = self._get_client()
        rooms = await client.sync()
        room_list = rooms.get("rooms", {}).get("join", {})

        result = []
        for room_id, room_data in room_list.items():
            room_info = self._parse_room_info(room_id, room_data)
            if filter_name and filter_name.lower() not in room_info.name.lower():
                continue
            result.append(room_info)

        return result

    async def get_room(self, room_id: str) -> RoomInfo:
        """Get information about a specific room.

        Args:
            room_id: Matrix room ID.

        Returns:
            RoomInfo for the room.

        Raises:
            RoomNotFoundError: If room not found.
        """
        client = self._get_client()
        room_list = await client.joined_rooms()

        if room_id not in room_list:
            raise RoomNotFoundError(f"Room {room_id} not found or not joined")

        # Get room state
        state = await client.room_state(room_id)
        name = ""
        topic = ""
        canonical_alias = None
        is_direct = False

        for event in state:
            if event.type == "m.room.name" and event.content:
                name = event.content.get("name", "")
            elif event.type == "m.room.topic" and event.content:
                topic = event.content.get("topic", "")
            elif event.type == "m.room.canonical_alias" and event.content:
                canonical_alias = event.content.get("alias")
            elif event.type == "m.room.join_rules" and event.content:
                is_direct = event.content.get("join_rule") == "invite"

        member_count = len(await client.room_members(room_id))

        return RoomInfo(
            room_id=room_id,
            name=name,
            canonical_alias=canonical_alias,
            topic=topic,
            member_count=member_count,
            is_direct=is_direct,
        )

    async def create_room(self, name: str, topic: str = "") -> str:
        """Create a new Matrix room.

        Args:
            name: Room name.
            topic: Room topic.

        Returns:
            New room ID.

        Raises:
            RoomManagementError: If room creation fails.
        """
        client = self._get_client()
        try:
            room_id = await client.create_room(name=name, topic=topic)
            logger.info(f"Created room {name}: {room_id}")
            return room_id
        except Exception as e:
            raise RoomManagementError(f"Failed to create room: {e}") from e

    async def set_room_topic(self, room_id: str, topic: str) -> None:
        """Set the topic of a room.

        Args:
            room_id: Matrix room ID.
            topic: New topic.

        Raises:
            RoomManagementError: If setting topic fails.
        """
        client = self._get_client()
        try:
            await client.room_send(
                room_id=room_id,
                message_type="m.room.topic",
                content={"topic": topic},
            )
        except Exception as e:
            raise RoomManagementError(f"Failed to set topic: {e}") from e

    def _parse_room_info(self, room_id: str, room_data: dict) -> RoomInfo:
        """Parse room data from sync response.

        Args:
            room_id: Room ID.
            room_data: Room data dict from sync.

        Returns:
            RoomInfo instance.
        """
        state = room_data.get("state", {})
        timeline = room_data.get("timeline", {})

        name = ""
        topic = ""
        canonical_alias = None

        for event in state.get("events", []):
            if event.get("type") == "m.room.name":
                name = event.get("content", {}).get("name", "")
            elif event.get("type") == "m.room.topic":
                topic = event.get("content", {}).get("topic", "")
            elif event.get("type") == "m.room.canonical_alias":
                canonical_alias = event.get("content", {}).get("alias")

        # Get member count from timeline (approximate)
        member_count = len(timeline.get("events", []))

        return RoomInfo(
            room_id=room_id,
            name=name,
            canonical_alias=canonical_alias,
            topic=topic,
            member_count=member_count,
        )
```

## Граничные случаи

- `list_rooms()` без комнат — пустой список
- `list_rooms(filter_name="nonexistent")` — пустой список
- `list_rooms(filter_name="")` — возвращает все комнаты
- filter_name case-insensitive — "TEST" находит "Test Room"
- `get_room()` несуществующий ID — RoomNotFoundError
- `get_room()` с пустым room_id — RoomNotFoundError
- `create_room()` с пустым именем — создаёт комнату с пустым именем
- `create_room()` с длинным именем — создаёт комнату
- `set_room_topic()` на несуществующей комнате — RoomManagementError
- `_parse_room_info()` без state events — name="", topic=""
- `_parse_room_info()` с canonical_alias — alias из events
