## Цель

## Spec

## Пошаговый план

1. Создать `worker/queue_producer.py`:

```python
"""Redis queue producer for task submission."""

import json
import logging
import uuid
from typing import Any, Dict, Optional

import redis

logger = logging.getLogger(__name__)


class QueueProducerError(Exception):
    """Queue producer error."""


class QueueProducer:
    """Produce tasks to Redis queue."""

    def __init__(
        self,
        redis_url: str = "redis://localhost:6379/0",
        queue_name: str = "briefer:tasks",
    ):
        """Initialize queue producer.

        Args:
            redis_url: Redis connection URL.
            queue_name: Queue name.
        """
        self.redis_url = redis_url
        self.queue_name = queue_name
        self._redis: Optional[redis.Redis] = None

    def _get_connection(self) -> redis.Redis:
        """Get Redis connection.

        Raises:
            QueueProducerError: If connection fails.
        """
        if self._redis is None:
            try:
                self._redis = redis.from_url(
                    self.redis_url,
                    decode_responses=False,
                    socket_timeout=5,
                    socket_connect_timeout=5,
                )
                self._redis.ping()
            except redis.ConnectionError as e:
                raise QueueProducerError(f"Redis connection failed: {e}") from e
        return self._redis

    def submit_task(
        self,
        task_data: Dict[str, Any],
        priority: int = 0,
    ) -> str:
        """Submit a task to the queue.

        Args:
            task_data: Task data dict.
            priority: Priority level (higher = more urgent).

        Returns:
            Task ID.

        Raises:
            QueueProducerError: On submission failure.
        """
        if not task_data or not task_data.get("task_id"):
            task_data["task_id"] = str(uuid.uuid4())

        task_id = task_data["task_id"]

        # Add priority as prefix for sorted queue
        # Redis sorted set: score = priority, member = JSON task
        task_json = json.dumps(task_data).encode("utf-8")

        try:
            redis_conn = self._get_connection()
            # Use sorted set for priority queue
            redis_conn.zadd(self.queue_name, {task_json: priority})
            logger.info(f"Submitted task {task_id} with priority {priority}")
            return task_id
        except redis.RedisError as e:
            raise QueueProducerError(f"Failed to submit task: {e}") from e

    def submit_batch(
        self,
        tasks: list,
        priority: int = 0,
    ) -> list:
        """Submit multiple tasks to the queue.

        Args:
            tasks: List of task data dicts.
            priority: Priority level.

        Returns:
            List of task IDs.
        """
        task_ids = []
        for task_data in tasks:
            task_id = self.submit_task(task_data, priority)
            task_ids.append(task_id)
        return task_ids

    def queue_length(self) -> int:
        """Get the current queue length.

        Returns:
            Number of tasks in queue.
        """
        try:
            redis_conn = self._get_connection()
            return redis_conn.zcard(self.queue_name)
        except redis.RedisError as e:
            logger.error(f"Failed to get queue length: {e}")
            return -1

    def close(self) -> None:
        """Close Redis connection."""
        if self._redis is not None:
            self._redis.close()
            self._redis = None
```

## Граничные случаи

- task_data без task_id — генерируется UUID
- task_data с task_id — используется существующий
- Пустой task_data — генерируется UUID
- priority = 0 — default
- priority = -1 — низкий приоритет
- priority = 100 — высокий приоритет
- Redis недоступен — QueueProducerError
- submit_batch с пустым списком — []
- submit_batch с 1 задачей — [task_id]
- queue_length после submit — 1
- close без connect — ничего не делает
