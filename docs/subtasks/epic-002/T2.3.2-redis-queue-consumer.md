# T2.3.2 — Create Redis Queue Consumer

## Цель
Реализовать потребление задач из Redis queue с обработкой ошибок и graceful shutdown.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| Task JSON из очереди | Deserialise + callback | Task result |
| Shutdown signal | Stop loop | Clean exit |

## Пошаговый план

1. Создать `worker/queue_consumer.py`:

```python
"""Redis queue consumer for task processing."""

import asyncio
import json
import logging
import signal
from typing import Any, Callable, Dict, Optional

import redis

logger = logging.getLogger(__name__)


class QueueConsumerError(Exception):
    """Queue consumer error."""


class QueueConsumer:
    """Consume tasks from Redis queue."""

    def __init__(
        self,
        redis_url: str = "redis://localhost:6379/0",
        queue_name: str = "briefer:tasks",
        callback: Optional[Callable[[Dict[str, Any]], Any]] = None,
    ):
        """Initialize queue consumer.

        Args:
            redis_url: Redis connection URL.
            queue_name: Queue name.
            callback: Task processing callback.
        """
        self.redis_url = redis_url
        self.queue_name = queue_name
        self.callback = callback
        self._redis: Optional[redis.Redis] = None
        self._running = False
        self._shutdown_event = asyncio.Event()

    def _get_connection(self) -> redis.Redis:
        """Get Redis connection.

        Raises:
            QueueConsumerError: If connection fails.
        """
        if self._redis is None:
            try:
                self._redis = redis.from_url(
                    self.redis_url,
                    decode_responses=True,
                    socket_timeout=5,
                    socket_connect_timeout=5,
                )
                self._redis.ping()
            except redis.ConnectionError as e:
                raise QueueConsumerError(f"Redis connection failed: {e}") from e
        return self._redis

    async def start(self) -> None:
        """Start consuming tasks from queue."""
        self._running = True
        logger.info(f"Starting queue consumer on {self.queue_name}")

        # Setup signal handlers
        loop = asyncio.get_running_loop()
        for sig in (signal.SIGINT, signal.SIGTERM):
            loop.add_signal_handler(sig, self._shutdown)

        try:
            while self._running and not self._shutdown_event.is_set():
                await self._process_next()
        except QueueConsumerError as e:
            logger.error(f"Queue consumer error: {e}")
        finally:
            self._running = False
            logger.info("Queue consumer stopped")

    def _shutdown(self) -> None:
        """Signal shutdown."""
        logger.info("Shutdown signal received")
        self._running = False
        self._shutdown_event.set()

    async def _process_next(self) -> None:
        """Process next task from queue."""
        try:
            redis_conn = self._get_connection()
            # Pop from sorted set (lowest score first = lowest priority)
            result = redis_conn.zpopmin(self.queue_name, count=1)

            if not result:
                await asyncio.sleep(0.1)  # No tasks, wait
                return

            task_json, _priority = result[0]
            task_data = json.loads(task_json)

            logger.info(f"Processing task: {task_data.get('task_id', 'unknown')}")

            if self.callback:
                try:
                    result_data = self.callback(task_data)
                    logger.info(f"Task completed: {task_data.get('task_id', 'unknown')}")
                except Exception as e:
                    logger.error(f"Task failed: {task_data.get('task_id', 'unknown')}: {e}")
                    raise
        except redis.RedisError as e:
            logger.error(f"Redis error while processing: {e}")
            await asyncio.sleep(1)  # Wait before retry

    def stop(self) -> None:
        """Stop the consumer."""
        self._running = False
        self._shutdown_event.set()

    def close(self) -> None:
        """Close Redis connection."""
        if self._redis is not None:
            self._redis.close()
            self._redis = None
        self._running = False
        self._shutdown_event.set()
```

## Граничные случаи

- Redis недоступен при старте — QueueConsumerError
- Redis недоступен при обработке — логирование + retry через 1 секунду
- Пустая очередь — sleep 0.1s, не блокирует
- Callback raises — логирование ошибки, не ломает consumer
- Callback возвращает None — OK
- Callback возвращает dict — OK
- SIGINT во время sleep — shutdown_event.set()
- SIGTERM во время обработки — shutdown_event.set()
- start() без callback — пропускает обработку, просто pop
- stop() во время start() — _running = False
- close() без start() — ничего не ломает
- queue_name с colon — OK (Redis key)
