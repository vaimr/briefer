```python
"""Matrix message handling."""

import logging
import re
from dataclasses import dataclass
from enum import Enum
from typing import Dict, List, Optional, Tuple

from matrix_nio import AsyncClient

from bot.matrix.client import MatrixClientWrapper
from bot.matrix.rooms import RoomInfo

logger = logging.getLogger(__name__)


class CommandType(Enum):
    """Bot command types."""
    HELP = "help"
    STATUS = "status"
    PROCESS = "process"
    CANCEL = "cancel"


@dataclass
class ParsedCommand:
    """Parsed bot command."""
    command: CommandType
    arguments: List[str] = None
    room_id: str = ""
    sender: str = ""

    def __post_init__(self):
        if self.arguments is None:
            self.arguments = []


class CommandParseError(Exception):
    """Failed to parse command."""


class MessageHandler:
    """Handle incoming Matrix messages."""

    COMMAND_PATTERN = re.compile(r"^!briefer\s+(\w+)(?:\s+(.+))?$")
    MEDIA_MIMES = {
        "m.image": ["image/png", "image/jpeg", "image/gif", "image/webp"],
        "m.audio": ["audio/mpeg", "audio/ogg", "audio/wav", "audio/webm", "audio/mp4"],
        "m.video": ["video/mp4", "video/webm", "video/ogg"],
        "m.file": ["application/pdf", "application/zip", "application/gzip"],
    }

    def __init__(self, client_wrapper: MatrixClientWrapper):
        self._wrapper = client_wrapper

    def parse_command(self, text: str, room_id: str = "", sender: str = "") -> ParsedCommand:
        """Parse a bot command from text.

        Args:
            text: Message text.
            room_id: Room ID where command was sent.
            sender: Sender user ID.

        Returns:
            ParsedCommand.

        Raises:
            CommandParseError: If text is not a valid command.
        """
        if not text or not text.strip():
            raise CommandParseError("Empty message")

        match = self.COMMAND_PATTERN.match(text.strip())
        if not match:
            raise CommandParseError(f"Not a valid command: {text[:50]}")

        command_name = match.group(1).lower()
        args_str = match.group(2) or ""
        args = args_str.split() if args_str else []

        try:
            command = CommandType(command_name)
        except ValueError:
            raise CommandParseError(f"Unknown command: {command_name}")

        return ParsedCommand(
            command=command,
            arguments=args,
            room_id=room_id,
            sender=sender,
        )

    def is_media_message(self, event_type: str, content: dict) -> bool:
        """Check if event is a media message.

        Args:
            event_type: Matrix event type (m.image, m.audio, etc.).
            content: Event content.

        Returns:
            True if media message.
        """
        if event_type not in self.MEDIA_MIMES:
            return False

        msg_type = content.get("msgtype", "")
        return msg_type in self.MEDIA_MIMES[event_type]

    def get_media_info(self, content: dict) -> Dict[str, str]:
        """Extract media info from content.

        Args:
            content: Matrix event content.

        Returns:
            Dict with media_url, info, thumbnail.
        """
        return {
            "url": content.get("url", ""),
            "info": content.get("info", {}),
            "body": content.get("body", ""),
            "msgtype": content.get("msgtype", ""),
        }

    async def handle_command(
        self,
        command: ParsedCommand,
        room_info: RoomInfo,
    ) -> Optional[str]:
        """Handle a parsed command.

        Args:
            command: Parsed command.
            room_info: Room info.

        Returns:
            Reply message text, or None if no reply needed.
        """
        client = self._wrapper.client

        if command.command == CommandType.HELP:
            return self._help_reply()
        elif command.command == CommandType.STATUS:
            return self._status_reply(room_info)
        elif command.command == CommandType.PROCESS:
            return self._process_reply(command.arguments, room_info)
        elif command.command == CommandType.CANCEL:
            return self._cancel_reply(room_info)

        return None

    def _help_reply(self) -> str:
        """Generate help reply."""
        return (
            "**Доступные команды:**\n"
            "!briefer help — показать справку\n"
            "!briefer status — статус обработки\n"
            "!briefer process <url> — обработать ссылку\n"
            "!briefer cancel — отменить обработку"
        )

    def _status_reply(self, room_info: RoomInfo) -> str:
        """Generate status reply."""
        return f"Статус комнаты: {room_info.name} (ID: {room_info.room_id})"

    def _process_reply(self, args: List[str], room_info: RoomInfo) -> str:
        """Generate process reply."""
        if not args:
            return "Укажите URL для обработки: !briefer process <url>"
        return f"Начинаю обработку: {args[0]}"

    def _cancel_reply(self, room_info: RoomInfo) -> str:
        """Generate cancel reply."""
        return f"Отмена обработки в {room_info.name}"
```

## Граничные случаи

- Пустая строка — CommandParseError
- Строка без `!briefer` — CommandParseError
- `!briefer help` — CommandType.HELP
- `!briefer help arg1 arg2` — HELP с аргументами
- `!briefer unknown` — CommandParseError (unknown command)
- `!briefer PROCESS URL` — case-insensitive, PROCESS → process
- `!briefer  multiple` — double space → parse
- `!briefer single` — без пробела после команды
- Media MIME проверка: image/png → is_media_message=True
- Media MIME проверка: text/plain → is_media_message=False
- get_media_info без url — url = ""
- is_media_message с m.sticker → False (не в MEDIA_MIMES)
