# T5.3.1 — Create Bot Error Handling for Results

## Цель
Реализовать обработку ошибок при доставке результатов в Matrix с retry-логикой.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — ID задачи; `error: Exception` — ошибка доставки; `room_id: str` — Matrix room ID |
| **Обработка** | 1. Классификация ошибки (transient vs permanent). 2. Retry с exponential backoff для transient ошибок. 3. Отправка уведомления об ошибке. |
| **Выход** | `bool` — True если delivery succeeded после retry, False если permanent failure |

## Пошаговый план

1. **Создать `bot/result_errors.py`**:
```python
import logging
import time
from typing import Optional

logger = logging.getLogger(__name__)

class DeliveryError(Exception):
    """Error delivering result to Matrix."""
    def __init__(self, task_id: str, message: str):
        self.task_id = task_id
        super().__init__(message)

class RetryableError(DeliveryError):
    """Transient error that can be retried."""
    pass

class PermanentError(DeliveryError):
    """Permanent error that should not be retried."""
    pass

class ResultErrorHandler:
    """Handle errors in result delivery to Matrix."""

    def __init__(
        self,
        matrix_client,
        room_id: str,
        max_retries: int = 3,
        base_delay: float = 1.0,
    ):
        self.matrix_client = matrix_client
        self.room_id = room_id
        self.max_retries = max_retries
        self.base_delay = base_delay

    def handle_delivery_error(
        self,
        task_id: str,
        error: Exception,
        delivery_fn=None,
    ) -> bool:
        """Handle a delivery error with retry logic.

        Args:
            task_id: Task identifier.
            error: The delivery error.
            delivery_fn: Optional retry function to call.

        Returns:
            True if succeeded after retry, False if permanent failure.
        """
        is_retryable = self._is_retryable(error)

        if not is_retryable:
            logger.error(
                "Permanent delivery error for task_id=%s: %s",
                task_id, error,
            )
            self.send_error_notification(task_id, error)
            return False

        # Retry with exponential backoff
        for attempt in range(1, self.max_retries + 1):
            try:
                if delivery_fn:
                    delivery_fn()
                logger.info(
                    "Retry %d/%d succeeded for task_id=%s",
                    attempt, self.max_retries, task_id,
                )
                return True
            except Exception as retry_error:
                logger.warning(
                    "Retry %d/%d failed for task_id=%s: %s",
                    attempt, self.max_retries, task_id, retry_error,
                )
                if attempt < self.max_retries:
                    delay = self.base_delay * (2 ** (attempt - 1))
                    time.sleep(delay)

        # All retries exhausted
        logger.error(
            "All retries exhausted for task_id=%s", task_id,
        )
        self.send_error_notification(task_id, error)
        return False

    def _is_retryable(self, error: Exception) -> bool:
        """Check if an error is transient (retryable)."""
        retryable_types = (
            ConnectionError,
            TimeoutError,
            OSError,
            RetryableError,
        )
        return isinstance(error, retryable_types)

    def send_error_notification(
        self,
        task_id: str,
        error: Exception,
        retry_count: int = 0,
    ):
        """Send an error notification to the Matrix room."""
        message = f"❌ Delivery failed for {task_id}\n"
        message += f"Error: {error}\n"
        if retry_count > 0:
            message += f"Retries: {retry_count}\n"
        message += "Please check the worker logs."

        try:
            self.matrix_client.send_message(
                room_id=self.room_id,
                message=message,
            )
        except Exception as e:
            logger.error("Failed to send error notification: %s", e)
```

2. **Написать unit-тесты**:
```python
# tests/unit/test_result_errors.py
import pytest
import time
from bot.result_errors import (
    ResultErrorHandler, DeliveryError, RetryableError, PermanentError,
)

class TestResultErrorHandler:
    @pytest.fixture
    def handler(self):
        mock_client = mocker.Mock()
        mock_client.send_message.return_value = None
        return ResultErrorHandler(
            matrix_client=mock_client,
            room_id="!room:server",
            max_retries=3,
            base_delay=0.01,  # Fast for tests
        )

    def test_handle_delivery_error_retryable_success(self, handler):
        call_count = 0
        def delivery_fn():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise ConnectionError("temp")
        result = handler.handle_delivery_error("t-001", ConnectionError("temp"), delivery_fn)
        assert result is True

    def test_handle_delivery_error_retryable_exhausted(self, handler):
        def delivery_fn():
            raise ConnectionError("always fails")
        result = handler.handle_delivery_error("t-001", ConnectionError("always"), delivery_fn)
        assert result is False

    def test_handle_delivery_error_permanent(self, handler):
        result = handler.handle_delivery_error(
            "t-001", ValueError("permanent"),
        )
        assert result is False

    def test_is_retryable_connection_error(self, handler):
        assert handler._is_retryable(ConnectionError()) is True

    def test_is_retryable_timeout_error(self, handler):
        assert handler._is_retryable(TimeoutError()) is True

    def test_is_retryable_os_error(self, handler):
        assert handler._is_retryable(OSError()) is True

    def test_is_retryable_retryable_error(self, handler):
        assert handler._is_retryable(RetryableError("t-001", "retry")) is True

    def test_is_retryable_value_error(self, handler):
        assert handler._is_retryable(ValueError()) is False

    def test_is_retryable_runtime_error(self, handler):
        assert handler._is_retryable(RuntimeError()) is False

    def test_is_retryable_permanent_error(self, handler):
        assert handler._is_retryable(PermanentError("t-001", "perm")) is False

    def test_send_error_notification(self, handler):
        handler.send_error_notification("t-001", ValueError("test"))
        handler.matrix_client.send_message.assert_called_once()
        call_args = handler.matrix_client.send_message.call_args
        assert "t-001" in call_args[1]["message"]
        assert "test" in call_args[1]["message"]

    def test_send_error_notification_with_retry_count(self, handler):
        handler.send_error_notification("t-001", ValueError("test"), retry_count=3)
        call_args = handler.matrix_client.send_message.call_args
        assert "Retries: 3" in call_args[1]["message"]

    def test_send_error_notification_fails(self, handler):
        handler.matrix_client.send_message.side_effect = Exception("notification failed")
        handler.send_error_notification("t-001", ValueError("test"))
        # Should not raise — logged instead

    def test_handle_delivery_error_no_delivery_fn(self, handler):
        result = handler.handle_delivery_error(
            "t-001", ConnectionError("no fn"),
        )
        assert result is False

    def test_handler_defaults(self):
        mock_client = mocker.Mock()
        h = ResultErrorHandler(matrix_client=mock_client, room_id="!r:s")
        assert h.max_retries == 3
        assert h.base_delay == 1.0

    def test_retry_backoff_timing(self, handler):
        """Verify retries happen with increasing delays."""
        start = time.monotonic()
        call_count = 0
        def delivery_fn():
            nonlocal call_count
            call_count += 1
            if call_count < 3:
                raise ConnectionError("fail")
        handler.base_delay = 0.01
        result = handler.handle_delivery_error("t-001", ConnectionError(), delivery_fn)
        elapsed = time.monotonic() - start
        assert result is False
        assert elapsed >= 0.03  # At least 2 delays: 0.01 + 0.02
```

## Граничные случаи
- **Retryable error, 1-я попытка succeeds**: возвращает True
- **Retryable error, все попытки fail**: возвращает False, отправляет уведомление
- **Permanent error**: сразу False, без retry
- **Delivery function не передан**: сразу False
- **Notification send fails**: логируется, не raises
- **base_delay=0.01** для тестов (быстрее чем 1.0)
