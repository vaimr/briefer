# T5.2.1 — Create Bot Result Consumer

## Цель
Создать `ResultConsumer` — обработчик результатов из Redis pub/sub, который отправляет результат в Matrix-комнату.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — ID задачи; `result: dict` — результат из Redis {text, segments, duration_ms, status}; `room_id: str` — Matrix room ID |
| **Обработка** | 1. Проверка статуса результата. 2. Если success → генерация PDF → отправка. 3. Если failed → отправка уведомления об ошибке. |
| **Выход** | Сообщение в Matrix-комнату: текст транскрипции или PDF с результатами |

## Пошаговый план

1. **Создать `bot/results_consumer.py`**:
```python
import logging
from typing import Any, Optional

logger = logging.getLogger(__name__)

class ResultConsumer:
    """Consume transcription results from Redis and deliver to Matrix."""

    def __init__(
        self,
        matrix_client,
        room_id: str,
        redis_channel: str = "transcription_results",
    ):
        self.matrix_client = matrix_client
        self.room_id = room_id
        self.redis_channel = redis_channel
        self._subscriber = None

    def consume(self, task_id: str, result: dict[str, Any]):
        """Process a transcription result and deliver to Matrix.

        Args:
            task_id: Task identifier.
            result: Result dict with status, text, segments, etc.
        """
        status = result.get("status", "unknown")
        logger.info("Consuming result for task_id=%s status=%s", task_id, status)

        if status == "success":
            self._handle_success(task_id, result)
        elif status == "failed":
            self._handle_failure(task_id, result)
        else:
            logger.warning("Unknown status '%s' for task_id=%s", status, task_id)
            self._send_notification(
                f"⚠️ Task {task_id}: Unknown status '{status}'"
            )

    def _handle_success(self, task_id: str, result: dict[str, Any]):
        """Handle a successful transcription result."""
        text = result.get("text", "")
        duration_ms = result.get("duration_ms", 0)

        # Generate PDF if text is long enough
        if len(text) > 500:
            try:
                pdf_path = self._generate_pdf(task_id, text)
                self._send_pdf(pdf_path)
                self._send_notification(f"✅ Transcription ready for {task_id} (PDF attached)")
            except Exception as e:
                logger.error("Failed to generate PDF for %s: %s", task_id, e)
                self._send_notification(f"❌ PDF generation failed for {task_id}: {e}")
                self._send_notification(text[:1000])  # Fallback: send text
        else:
            # Short text: send as message
            self._send_notification(f"📝 Transcription for {task_id}:\n{text[:2000]}")

    def _handle_failure(self, task_id: str, result: dict[str, Any]):
        """Handle a failed transcription result."""
        error_msg = result.get("error_msg", "Unknown error")
        self._send_notification(
            f"❌ Transcription failed for {task_id}\nError: {error_msg}"
        )

    def _generate_pdf(self, task_id: str, text: str) -> str:
        """Generate a PDF from transcription text."""
        from worker.pdf_generator import PDFGenerator
        generator = PDFGenerator()
        return generator.generate(task_id, text)

    def _send_pdf(self, file_path: str):
        """Send a PDF file to the Matrix room."""
        self.matrix_client.upload_media(
            file_path,
            room_id=self.room_id,
            message=f"Transcription result",
        )

    def _send_notification(self, text: str):
        """Send a text notification to the Matrix room."""
        self.matrix_client.send_message(
            room_id=self.room_id,
            message=text,
        )
```

```python
# tests/unit/test_results_consumer.py
import pytest
from worker.result_consumer import ResultConsumer

class TestResultConsumer:
    @pytest.fixture
    def mock_client(self):
        return {
            "send_message": mocker.Mock(),
            "upload_media": mocker.Mock(),
        }

    @pytest.fixture
    def consumer(self):
        return ResultConsumer(
            matrix_client=mocker.Mock(),
            room_id="!test:room",
        )

    def test_consume_success_short_text(self, consumer, mocker):
        mocker.patch.object(consumer, '_send_notification')
        result = {"status": "success", "text": "Hello world", "duration_ms": 1000}
        consumer.consume("task-001", result)
        consumer._send_notification.assert_called_once()
        call_args = consumer._send_notification.call_args[0][0]
        assert "Transcription" in call_args
        assert "Hello world" in call_args

    def test_consume_success_long_text(self, consumer, mocker):
        mocker.patch.object(consumer, '_send_notification')
        mocker.patch.object(consumer, '_generate_pdf', return_value="/tmp/test.pdf")
        long_text = "word " * 200  # > 500 chars
        result = {"status": "success", "text": long_text, "duration_ms": 5000}
        consumer.consume("task-001", result)
        consumer._generate_pdf.assert_called_once_with("task-001", long_text)
        consumer._send_pdf.assert_called_once_with("/tmp/test.pdf")

    def test_consume_failure(self, consumer, mocker):
        mocker.patch.object(consumer, '_send_notification')
        result = {"status": "failed", "error_msg": "Whisper error"}
        consumer.consume("task-001", result)
        consumer._send_notification.assert_called_once()
        call_args = consumer._send_notification.call_args[0][0]
        assert "failed" in call_args.lower()
        assert "Whisper error" in call_args

    def test_consume_unknown_status(self, consumer, mocker):
        mocker.patch.object(consumer, '_send_notification')
        result = {"status": "unknown_status"}
        consumer.consume("task-001", result)
        consumer._send_notification.assert_called_once()
        call_args = consumer._send_notification.call_args[0][0]
        assert "Unknown status" in call_args

    def test_consume_success_no_text(self, consumer, mocker):
        mocker.patch.object(consumer, '_send_notification')
        result = {"status": "success", "text": "", "duration_ms": 0}
        consumer.consume("task-001", result)
        consumer._send_notification.assert_called_once()

    def test_consume_success_pdf_generation_fails(self, consumer, mocker):
        mocker.patch.object(consumer, '_generate_pdf', side_effect=RuntimeError("PDF error"))
        long_text = "word " * 200
        result = {"status": "success", "text": long_text}
        consumer.consume("task-001", result)
        # Should fallback to sending text
        consumer._send_notification.assert_called()

    def test_consume_empty_result(self, consumer, mocker):
        mocker.patch.object(consumer, '_send_notification')
        consumer.consume("task-001", {})
        consumer._send_notification.assert_called_once()

    def test_generate_pdf(self, consumer, mocker):
        mock_generator = mocker.Mock(return_value="/tmp/out.pdf")
        mocker.patch("bot.results_consumer.PDFGenerator", return_value=mock_generator)
        path = consumer._generate_pdf("task-001", "text")
        assert path == "/tmp/out.pdf"

    def test_send_notification(self, consumer, mocker):
        mock_client = mocker.Mock()
        consumer.matrix_client = mock_client
        consumer._send_notification("Hello")
        mock_client.send_message.assert_called_once_with(
            room_id="!test:room",
            message="Hello",
        )

    def test_send_pdf(self, consumer, mocker):
        mock_client = mocker.Mock()
        consumer.matrix_client = mock_client
        consumer._send_pdf("/tmp/test.pdf")
        mock_client.upload_media.assert_called_once_with(
            "/tmp/test.pdf",
            room_id="!test:room",
            message="Transcription result",
        )

    def test_consumer_initialization(self):
        mock_client = mocker.Mock()
        consumer = ResultConsumer(
            matrix_client=mock_client,
            room_id="!room:server",
            redis_channel="custom_channel",
        )
        assert consumer.room_id == "!room:server"
        assert consumer.redis_channel == "custom_channel"
        assert consumer.matrix_client is mock_client
```

## Граничные случаи
- **Пустой result dict**: обрабатывается как failure (status="unknown")
- **Короткий текст** (< 500 символов): отправляется как сообщение
- **Длинный текст** (≥ 500 символов): генерируется PDF
- **PDF generation failed**: fallback — отправка текста
- **Неизвестный status**: уведомление с предупреждением
- **Пустой текст при success**: отправляется пустое сообщение
