# T5.1.1 — Create Redis Pub/Sub Result Publisher

## Цель
Реализовать публикацию результатов транскрипции в Redis pub/sub канал для уведомления бота о готовности результата.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — ID задачи; `result: dict` — результат транскрипции {text, segments, duration_ms, status}; `channel: str` — имя pub/sub канала; `redis_url: str` — URL Redis (по умолчанию redis://localhost:6379) |
| **Обработка** | 1. Сериализация result в JSON. 2. Отправка через Redis pub/sub publish. 3. Обработка ошибок подключения. |
| **Выход** | `int` — количество подписчиков, получивших сообщение; `0` при ошибке |

## Пошаговый план

1. **Создать `worker/result_publisher.py`**:
```python
import json
import logging
from typing import Any, Optional

logger = logging.getLogger(__name__)

class ResultPublisher:
    """Publish transcription results via Redis pub/sub."""

    def __init__(
        self,
        redis_url: str = "redis://localhost:6379/0",
        channel: str = "transcription_results",
    ):
        self.redis_url = redis_url
        self.channel = channel
        self._redis = None

    def _get_redis(self):
        """Lazy-connect to Redis."""
        if self._redis is None:
            import redis
            self._redis = redis.from_url(self.redis_url, decode_responses=True)
        return self._redis

    def publish(self, task_id: str, result: dict[str, Any]) -> int:
        """Publish a transcription result to the pub/sub channel.

        Args:
            task_id: Unique task identifier.
            result: Result dict with text, segments, duration_ms, status.

        Returns:
            Number of subscribers that received the message.
        """
        message = {
            "task_id": task_id,
            "result": result,
            "timestamp": __import__("datetime").datetime.now().isoformat(),
        }
        try:
            r = self._get_redis()
            message_json = json.dumps(message, ensure_ascii=False)
            return r.publish(self.channel, message_json)
        except Exception as e:
            logger.error("Failed to publish result for task_id=%s: %s", task_id, e)
            return 0

    def publish_error(self, task_id: str, error_msg: str) -> int:
        """Publish an error message to the pub/sub channel.

        Args:
            task_id: Unique task identifier.
            error_msg: Error message string.

        Returns:
            Number of subscribers that received the message.
        """
        result = {
            "status": "failed",
            "error_msg": error_msg,
            "text": "",
            "segments": [],
            "duration_ms": 0,
        }
        return self.publish(task_id, result)

    def get_channel(self) -> str:
        """Return the pub/sub channel name."""
        return self.channel

    def close(self):
        """Close Redis connection."""
        if self._redis:
            self._redis.close()
            self._redis = None
```

2. **Написать unit-тесты**:
```python
# tests/unit/test_result_publisher.py
import pytest
import json
from worker.result_publisher import ResultPublisher

class TestResultPublisher:
    @pytest.fixture
    def publisher(self):
        return ResultPublisher(
            redis_url="redis://localhost:6379/0",
            channel="test_results",
        )

    def test_get_channel(self, publisher):
        assert publisher.get_channel() == "test_results"

    def test_publish_success(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        result = publisher.publish("task-001", {"text": "Hello", "status": "success"})
        assert result == 1

    def test_publish_sends_json(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        publisher.publish("task-001", {"text": "Hello", "status": "success"})
        call_args = mock_redis.publish.call_args
        message = json.loads(call_args[0][1])
        assert message["task_id"] == "task-001"
        assert message["result"]["text"] == "Hello"
        assert "timestamp" in message

    def test_publish_error(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.side_effect = ConnectionError("Redis down")
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        result = publisher.publish("task-001", {"text": "Hello"})
        assert result == 0

    def test_publish_error_message(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        result = publisher.publish_error("task-001", "Whisper failed")
        assert result == 1

    def test_publish_error_contains_status_failed(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        publisher.publish_error("task-001", "Connection error")
        call_args = mock_redis.publish.call_args
        message = json.loads(call_args[0][1])
        assert message["result"]["status"] == "failed"
        assert message["result"]["error_msg"] == "Connection error"

    def test_publish_empty_text_on_error(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        publisher.publish_error("task-001", "error")
        call_args = mock_redis.publish.call_args
        message = json.loads(call_args[0][1])
        assert message["result"]["text"] == ""
        assert message["result"]["segments"] == []
        assert message["result"]["duration_ms"] == 0

    def test_close_no_redis(self, publisher):
        publisher.close()  # Should not raise

    def test_close_with_redis(self, publisher, mocker):
        mock_redis = mocker.Mock()
        publisher._redis = mock_redis
        publisher.close()
        mock_redis.close.assert_called_once()
        assert publisher._redis is None

    def test_default_channel(self):
        p = ResultPublisher()
        assert p.get_channel() == "transcription_results"

    def test_default_redis_url(self):
        p = ResultPublisher()
        assert p.redis_url == "redis://localhost:6379/0"

    def test_publish_utf8_text(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        publisher.publish("task-001", {"text": "Привет мир! 你好"})
        call_args = mock_redis.publish.call_args
        message = json.loads(call_args[0][1])
        assert message["result"]["text"] == "Привет мир! 你好"

    def test_publish_segments_included(self, publisher, mocker):
        mock_redis = mocker.Mock()
        mock_redis.publish.return_value = 1
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        segments = [{"id": 0, "start": 0, "end": 1.0, "text": "Hello"}]
        publisher.publish("task-001", {"text": "Hello", "segments": segments, "duration_ms": 1000})
        call_args = mock_redis.publish.call_args
        message = json.loads(call_args[0][1])
        assert message["result"]["segments"] == segments
        assert message["result"]["duration_ms"] == 1000

    def test_publish_no_connection(self, publisher, mocker):
        """When Redis is not connected yet, _get_redis creates connection."""
        mock_redis = mocker.Mock()
        mock_redis.publish.side_effect = ConnectionError("not connected")
        mocker.patch.object(publisher, '_get_redis', return_value=mock_redis)
        result = publisher.publish("task-001", {"text": "test"})
        assert result == 0
```

## Граничные случаи
- **Redis недоступен**: возвращает `0`, логирует ошибку
- **Пустой result dict**: отправляется с пустыми полями
- **None result**: raises TypeError (не обрабатывается — caller обязан передать dict)
- **UTF-8 текст**: корректно сериализуется
- **Повторная публикация**: перезаписывает состояние
- **Multiple publish calls**: каждый вызов — отдельное сообщение
