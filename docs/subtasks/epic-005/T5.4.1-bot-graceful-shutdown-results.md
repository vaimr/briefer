# T5.4.1 — Add Bot Graceful Shutdown for Results

## Цель
Реализовать graceful shutdown consumer'а результатов: корректное завершение Redis pub/sub listener и освобождение ресурсов.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `signal: int` — сигнал ОС (SIGTERM, SIGINT); `frame` — current stack frame |
| **Обработка** | 1. Регистрация signal handler. 2. При получении сигнала: stop subscriber → cleanup Redis → exit. 3. Ожидание завершения текущих операций. |
| **Выход** | Process exits cleanly with code 0 |

## Пошаговый план

1. **Создать `bot/results_shutdown.py`**:
```python
import asyncio
import logging
import signal
import sys
from typing import Optional, Callable

logger = logging.getLogger(__name__)

class ResultsShutdownHandler:
    """Handle graceful shutdown of the results consumer."""

    def __init__(
        self,
        subscriber=None,
        cleanup_callbacks: Optional[list[Callable]] = None,
    ):
        self.subscriber = subscriber
        self.cleanup_callbacks = cleanup_callbacks or []
        self._shutdown_event = asyncio.Event()

    def register_signal_handlers(self):
        """Register SIGTERM and SIGINT handlers."""
        for sig in (signal.SIGTERM, signal.SIGINT):
            signal.signal(sig, self._signal_handler)
        logger.info("Registered signal handlers for SIGTERM, SIGINT")

    def _signal_handler(self, signum, frame):
        """Handle OS signal."""
        sig_name = signal.Signals(signum).name
        logger.info("Received %s, initiating graceful shutdown...", sig_name)
        asyncio.get_event_loop().create_task(self.shutdown())

    async def shutdown(self):
        """Execute graceful shutdown sequence."""
        if self._shutdown_event.is_set():
            return  # Already shutting down
        self._shutdown_event.set()

        logger.info("Starting graceful shutdown...")

        # Stop subscriber
        if self.subscriber:
            self.subscriber.stop()
            logger.info("Subscriber stopped")

        # Run cleanup callbacks
        for cb in self.cleanup_callbacks:
            try:
                if asyncio.iscoroutinefunction(cb):
                    await cb()
                else:
                    cb()
                logger.info("Cleanup callback executed")
            except Exception as e:
                logger.error("Cleanup callback failed: %s", e)

        logger.info("Graceful shutdown complete")

    def add_cleanup_callback(self, callback: Callable):
        """Add a cleanup callback."""
        self.cleanup_callbacks.append(callback)

    def is_shutting_down(self) -> bool:
        """Check if shutdown is in progress."""
        return self._shutdown_event.is_set()
```

2. **Написать unit-тесты**:
```python
# tests/unit/test_results_shutdown.py
import pytest
import asyncio
import signal
from bot.results_shutdown import ResultsShutdownHandler

class TestResultsShutdownHandler:
    @pytest.fixture
    def handler(self):
        return ResultsShutdownHandler()

    def test_is_shutting_down_false(self, handler):
        assert handler.is_shutting_down() is False

    def test_is_shutting_down_true(self, handler):
        handler._shutdown_event.set()
        assert handler.is_shutting_down() is True

    def test_shutdown_stops_subscriber(self, handler, mocker):
        mock_sub = mocker.Mock()
        handler.subscriber = mock_sub
        asyncio.run(handler.shutdown())
        mock_sub.stop.assert_called_once()

    def test_shutdown_no_subscriber(self, handler):
        asyncio.run(handler.shutdown())  # Should not raise

    def test_shutdown_run_cleanup_callbacks(self, handler, mocker):
        mock_cb = mocker.Mock()
        handler.add_cleanup_callback(mock_cb)
        asyncio.run(handler.shutdown())
        mock_cb.assert_called_once()

    def test_shutdown_async_cleanup_callback(self, handler, mocker):
        async def async_cb():
            pass
        handler.add_cleanup_callback(async_cb)
        asyncio.run(handler.shutdown())  # Should not raise

    def test_shutdown_callback_exception(self, handler, mocker):
        def failing_cb():
            raise ValueError("cleanup failed")
        handler.add_cleanup_callback(failing_cb)
        asyncio.run(handler.shutdown())  # Should not raise

    def test_shutdown_idempotent(self, handler, mocker):
        mock_sub = mocker.Mock()
        handler.subscriber = mock_sub
        asyncio.run(handler.shutdown())
        asyncio.run(handler.shutdown())
        mock_sub.stop.assert_called_once()  # Called only once

    def test_add_cleanup_callback(self, handler):
        cb = mocker.Mock()
        handler.add_cleanup_callback(cb)
        assert cb in handler.cleanup_callbacks

    def test_default_cleanup_callbacks_empty(self):
        h = ResultsShutdownHandler()
        assert h.cleanup_callbacks == []

    def test_register_signal_handlers(self, handler):
        # Should not raise
        handler.register_signal_handlers()

    def test_shutdown_event_prevents_double_shutdown(self, handler):
        """Second shutdown is no-op due to event flag."""
        asyncio.run(handler.shutdown())
        asyncio.run(handler.shutdown())
        # No exception, idempotent
```

## Граничные случаи
- **Двойной shutdown**: idempotent, second call is no-op
- **Нет subscriber**: не raises
- **Cleanup callback raises exception**: logs, не ломает shutdown
- **Async cleanup callback**: await'ится корректно
- **Signal во время shutdown**: игнорируется (event set)
- **Пустой cleanup_callbacks**: не raises
