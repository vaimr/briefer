# T5.1.2 — Create Redis Pub/Sub Result Subscriber

## Цель
Реализовать async-подписку на Redis pub/sub канал для получения результатов транскрипции в реальном времени.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `callback: Callable[[str, dict], None]` — функция-обработчик; `channel: str` — имя канала; `redis_url: str` — URL Redis |
| **Обработка** | 1. Подписка на канал через Redis pub/sub. 2. Async-слушание сообщений. 3. Парсинг JSON. 4. Вызов callback с task_id и result. 5. Обработка ошибок парсинга. |
| **Выход** | Async-генератор или поток, вызывающий callback для каждого сообщения |

## Пошаговый план

1. **Создать `worker/result_subscriber.py`**:
```python
import json
import logging
import asyncio
from typing import Any, Callable, Optional

logger = logging.getLogger(__name__)

class ResultSubscriber:
    """Subscribe to transcription results via Redis pub/sub."""

    def __init__(
        self,
        redis_url: str = "redis://localhost:6379/0",
        channel: str = "transcription_results",
    ):
        self.redis_url = redis_url
        self.channel = channel
        self._callback: Optional[Callable[[str, dict], None]] = None
        self._pubsub = None
        self._redis = None
        self._running = False

    def subscribe(self, callback: Callable[[str, dict], None]):
        """Subscribe to the results channel.

        Args:
            callback: Function(task_id, result) called for each message.
        """
        self._callback = callback

    def _get_redis(self):
        """Lazy-connect to Redis."""
        if self._redis is None:
            import redis
            self._redis = redis.from_url(self.redis_url, decode_responses=True)
        return self._redis

    async def listen(self):
        """Start listening for messages. Runs until stop() is called."""
        if self._callback is None:
            raise RuntimeError("No callback set. Call subscribe() first.")

        r = self._get_redis()
        self._pubsub = r.pubsub()
        self._pubsub.subscribe(self.channel)
        self._running = True

        logger.info("Subscriber listening on channel=%s", self.channel)

        try:
            while self._running:
                message = self._pubsub.get_message(ignore_subscribe_messages=True, timeout=1.0)
                if message and message['type'] == 'message':
                    await self._handle_message(message['data'])
        except Exception as e:
            logger.error("Subscriber error: %s", e, exc_info=True)
        finally:
            self._running = False
            self._cleanup()

    async def _handle_message(self, raw_data: str):
        """Parse and dispatch a message."""
        try:
            data = json.loads(raw_data)
            task_id = data.get("task_id", "unknown")
            result = data.get("result", {})
            logger.debug("Received result for task_id=%s status=%s", task_id, result.get("status"))
            self._callback(task_id, result)
        except json.JSONDecodeError as e:
            logger.error("Failed to parse message: %s", e)
        except Exception as e:
            logger.error("Error processing message: %s", e)

    def stop(self):
        """Stop the subscriber."""
        self._running = False

    def _cleanup(self):
        """Clean up Redis resources."""
        if self._pubsub:
            try:
                self._pubsub.unsubscribe()
                self._pubsub.close()
            except Exception:
                pass
            self._pubsub = None
        if self._redis:
            try:
                self._redis.close()
            except Exception:
                pass
            self._redis = None
```

```python
# tests/unit/test_result_subscriber.py
import pytest
import asyncio
import json
from worker.result_subscriber import ResultSubscriber

class TestResultSubscriber:
    @pytest.fixture
    def subscriber(self):
        return ResultSubscriber(
            redis_url="redis://localhost:6379/0",
            channel="test_channel",
        )

    def test_subscribe_sets_callback(self, subscriber):
        def cb(tid, r):
            pass
        subscriber.subscribe(cb)
        assert subscriber._callback is not None

    def test_subscribe_without_callback_raises(self, subscriber):
        subscriber.subscribe(lambda t, r: None)
        subscriber._callback = None  # Clear it
        with pytest.raises(RuntimeError, match="No callback set"):
            asyncio.run(subscriber.listen())

    def test_handle_message_valid_json(self, subscriber, mocker):
        mock_callback = mocker.Mock()
        subscriber._callback = mock_callback
        msg = json.dumps({"task_id": "t-001", "result": {"status": "success", "text": "Hello"}})
        asyncio.run(subscriber._handle_message(msg))
        mock_callback.assert_called_once_with("t-001", {"status": "success", "text": "Hello"})

    def test_handle_message_invalid_json(self, subscriber, mocker):
        mock_callback = mocker.Mock()
        subscriber._callback = mock_callback
        asyncio.run(subscriber._handle_message("{invalid"))
        mock_callback.assert_not_called()

    def test_handle_message_callback_exception(self, subscriber, mocker):
        mock_callback = mocker.Mock(side_effect=ValueError("callback error"))
        subscriber._callback = mock_callback
        msg = json.dumps({"task_id": "t-001", "result": {"status": "success"}})
        asyncio.run(subscriber._handle_message(msg))
        mock_callback.assert_called_once()  # Called but exception caught

    def test_handle_message_missing_task_id(self, subscriber, mocker):
        mock_callback = mocker.Mock()
        subscriber._callback = mock_callback
        msg = json.dumps({"result": {"status": "success"}})
        asyncio.run(subscriber._handle_message(msg))
        mock_callback.assert_called_once_with("unknown", {"status": "success"})

    def test_handle_message_missing_result(self, subscriber, mocker):
        mock_callback = mocker.Mock()
        subscriber._callback = mock_callback
        msg = json.dumps({"task_id": "t-001"})
        asyncio.run(subscriber._handle_message(msg))
        mock_callback.assert_called_once_with("t-001", {})

    def test_stop_sets_running_false(self, subscriber):
        subscriber._running = True
        subscriber.stop()
        assert subscriber._running is False

    def test_cleanup_no_pubsub(self, subscriber):
        subscriber._cleanup()  # Should not raise

    def test_cleanup_with_pubsub(self, subscriber, mocker):
        mock_pubsub = mocker.Mock()
        subscriber._pubsub = mock_pubsub
        subscriber._cleanup()
        mock_pubsub.unsubscribe.assert_called_once()
        mock_pubsub.close.assert_called_once()
        assert subscriber._pubsub is None

    def test_cleanup_with_redis(self, subscriber, mocker):
        mock_redis = mocker.Mock()
        subscriber._redis = mock_redis
        subscriber._cleanup()
        mock_redis.close.assert_called_once()
        assert subscriber._redis is None

    def test_default_channel(self):
        s = ResultSubscriber()
        assert s.channel == "transcription_results"

    def test_default_redis_url(self):
        s = ResultSubscriber()
        assert s.redis_url == "redis://localhost:6379/0"

    def test_handle_message_utf8(self, subscriber, mocker):
        mock_callback = mocker.Mock()
        subscriber._callback = mock_callback
        msg = json.dumps({"task_id": "t-001", "result": {"text": "Привет"}})
        asyncio.run(subscriber._handle_message(msg))
        mock_callback.assert_called_once_with("t-001", {"text": "Привет"})

    def test_handle_message_empty_result(self, subscriber, mocker):
        mock_callback = mocker.Mock()
        subscriber._callback = mock_callback
        msg = json.dumps({"task_id": "t-001", "result": {}})
        asyncio.run(subscriber._handle_message(msg))
        mock_callback.assert_called_once_with("t-001", {})
```

## Граничные случаи
- **Callback не установлен**: raises RuntimeError при listen()
- **Invalid JSON**: логирует ошибку, не вызывает callback
- **Missing task_id**: использует "unknown"
- **Missing result**: использует пустой dict {}
- **Callback raises exception**: exception catch, не ломает listener
- **Redis disconnect**: exception catch, listener завершается
- **Повторный subscribe()**: перезаписывает callback
