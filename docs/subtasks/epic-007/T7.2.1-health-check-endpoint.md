# T7.2.1 — Health Check Endpoint

## Цель
Реализовать health check для bot и worker: проверка Redis, Whisper (worker), LLM (worker).

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `service: str` — "bot" или "worker"; `redis_client` — Redis-клиент (опционально); `whisper_model: str` — модель Whisper (только worker) |
| **Обработка** | 1. Проверка Redis через `ping()`. 2. Для worker — проверка загрузки Whisper модели. 3. Для worker — проверка LLM API ключа. 4. Сбор статусов в `HealthStatus`. |
| **Выход** | `HealthStatus` — объект с полями: status (healthy/degraded/unhealthy), checks (dict), timestamp |

## Пошаговый план

1. **Создать `bot/health.py`** (и `worker/health.py` как расширение):
```python
import logging
import time
from dataclasses import dataclass, field
from typing import Optional

logger = logging.getLogger(__name__)


@dataclass
class CheckResult:
    """Result of a health check."""
    name: str
    healthy: bool
    message: str = ""
    duration_ms: float = 0.0


@dataclass
class HealthStatus:
    """Overall health status."""
    status: str = "healthy"  # healthy, degraded, unhealthy
    checks: dict[str, CheckResult] = field(default_factory=dict)
    timestamp: float = field(default_factory=time.time)

    @property
    def is_healthy(self) -> bool:
        """Check if all checks are healthy."""
        return all(c.healthy for c in self.checks.values()) if self.checks else True


class HealthChecker:
    """Health checker for bot and worker services."""

    def __init__(self, service_name: str = "bot"):
        self.service_name = service_name
        self._redis_client = None

    def set_redis_client(self, redis_client) -> None:
        """Set the Redis client for health checks."""
        self._redis_client = redis_client

    async def check_redis(self) -> CheckResult:
        """Check Redis connectivity."""
        start = time.monotonic()
        try:
            if self._redis_client is None:
                return CheckResult(
                    name="redis",
                    healthy=False,
                    message="Redis client not configured",
                )
            await self._redis_client.ping()
            duration = (time.monotonic() - start) * 1000
            return CheckResult(
                name="redis",
                healthy=True,
                message="Connected",
                duration_ms=duration,
            )
        except Exception as e:
            duration = (time.monotonic() - start) * 1000
            return CheckResult(
                name="redis",
                healthy=False,
                message=f"Redis error: {e}",
                duration_ms=duration,
            )

    async def check_whisper(self) -> CheckResult:
        """Check Whisper model availability (worker only)."""
        return CheckResult(
            name="whisper",
            healthy=False,
            message="Not applicable for bot",
        )

    async def check_llm(self) -> CheckResult:
        """Check LLM API availability (worker only)."""
        return CheckResult(
            name="llm",
            healthy=False,
            message="Not applicable for bot",
        )

    async def health_check(self) -> HealthStatus:
        """Run all health checks."""
        checks = {
            "redis": await self.check_redis(),
            "whisper": await self.check_whisper(),
            "llm": await self.check_llm(),
        }

        healthy_count = sum(1 for c in checks.values() if c.healthy)
        total_count = len(checks)

        if healthy_count == total_count:
            status = "healthy"
        elif healthy_count > 0:
            status = "degraded"
        else:
            status = "unhealthy"

        return HealthStatus(
            status=status,
            checks=checks,
        )


class WorkerHealthChecker(HealthChecker):
    """Health checker for worker service with Whisper and LLM checks."""

    def __init__(self, whisper_model: Optional[str] = None):
        super().__init__(service_name="worker")
        self._whisper_model = whisper_model
        self._whisper_loaded = False
        self._llm_api_key = None

    def set_whisper_model(self, model_path: str) -> None:
        """Set Whisper model path."""
        self._whisper_model = model_path

    def set_llm_api_key(self, api_key: str) -> None:
        """Set LLM API key."""
        self._llm_api_key = api_key

    async def check_whisper(self) -> CheckResult:
        """Check Whisper model availability."""
        start = time.monotonic()
        if not self._whisper_model:
            return CheckResult(
                name="whisper",
                healthy=False,
                message="Whisper model not configured",
            )
        if not self._whisper_loaded:
            return CheckResult(
                name="whisper",
                healthy=False,
                message="Whisper model not loaded",
            )
        duration = (time.monotonic() - start) * 1000
        return CheckResult(
            name="whisper",
            healthy=True,
            message=f"Model: {self._whisper_model}",
            duration_ms=duration,
        )

    async def check_llm(self) -> CheckResult:
        """Check LLM API availability."""
        start = time.monotonic()
        if not self._llm_api_key:
            return CheckResult(
                name="llm",
                healthy=False,
                message="LLM API key not configured",
            )
        duration = (time.monotonic() - start) * 1000
        return CheckResult(
            name="llm",
            healthy=True,
            message="API key configured",
            duration_ms=duration,
        )
```

class TestCheckResult:
    def test_default_values(self):
        cr = CheckResult(name="test", healthy=True)
        assert cr.name == "test"
        assert cr.healthy is True
        assert cr.message == ""
        assert cr.duration_ms == 0.0

    def test_with_message(self):
        cr = CheckResult(name="test", healthy=False, message="error")
        assert cr.message == "error"

    def test_with_duration(self):
        cr = CheckResult(name="test", healthy=True, duration_ms=42.0)
        assert cr.duration_ms == 42.0


class TestHealthStatus:
    def test_all_healthy(self):
        checks = {
            "redis": CheckResult(name="redis", healthy=True),
            "whisper": CheckResult(name="whisper", healthy=True),
        }
        hs = HealthStatus(checks=checks)
        assert hs.status == "healthy"
        assert hs.is_healthy is True

    def test_all_unhealthy(self):
        checks = {
            "redis": CheckResult(name="redis", healthy=False, message="error"),
            "whisper": CheckResult(name="whisper", healthy=False, message="error"),
        }
        hs = HealthStatus(checks=checks)
        assert hs.status == "unhealthy"
        assert hs.is_healthy is False

    def test_degraded(self):
        checks = {
            "redis": CheckResult(name="redis", healthy=True),
            "whisper": CheckResult(name="whisper", healthy=False, message="error"),
        }
        hs = HealthStatus(checks=checks)
        assert hs.status == "degraded"
        assert hs.is_healthy is False

    def test_empty_checks_healthy(self):
        hs = HealthStatus()
        assert hs.is_healthy is True

    def test_timestamp_set(self):
        before = time.time()
        hs = HealthStatus()
        after = time.time()
        assert before <= hs.timestamp <= after
