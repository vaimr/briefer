# T7.1.1 — JSON Logger

## Цель
Реализовать JSON-форматтер для структурированного логирования. Единый JsonFormatter для bot и worker.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `record: logging.LogRecord` — стандартный LogRecord из logging модуля |
| **Обработка** | 1. Извлечь level, message, name, timestamp. 2. Добавить task_id/correlation_id из контекста. 3. Сериализовать в JSON-строку. |
| **Выход** | `str` — JSON-строка с полями: timestamp, level, logger, message, task_id, correlation_id |

## Пошаговый план

1. **Создать `bot/json_logger.py`** (и `worker/json_logger.py` как импортер):
```python
import json
import logging
import time
from typing import Any, Optional

# Context vars for task_id and correlation_id
import contextvars

TASK_ID_VAR = contextvars.ContextVar("task_id", default=None)
CORRELATION_ID_VAR = contextvars.ContextVar("correlation_id", default=None)


class JsonFormatter(logging.Formatter):
    """JSON log formatter with task_id and correlation_id support."""

    def __init__(
        self,
        extra_fields: Optional[list[str]] = None,
        include_stacktrace: bool = False,
    ):
        super().__init__()
        self.extra_fields = extra_fields or []
        self.include_stacktrace = include_stacktrace

    def format(self, record: logging.LogRecord) -> str:
        """Format a log record as JSON."""
        log_data: dict[str, Any] = {
            "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S", time.gmtime(record.created)),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "task_id": TASK_ID_VAR.get(),
            "correlation_id": CORRELATION_ID_VAR.get(),
        }

        # Add extra fields from record
        for field in self.extra_fields:
            if hasattr(record, field):
                log_data[field] = getattr(record, field)

        # Add exception info if present
        if record.exc_info and record.exc_info[0] is not None:
            log_data["exception"] = self.formatException(record.exc_info)

        # Add stacktrace if requested
        if self.include_stacktrace and record.stack_info:
            log_data["stack_info"] = record.stack_info

        return json.dumps(log_data, ensure_ascii=False)


def set_task_id(task_id: Optional[str]) -> None:
    """Set the current task_id in context."""
    TASK_ID_VAR.set(task_id)


def set_correlation_id(correlation_id: Optional[str]) -> None:
    """Set the current correlation_id in context."""
    CORRELATION_ID_VAR.set(correlation_id)


def get_task_id() -> Optional[str]:
    """Get the current task_id from context."""
    return TASK_ID_VAR.get()


def get_correlation_id() -> Optional[str]:
    """Get the current correlation_id from context."""
    return CORRELATION_ID_VAR.get()


def setup_json_logging(
    level: int = logging.INFO,
    extra_fields: Optional[list[str]] = None,
) -> None:
    """Configure the root logger with JSON formatter.

    Args:
        level: Logging level.
        extra_fields: Additional fields to include in JSON output.
    """
    handler = logging.StreamHandler()
    handler.setFormatter(JsonFormatter(extra_fields=extra_fields))
    handler.setLevel(level)

    root_logger = logging.getLogger()
    root_logger.addHandler(handler)
    root_logger.setLevel(level)
```

2. **Написать unit-тесты**:
```python
# tests/unit/test_json_logger.py
import pytest
import json
import logging
from bot.json_logger import (
    JsonFormatter, set_task_id, set_correlation_id,
    get_task_id, get_correlation_id, setup_json_logging,
)


class TestJsonFormatter:
    def test_basic_format(self):
        formatter = JsonFormatter()
        record = logging.LogRecord(
            name="test",
            level=logging.INFO,
            pathname="test.py",
            lineno=1,
            msg="Hello world",
            args=(),
            exc_info=None,
        )
        result = json.loads(formatter.format(record))
        assert result["level"] == "INFO"
        assert result["message"] == "Hello world"
        assert result["logger"] == "test"
        assert "timestamp" in result
        assert result["task_id"] is None
        assert result["correlation_id"] is None

    def test_with_task_id(self):
        formatter = JsonFormatter()
        set_task_id("t-001")
        record = logging.LogRecord(
            name="test", level=logging.WARNING,
            pathname="test.py", lineno=1,
            msg="Warning message", args=(), exc_info=None,
        )
        result = json.loads(formatter.format(record))
        assert result["task_id"] == "t-001"
        set_task_id(None)  # Cleanup

    def test_with_correlation_id(self):
        formatter = JsonFormatter()
        set_correlation_id("corr-001")
        record = logging.LogRecord(
            name="test", level=logging.DEBUG,
            pathname="test.py", lineno=1,
            msg="Debug message", args=(), exc_info=None,
        )
        result = json.loads(formatter.format(record))
        assert result["correlation_id"] == "corr-001"
        set_correlation_id(None)  # Cleanup

    def test_with_exception(self):
        formatter = JsonFormatter()
        try:
            raise ValueError("test error")
        except ValueError:
            record = logging.LogRecord(
                name="test", level=logging.ERROR,
                pathname="test.py", lineno=1,
                msg="Error occurred", args=(), exc_info=True,
            )
            result = json.loads(formatter.format(record))
            assert "exception" in result
            assert "test error" in result["exception"]

    def test_extra_fields(self):
        formatter = JsonFormatter(extra_fields=["module", "function"])
        record = logging.LogRecord(
            name="test", level=logging.INFO,
            pathname="test.py", lineno=1,
            msg="Test", args=(), exc_info=None,
        )
        record.module = "test_module"
        record.function = "test_func"
        result = json.loads(formatter.format(record))
        assert result["module"] == "test_module"
        assert result["function"] == "test_func"

    def test_output_is_valid_json(self):
        formatter = JsonFormatter()
        record = logging.LogRecord(
            name="test", level=logging.INFO,
            pathname="test.py", lineno=1,
            msg="Test", args=(), exc_info=None,
        )
        result = formatter.format(record)
        parsed = json.loads(result)
        assert isinstance(parsed, dict)

    def test_all_required_fields_present(self):
        formatter = JsonFormatter()
        record = logging.LogRecord(
            name="test", level=logging.INFO,
            pathname="test.py", lineno=1,
            msg="Test", args=(), exc_info=None,
        )
        result = json.loads(formatter.format(record))
        for field in ("timestamp", "level", "logger", "message", "task_id", "correlation_id"):
            assert field in result

    def test_different_levels(self):
        formatter = JsonFormatter()
        for level in (logging.DEBUG, logging.INFO, logging.WARNING, logging.ERROR, logging.CRITICAL):
            record = logging.LogRecord(
                name="test", level=level,
                pathname="test.py", lineno=1,
                msg="Test", args=(), exc_info=None,
            )
            result = json.loads(formatter.format(record))
            assert result["level"] == logging.getLevelName(level)


class TestContextVars:
    def test_set_and_get_task_id(self):
        set_task_id("t-001")
        assert get_task_id() == "t-001"
        set_task_id(None)

    def test_set_and_get_correlation_id(self):
        set_correlation_id("corr-001")
        assert get_correlation_id() == "corr-001"
        set_correlation_id(None)

    def test_default_is_none(self):
        assert get_task_id() is None
        assert get_correlation_id() is None

    def test_context_isolation(self):
        """Context vars should be isolated per context."""
        import contextvars
        task_id_var = contextvars.ContextVar("test_var", default=None)

        def set_and_get():
            task_id_var.set("ctx-1")
            return task_id_var.get()

        assert set_and_get() == "ctx-1"
        assert task_id_var.get() is None


class TestSetupJsonLogging:
    def test_setup_does_not_raise(self):
        setup_json_logging(level=logging.DEBUG)
        # Should not raise

    def test_setup_adds_handler(self):
        import logging
        logger = logging.getLogger()
        handler_count = len(logger.handlers)
        setup_json_logging(level=logging.DEBUG)
        # Handler added
        assert len(logger.handlers) >= handler_count
```

## Граничные случаи
- **Context vars по умолчанию**: None
- **Exception в log**: добавляется поле "exception"
- **Extra fields**: только присутствующие в LogRecord
- **Стандартный LogRecord**: без task_id/correlation_id → None в JSON
- **Установка None**: сбрасывает context var к default
