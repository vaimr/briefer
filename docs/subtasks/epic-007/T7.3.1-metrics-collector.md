# T7.3.1 — Metrics Collector

## Цель
Реализовать сбор метрик для мониторинга: счётчики, тайминги, gauge-значения.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `prefix: str` — префикс для имён метрик; `counter_name: str` — имя счётчика; `metric_name: str` — имя тайминга |
| **Обработка** | 1. `increment()` — инкремент счётчика. 2. `timing()` — запись длительности. 3. `gauge()` — установка gauge-значения. 4. `get_metrics()` — получение всех метрик. |
| **Выход** | `dict[str, Any]` — словарь со всеми метриками |

## Пошаговый план

1. **Создать `worker/metrics.py`**:
```python
import logging
import time
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Any, Optional


logger = logging.getLogger(__name__)


@dataclass
class MetricSnapshot:
    """Snapshot of a single metric."""
    name: str
    value: Any
    type: str  # counter, timing, gauge
    timestamp: float = field(default_factory=time.time)


class MetricsCollector:
    """In-memory metrics collector for bot and worker services."""

    def __init__(self, prefix: str = "briefer"):
        self.prefix = prefix
        self._counters: dict[str, int] = defaultdict(int)
        self._timings: dict[str, list[float]] = defaultdict(list)
        self._gauges: dict[str, float] = {}
        self._max_timings: int = 1000  # Keep last N timings per metric

    def increment(self, counter: str, value: int = 1) -> None:
        """Increment a counter metric.

        Args:
            counter: Name of the counter.
            value: Amount to increment by (default 1).
        """
        self._counters[counter] += value
        logger.debug("Counter incremented: %s += %d", counter, value)

    def timing(self, metric: str, duration: float) -> None:
        """Record a timing metric.

        Args:
            metric: Name of the timing metric.
            duration: Duration in seconds.
        """
        self._timings[metric].append(duration)
        # Trim to max_timings
        if len(self._timings[metric]) > self._max_timings:
            self._timings[metric] = self._timings[metric][-self._max_timings:]

    def gauge(self, metric: str, value: float) -> None:
        """Set a gauge metric.

        Args:
            metric: Name of the gauge metric.
            value: Current value.
        """
        self._gauges[metric] = value

    def get_counter(self, counter: str) -> int:
        """Get the current value of a counter."""
        return self._counters.get(counter, 0)

    def get_timing_stats(self, metric: str) -> dict[str, float]:
        """Get statistics for a timing metric.

        Returns:
            Dict with 'count', 'avg', 'min', 'max', 'p50', 'p95', 'p99'.
        """
        values = self._timings.get(metric, [])
        if not values:
            return {
                "count": 0,
                "avg": 0.0,
                "min": 0.0,
                "max": 0.0,
                "p50": 0.0,
                "p95": 0.0,
                "p99": 0.0,
            }

        sorted_vals = sorted(values)
        count = len(sorted_vals)
        avg = sum(sorted_vals) / count
        min_val = sorted_vals[0]
        max_val = sorted_vals[-1]

        def percentile(p):
            idx = int(len(sorted_vals) * p / 100)
            idx = min(idx, len(sorted_vals) - 1)
            return sorted_vals[idx]

        return {
            "count": count,
            "avg": avg,
            "min": min_val,
            "max": max_val,
            "p50": percentile(50),
            "p95": percentile(95),
            "p99": percentile(99),
        }

    def get_gauge(self, metric: str) -> Optional[float]:
        """Get the current value of a gauge."""
        return self._gauges.get(metric)

    def get_metrics(self) -> dict[str, Any]:
        """Get all metrics as a dictionary.

        Returns:
            Dict with 'counters', 'timings', 'gauges'.
        """
        return {
            "counters": dict(self._counters),
            "timings": {
                name: self.get_timing_stats(name)
                for name in self._timings
            },
            "gauges": dict(self._gauges),
            "prefix": self.prefix,
        }

    def reset(self) -> None:
        """Reset all metrics."""
        self._counters.clear()
        self._timings.clear()
        self._gauges.clear()
        logger.info("All metrics reset")

    def snapshot(self) -> list[MetricSnapshot]:
        """Get a snapshot of all metrics."""
        snapshots = []

        for name, value in self._counters.items():
            snapshots.append(MetricSnapshot(
                name=f"{self.prefix}.{name}",
                value=value,
                type="counter",
            ))

        for name, stats in self._timings.items():
            snapshots.append(MetricSnapshot(
                name=f"{self.prefix}.{name}.avg",
                value=stats["avg"],
                type="timing",
            ))

        for name, value in self._gauges.items():
            snapshots.append(MetricSnapshot(
                name=f"{self.prefix}.{name}",
                value=value,
                type="gauge",
            ))

        return snapshots


def metrics_decorator(collector: MetricsCollector):
    """Decorator to automatically record timing for a function.

    Usage:
        @metrics_decorator(collector)
        async def my_function():
            ...
    """
    def decorator(func):
        async def async_wrapper(*args, **kwargs):
            start = time.monotonic()
            try:
                result = await func(*args, **kwargs)
                return result
            finally:
                duration = time.monotonic() - start
                collector.timing(f"{func.__name__}.duration", duration)
                collector.increment(f"{func.__name__}.calls")

        def sync_wrapper(*args, **kwargs):
            start = time.monotonic()
            try:
                result = func(*args, **kwargs)
                return result
            finally:
                duration = time.monotonic() - start
                collector.timing(f"{func.__name__}.duration", duration)
                collector.increment(f"{func.__name__}.calls")

        import asyncio
        if asyncio.iscoroutinefunction(func):
            return async_wrapper
        return sync_wrapper

    return decorator
```

# tests/unit/test_metrics.py
import pytest
import asyncio
import time
from worker.metrics import MetricsCollector, MetricSnapshot, metrics_decorator


class TestMetricsCollector:
    @pytest.fixture
    def collector(self):
        return MetricsCollector(prefix="test")

    def test_increment(self, collector):
        collector.increment("transcription.calls")
        assert collector.get_counter("transcription.calls") == 1

    def test_increment_multiple(self, collector):
        collector.increment("transcription.calls", 5)
        assert collector.get_counter("transcription.calls") == 5

    def test_increment_default_value(self, collector):
        collector.increment("transcription.calls")
        assert collector.get_counter("transcription.calls") == 1

    def test_increment_different_counters(self, collector):
        collector.increment("transcription.calls")
        collector.increment("transcription.success")
        assert collector.get_counter("transcription.calls") == 1
        assert collector.get_counter("transcription.success") == 1

    def test_timing(self, collector):
        collector.timing("transcription.duration", 1.5)
        stats = collector.get_timing_stats("transcription.duration")
        assert stats["count"] == 1
        assert stats["avg"] == 1.5
        assert stats["min"] == 1.5
        assert stats["max"] == 1.5

    def test_timing_multiple(self, collector):
        collector.timing("transcription.duration", 1.0)
        collector.timing("transcription.duration", 2.0)
        collector.timing("transcription.duration", 3.0)
        stats = collector.get_timing_stats("transcription.duration")
        assert stats["count"] == 3
        assert stats["avg"] == 2.0
        assert stats["min"] == 1.0
        assert stats["max"] == 3.0

    def test_timing_empty(self, collector):
        stats = collector.get_timing_stats("nonexistent")
        assert stats["count"] == 0
        assert stats["avg"] == 0.0

    def test_gauge(self, collector):
        collector.gauge("transcription.pending", 5)
        assert collector.get_gauge("transcription.pending") == 5

    def test_gauge_overwrite(self, collector):
        collector.gauge("transcription.pending", 5)
        collector.gauge("transcription.pending", 10)
        assert collector.get_gauge("transcription.pending") == 10

    def test_gauge_not_set(self, collector):
        assert collector.get_gauge("nonexistent") is None

    def test_get_metrics(self, collector):
        collector.increment("calls", 3)
        collector.timing("duration", 1.5)
        collector.gauge("pending", 5)
        metrics = collector.get_metrics()
        assert metrics["prefix"] == "test"
        assert metrics["counters"]["calls"] == 3
        assert metrics["timings"]["duration"]["count"] == 1
        assert metrics["gauges"]["pending"] == 5

    def test_reset(self, collector):
        collector.increment("calls", 3)
        collector.timing("duration", 1.5)
        collector.gauge("pending", 5)
        collector.reset()
        assert collector.get_counter("calls") == 0
        assert collector.get_gauge("pending") is None
        stats = collector.get_timing_stats("duration")
        assert stats["count"] == 0

    def test_default_prefix(self):
        c = MetricsCollector()
        assert c.prefix == "briefer"

    def test_custom_prefix(self):
        c = MetricsCollector(prefix="mybot")
        assert c.prefix == "mybot"

    def test_max_timings_trim(self, collector):
        collector._max_timings = 5
        for i in range(10):
            collector.timing("duration", float(i))
        stats = collector.get_timing_stats("duration")
        assert stats["count"] == 5  # Only last 5

    def test_snapshot(self, collector):
        collector.increment("calls", 3)
        collector.timing("duration", 1.5)
        collector.gauge("pending", 5)
        snapshots = collector.snapshot()
        names = [s.name for s in snapshots]
        assert "test.calls" in names
        assert "test.duration.avg" in names
        assert "test.pending" in names

    def test_snapshot_types(self, collector):
        collector.increment("calls", 3)
        collector.timing("duration", 1.5)
        collector.gauge("pending", 5)
        snapshots = collector.snapshot()
        types = {s.name: s.type for s in snapshots}
        assert types["test.calls"] == "counter"
        assert types["test.duration.avg"] == "timing"
        assert types["test.pending"] == "gauge"

    def test_timing_percentiles(self, collector):
        for i in range(100):
            collector.timing("duration", float(i))
        stats = collector.get_timing_stats("duration")
        assert stats["p50"] == 50.0
        assert stats["p95"] == 95.0
        assert stats["p99"] == 99.0

    def test_counter_initial_value(self, collector):
        assert collector.get_counter("nonexistent") == 0


class TestMetricsDecorator:

        async def my_func():
            return "ok"

        await my_func()
        assert collector.get_counter("my_func.calls") == 1

    def test_decorator_sync_timing(self, collector):
        @metrics_decorator(collector)

        assert result == "ok"

    def test_decorator_sync_calls(self, collector):
        @metrics_decorator(collector)
        def my_func():
