# T3.3.1 — Create Task Processing Pipeline

## Цель
Создать `TranscriptionPipeline` — оркестратор полного цикла обработки задачи: download → convert → transcribe → save.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — уникальный ID задачи; `audio_path: str` — путь к загруженному аудиофайлу; `config: PipelineConfig` — параметры пайплайна (audio_dir, output_dir, temp_dir, vad_enabled) |
| **Обработка** | 1. Валидация входных данных. 2. Конвертация аудио (если нужно) → temp-файл. 3. VAD-детекция (опционально). 4. Транскрипция через Whisper. 5. Сохранение результата. 6. Очистка temp-файлов. |
| **Выход** | `PipelineResult` — dataclass с полями: task_id, transcription_text, segments[], duration_ms, status="success"/"failed", error_msg (если failed) |

## Пошаговый план

1. **Создать `worker/pipeline.py`** с классом `TranscriptionPipeline`:
```python
import os
import json
import shutil
import tempfile
import logging
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional

logger = logging.getLogger(__name__)

@dataclass
class PipelineResult:
    task_id: str
    transcription_text: str
    segments: list = field(default_factory=list)
    duration_ms: float = 0.0
    status: str = "pending"  # success | failed
    error_msg: Optional[str] = None
    timestamp: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())

@dataclass
class PipelineConfig:
    audio_dir: str = "/tmp/briefer/audio"
    output_dir: str = "/tmp/briefer/output"
    temp_dir: str = "/tmp/briefer/temp"
    vad_enabled: bool = False
    vad_threshold: float = 0.5
    chunk_size: int = 3000

class TranscriptionPipeline:
    """Orchestrates: download → convert → vad → transcribe → save → cleanup."""

    def __init__(self, config: Optional[PipelineConfig] = None):
        self.config = config or PipelineConfig()
        # Ensure directories exist
        for d in [self.config.audio_dir, self.config.output_dir, self.config.temp_dir]:
            Path(d).mkdir(parents=True, exist_ok=True)

    def process(self, task_id: str, audio_path: str) -> PipelineResult:
        """Execute full transcription pipeline for a task.

        Steps:
        1. Validate input audio_path exists.
        2. Convert audio to whisper-compatible format (16kHz mono WAV).
        3. Optional VAD silence removal.
        4. Transcribe with Whisper.
        5. Save results.
        6. Cleanup temp files.

        Args:
            task_id: Unique task identifier.
            audio_path: Path to the downloaded audio file.

        Returns:
            PipelineResult with transcription or error details.
        """
        logger.info("Pipeline started for task_id=%s audio=%s", task_id, audio_path)

        # Step 1: Validate
        if not os.path.isfile(audio_path):
            return PipelineResult(
                task_id=task_id,
                transcription_text="",
                segments=[],
                status="failed",
                error_msg=f"Audio file not found: {audio_path}",
            )

        temp_files = []
        try:
            # Step 2: Convert audio
            converted_path = self._convert_audio(audio_path, task_id)
            temp_files.append(converted_path)
            logger.info("Audio converted for task_id=%s", task_id)

            # Step 3: VAD (optional)
            vad_path = converted_path
            if self.config.vad_enabled:
                vad_path = self._apply_vad(converted_path, task_id)
                if vad_path != converted_path:
                    temp_files.append(vad_path)
                logger.info("VAD applied for task_id=%s", task_id)

            # Step 4: Transcribe
            result = self._transcribe(vad_path, task_id)
            logger.info("Transcription completed for task_id=%s", task_id)

            # Step 5: Save results
            self._save_result(task_id, result)
            logger.info("Results saved for task_id=%s", task_id)

            result.status = "success"
            return result

        except Exception as e:
            logger.error("Pipeline failed for task_id=%s: %s", task_id, e, exc_info=True)
            return PipelineResult(
                task_id=task_id,
                transcription_text="",
                segments=[],
                status="failed",
                error_msg=str(e),
            )
        finally:
            # Step 6: Cleanup
            self._cleanup_temp(temp_files)
            logger.info("Temp files cleaned for task_id=%s", task_id)

    def _convert_audio(self, audio_path: str, task_id: str) -> str:
        """Convert audio to 16kHz mono WAV for Whisper.

        If file is already 16kHz mono WAV, return original path.
        Otherwise, convert and return new temp path.
        """
        import subprocess

        # Check if already suitable
        if self._is_whisper_ready(audio_path):
            return audio_path

        # Convert using ffmpeg
        output_path = os.path.join(
            self.config.temp_dir,
            f"{task_id}_converted.wav"
        )
        cmd = [
            "ffmpeg", "-y",
            "-i", audio_path,
            "-ar", "16000",
            "-ac", "1",
            "-c:a", "pcm_s16le",
            output_path,
        ]
        result = subprocess.run(cmd, capture_output=True, text=True)
        if result.returncode != 0:
            raise RuntimeError(f"ffmpeg failed: {result.stderr}")
        return output_path

    def _is_whisper_ready(self, audio_path: str) -> bool:
        """Check if audio is already 16kHz mono WAV."""
        import soundfile as sf
        try:
            info = sf.info(audio_path)
            return info.samplerate == 16000 and info.channels == 1 and audio_path.endswith('.wav')
        except Exception:
            return False

    def _apply_vad(self, audio_path: str, task_id: str) -> str:
        """Apply VAD silence removal. Returns path to cleaned audio."""
        from worker.vad import VoiceActivityDetector, SilenceSegment
        detector = VoiceActivityDetector(threshold=self.config.vad_threshold)
        segments = detector.detect_silence(audio_path)
        if not segments:
            return audio_path  # No silence to remove

        output_path = os.path.join(
            self.config.temp_dir,
            f"{task_id}_vad.wav"
        )
        detector.remove_silence_segments(audio_path, segments, output_path)
        return output_path

    def _transcribe(self, audio_path: str, task_id: str) -> PipelineResult:
        """Transcribe audio using Whisper."""
        from worker.transcription import WhisperTranscriber
        transcriber = WhisperTranscriber()
        result = transcriber.transcribe(audio_path)
        return PipelineResult(
            task_id=task_id,
            transcription_text=result.get("text", ""),
            segments=result.get("segments", []),
            duration_ms=result.get("duration_ms", 0),
        )

    def _save_result(self, task_id: str, result: PipelineResult):
        """Save pipeline result as JSON."""
        output_file = os.path.join(
            self.config.output_dir,
            f"{task_id}.json"
        )
        data = {
            "task_id": result.task_id,
            "text": result.transcription_text,
            "segments": result.segments,
            "duration_ms": result.duration_ms,
            "status": result.status,
            "error_msg": result.error_msg,
            "timestamp": result.timestamp,
        }
        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def _cleanup_temp(self, temp_files: list[str]):
        """Remove temporary files."""
        for path in temp_files:
            try:
                if os.path.isfile(path):
                    os.remove(path)
                    logger.debug("Removed temp file: %s", path)
            except OSError:
                pass

    def cleanup_temp_files(self, older_than_hours: int = 1):
        """Clean all temp files older than N hours."""
        cutoff = datetime.now()
        removed = 0
        for filename in os.listdir(self.config.temp_dir):
            filepath = os.path.join(self.config.temp_dir, filename)
            if os.path.isfile(filepath):
                mtime = datetime.fromtimestamp(os.path.getmtime(filepath))
                if (cutoff - mtime).total_seconds() > older_than_hours * 3600:
                    os.remove(filepath)
                    removed += 1
        logger.info("Cleaned %d temp files older than %dh", removed, older_than_hours)
        return removed
```

## Граничные случаи
- **Неexistent аудиофайл**: возвращает `PipelineResult(status="failed", error_msg="not found")`
- **Аудио уже whisper-ready** (16kHz mono WAV): пропуск конвертации
- **FFmpeg падает**: raises `RuntimeError` → pipeline возвращает `failed`
- **VAD включён, но тишины нет**: возвращает оригинальный путь
- **Исключение в транскрипции**: catch → `failed` result, cleanup всё равно выполняется
- **Пустой segments list**: корректно сохраняется в JSON
- **Очень длинный файл**: обработка по частям (chunking) — см. T3.2.2
