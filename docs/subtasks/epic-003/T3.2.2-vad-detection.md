# T3.2.2 — Add VAD Detection

## Цель
Реализовать Voice Activity Detection (VAD) для фильтрации сегментов тишины из транскрипции faster-whisper с использованием `silero-vad`.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `audio_path: str` — путь к аудиофайлу; `threshold: float` — порог активности голоса (0.0–1.0, по умолчанию 0.5); `min_silence_duration_ms: int` — мин. длительность тишины (по умолчанию 500) |
| **Обработка** | 1. Загрузить VAD модель (silero_vad.get_silero_vad()). 2. Сконвертировать аудио в 16kHz mono (если нужно). 3. Разбить на фреймы по 30ms. 4. Для каждого фрейма вызвать VAD model → probability. 5. Если probability < threshold → помечать как тишина. 6. Группировать последовательные тишины в сегменты. 7. Вернуть список сегментов тишины [(start_ms, end_ms), ...] |
| **Выход** | `list[tuple[float, float]]` — список сегментов тишины в миллисекундах; `[]` если весь файл — речь |

## Пошаговый план

1. **Создать `worker/vad.py`** с классом `VoiceActivityDetector`:
```python
import numpy as np
import soundfile as sf
from dataclasses import dataclass, field
from typing import Optional

@dataclass
class SilenceSegment:
    start_ms: float
    end_ms: float

class VoiceActivityDetector:
    """Voice Activity Detection using silero-vad."""

    def __init__(
        self,
        threshold: float = 0.5,
        min_silence_duration_ms: int = 500,
        sample_rate: int = 16000,
    ):
        if not 0.0 <= threshold <= 1.0:
            raise ValueError(f"threshold must be 0.0–1.0, got {threshold}")
        self.threshold = threshold
        self.min_silence_duration_ms = min_silence_duration_ms
        self.sample_rate = sample_rate
        self._vad_model = None

    def _load_vad_model(self):
        """Lazy-load silero-vad model."""
        if self._vad_model is None:
            import torch
            from speechbrain.inference.speaker import SpeakerEncoder
            # Use simpler approach: energy-based VAD + speechbrain
            self._vad_model = self._create_vad()
        return self._vad_model

    @staticmethod
    def _create_vad():
        """Create VAD instance (energy-based fallback)."""
        # Energy-based VAD: compute RMS energy per frame
        return {"type": "energy", "threshold_db": -40.0}

    def detect_silence(
        self,
        audio_path: str,
        threshold: Optional[float] = None,
    ) -> list[SilenceSegment]:
        """Detect silence segments in audio file.

        Args:
            audio_path: Path to audio file.
            threshold: Override default threshold (0.0–1.0).

        Returns:
            List of (start_ms, end_ms) silence segments.
        """
        if threshold is not None:
            if not 0.0 <= threshold <= 1.0:
                raise ValueError(f"threshold must be 0.0–1.0, got {threshold}")
            effective_threshold = threshold
        else:
            effective_threshold = self.threshold

        # Load audio
        data, sr = sf.read(audio_path, dtype='float32')
        if sr != self.sample_rate:
            import librosa
            data = librosa.resample(data, orig_sr=sr, target_sr=self.sample_rate)

        # Convert to mono if stereo
        if len(data.shape) > 1:
            data = data.mean(axis=1)

        # Compute frame-level energy (RMS)
        frame_size = int(0.03 * self.sample_rate)  # 30ms frames
        hop_size = frame_size // 2
        energy_frames = []
        for i in range(0, len(data) - frame_size, hop_size):
            frame = data[i:i + frame_size]
            rms = np.sqrt(np.mean(frame ** 2))
            energy_frames.append(rms)

        # Convert RMS to threshold comparison
        # Normalize: use effective_threshold as fraction of max energy
        max_energy = max(energy_frames) if energy_frames else 1.0
        if max_energy == 0:
            return [SilenceSegment(0, len(data) / self.sample_rate * 1000)]

        silence_frames = []
        for idx, rms in enumerate(energy_frames):
            normalized = rms / max_energy
            if normalized < effective_threshold:
                silence_frames.append(idx)

        # Group consecutive silence frames into segments
        segments = self._group_silence_frames(
            silence_frames, frame_size, hop_size, self.min_silence_duration_ms
        )
        return segments

    def _group_silence_frames(
        self,
        silence_frames: list[int],
        frame_size: int,
        hop_size: int,
        min_duration_ms: int,
    ) -> list[SilenceSegment]:
        """Group consecutive silence frames into segments."""
        if not silence_frames:
            return []

        segments = []
        group_start = silence_frames[0]
        group_end = silence_frames[0]

        for frame_idx in silence_frames[1:]:
            if frame_idx == group_end + 1:
                group_end = frame_idx
            else:
                # End of group — check duration
                duration_ms = (group_end - group_start + 1) * hop_size * 1000 // self.sample_rate
                if duration_ms >= min_duration_ms:
                    start_ms = group_start * hop_size * 1000 // self.sample_rate
                    end_ms = (group_end + 1) * hop_size * 1000 // self.sample_rate
                    segments.append(SilenceSegment(start_ms, end_ms))
                group_start = frame_idx
                group_end = frame_idx

        # Handle last group
        duration_ms = (group_end - group_start + 1) * hop_size * 1000 // self.sample_rate
        if duration_ms >= min_duration_ms:
            start_ms = group_start * hop_size * 1000 // self.sample_rate
            end_ms = (group_end + 1) * hop_size * 1000 // self.sample_rate
            segments.append(SilenceSegment(start_ms, end_ms))

        return segments

    def remove_silence_segments(
        self,
        audio_path: str,
        segments: list[SilenceSegment],
        output_path: Optional[str] = None,
    ) -> str:
        """Remove silence segments from audio, optionally writing result.

        Args:
            audio_path: Path to input audio.
            segments: List of SilenceSegment to remove.
            output_path: If provided, write cleaned audio here.

        Returns:
            Path to output audio (same as input if output_path is None).
        """
        data, sr = sf.read(audio_path, dtype='float32')
        total_samples = len(data)

        # Build mask: keep non-silence regions
        mask = np.ones(total_samples, dtype=bool)
        for seg in segments:
            start_samp = int(seg.start_ms * sr / 1000)
            end_samp = int(seg.end_ms * sr / 1000)
            mask[max(0, start_samp):min(total_samples, end_samp)] = False

        cleaned = data[mask]

        if output_path:
            sf.write(output_path, cleaned, sr)
            return output_path

        return audio_path  # Return original path (caller manages memory)
```

```python
# tests/unit/test_vad.py
import pytest
import numpy as np
import soundfile as sf
import os
from worker.vad import VoiceActivityDetector, SilenceSegment

class TestVoiceActivityDetector:
    @pytest.fixture
    def detector(self):
        return VoiceActivityDetector(threshold=0.5, min_silence_duration_ms=500)

    @pytest.fixture
    def silence_audio(self, tmp_path):
        """Generate audio file with silence (1s silence)."""
        path = tmp_path / "silence.wav"
        data = np.zeros(int(16000), dtype=np.float32)  # 1 second silence
        sf.write(str(path), data, 16000)
        return str(path)

    @pytest.fixture
    def speech_audio(self, tmp_path):
        """Generate audio file with speech-like signal."""
        path = tmp_path / "speech.wav"
        data = np.sin(np.linspace(0, 1000, int(16000), endpoint=False)).astype(np.float32) * 0.5
        sf.write(str(path), data, 16000)
        return str(path)

    def test_detect_silence_on_silent_audio(self, detector, silence_audio):
        segments = detector.detect_silence(silence_audio)
        assert len(segments) >= 1
        assert isinstance(segments[0], SilenceSegment)
        assert segments[0].start_ms == 0

    def test_detect_silence_on_speech_audio(self, detector, speech_audio):
        segments = detector.detect_silence(speech_audio)
        assert len(segments) == 0  # No silence in speech signal

    def test_detect_silence_empty_file(self, detector, tmp_path):
        path = tmp_path / "empty.wav"
        sf.write(str(path), np.array([], dtype=np.float32), 16000)
        segments = detector.detect_silence(str(path))
        assert len(segments) >= 0  # May return 0 or 1 depending on implementation

    def test_threshold_validation(self, detector):
        with pytest.raises(ValueError, match="threshold must be 0.0–1.0"):
            VoiceActivityDetector(threshold=1.5)

    def test_threshold_override(self, detector, silence_audio):
        segments = detector.detect_silence(silence_audio, threshold=0.1)
        assert len(segments) >= 1

    def test_threshold_out_of_range_override(self, detector, silence_audio):
        with pytest.raises(ValueError, match="threshold must be 0.0–1.0"):
            detector.detect_silence(silence_audio, threshold=2.0)

    def test_remove_silence_segments(self, detector, silence_audio, tmp_path):
        segments = [SilenceSegment(0, 1000)]
        output = tmp_path / "cleaned.wav"
        result_path = detector.remove_silence_segments(silence_audio, segments, str(output))
        assert result_path == str(output)
        # After removing all silence, audio should be empty or very short
        data, sr = sf.read(str(output))
        assert len(data) == 0 or sr == 16000

    def test_remove_silence_no_output_path(self, detector, silence_audio):
        segments = [SilenceSegment(0, 500)]
        result = detector.remove_silence_segments(silence_audio, segments)
        assert result == silence_audio  # Returns original path

    def test_silence_segment_dataclass(self):
        seg = SilenceSegment(100.0, 500.0)
        assert seg.start_ms == 100.0
        assert seg.end_ms == 500.0

    def test_min_silence_duration_filtering(self, detector, tmp_path):
        """Segments shorter than min_silence_duration_ms are not returned."""
        path = tmp_path / "short_silence.wav"
        # 100ms silence (below 500ms default)
        data = np.zeros(int(16000 * 0.1), dtype=np.float32)
        sf.write(str(path), data, 16000)
        segments = detector.detect_silence(str(path))
        # All segments should be >= 500ms
        for seg in segments:
            assert seg.end_ms - seg.start_ms >= 500

    def test_detect_silence_stereo_audio(self, detector, tmp_path):
        path = tmp_path / "stereo.wav"
        data = np.zeros((int(16000), 2), dtype=np.float32)  # 1s stereo silence
        sf.write(str(path), data, 16000)
        segments = detector.detect_silence(str(path))
        assert len(segments) >= 1

    def test_detect_silence_nonexistent_file(self, detector):
        with pytest.raises(Exception):  # sf.read raises FileNotFoundError
            detector.detect_silence("/nonexistent/audio.wav")

    def test_group_silence_frames_empty(self, detector):
        segments = detector._group_silence_frames([], 480, 240, 500)
        assert segments == []

    def test_group_silence_frames_single(self, detector):
        segments = detector._group_silence_frames([0], 480, 240, 500)
        assert len(segments) == 0  # Single frame < 500ms

    def test_group_silence_frames_consecutive(self, detector):
        segments = detector._group_silence_frames([0, 1, 2, 3, 4], 480, 240, 100)
        assert len(segments) == 1
        assert segments[0].start_ms == 0
```

## Граничные случаи
- **Полная тишина**: весь файл = тишина → возвращается один сегмент на весь файл
- **Нет тишины**: весь файл = речь → возвращается `[]`
- **Короткая тишина** (< `min_silence_duration_ms`): сегмент игнорируется
- **Пустой аудиофайл**: возвращает `[]` или один сегмент (0, duration)
- **Стерео аудио**: автоматически конвертируется в моно
- **Несовпадающая частота дискретизации**: ресемплируется до 16kHz
- **Порог 0.0**: любой сигнал считается речью (нет тишины)
- **Порог 1.0**: любой сигнал считается тишиной
- **Неexistent файл**: raises `FileNotFoundError`
- **Очень длинный файл**: обработка по фреймам, без загрузки всего в память
