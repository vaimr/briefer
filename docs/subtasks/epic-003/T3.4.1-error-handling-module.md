# T3.4.1 — Create Error Handling Module

## Цель
Создать иерархию кастомных исключений для модулей worker и утилиту форматирования ошибок для структурированного логирования.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `error: Exception` — исходное исключение; `task_id: str` — ID задачи для контекста; `extra: dict` — дополнительные поля |
| **Обработка** | 1. Создание иерархии исключений. 2. Форматирование в dict с полями: task_id, error_type, message, traceback, extra. |
| **Выход** | `dict` с полями: task_id, error_type, message, traceback (опционально), extra |

## Пошаговый план

1. **Создать `worker/errors.py`**:
```python
import traceback
import logging
from dataclasses import dataclass, field
from typing import Optional, Any

logger = logging.getLogger(__name__)

class TranscriptionError(Exception):
    """Base exception for all transcription-related errors."""
    def __init__(self, message: str, task_id: Optional[str] = None):
        self.task_id = task_id
        super().__init__(message)

class ConversionError(TranscriptionError):
    """Raised when audio conversion fails."""
    pass

class ValidationError(TranscriptionError):
    """Raised when input validation fails."""
    pass

class PipelineError(TranscriptionError):
    """Raised when pipeline processing fails."""
    pass

@dataclass
class ErrorInfo:
    """Structured error information for logging."""
    task_id: Optional[str] = None
    error_type: str = ""
    message: str = ""
    traceback_str: Optional[str] = None
    extra: dict = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        """Convert to dict for structured logging."""
        return {
            "task_id": self.task_id,
            "error_type": self.error_type,
            "message": self.message,
            "traceback": self.traceback_str,
            "extra": self.extra,
        }

def format_error(
    error: Exception,
    task_id: Optional[str] = None,
    extra: Optional[dict] = None,
) -> dict[str, Any]:
    """Format an exception into a structured dict for logging.

    Args:
        error: The exception to format.
        task_id: Optional task ID for context.
        extra: Additional fields to include.

    Returns:
        Dict with task_id, error_type, message, traceback, extra.
    """
    error_type = type(error).__name__
    error_module = type(error).__module__
    full_type = f"{error_module}.{error_type}"

    tb_str = None
    if error.__traceback__:
        tb_str = ''.join(traceback.format_exception(error))

    info = ErrorInfo(
        task_id=task_id,
        error_type=full_type,
        message=str(error),
        traceback_str=tb_str,
        extra=extra or {},
    )
    return info.to_dict()

def log_error(error: Exception, task_id: Optional[str] = None, extra: Optional[dict] = None):
    """Log an error with structured formatting.

    Args:
        error: The exception to log.
        task_id: Optional task ID for context.
        extra: Additional fields to include.
    """
    error_dict = format_error(error, task_id, extra)
    logger.error(
        "Error in task %s: %s (%s)",
        task_id, error_dict["error_type"], error_dict["message"]
    )
    return error_dict
```

# tests/unit/test_errors.py
import pytest
from worker.errors import (
    TranscriptionError, ConversionError, ValidationError, PipelineError,
    ErrorInfo, format_error, log_error,
)

class TestExceptionHierarchy:
    def test_transcription_error_base(self):
        e = TranscriptionError("base error")
        assert str(e) == "base error"
        assert e.task_id is None

    def test_transcription_error_with_task_id(self):
        e = TranscriptionError("task error", task_id="t-001")
        assert e.task_id == "t-001"

    def test_conversion_error_inherits_transcription(self):
        e = ConversionError("conversion failed")
        assert isinstance(e, TranscriptionError)

    def test_validation_error_inherits_transcription(self):
        e = ValidationError("validation failed")
        assert isinstance(e, TranscriptionError)

    def test_pipeline_error_inherits_transcription(self):
        e = PipelineError("pipeline failed")
        assert isinstance(e, TranscriptionError)

    def test_error_hierarchy_chain(self):
        e = PipelineError("deep error", task_id="t-002")
        assert isinstance(e, TranscriptionError)
        assert e.task_id == "t-002"

class TestErrorInfo:
    def test_error_info_to_dict(self):
        info = ErrorInfo(
            task_id="t-001",
            error_type="ConversionError",
            message="ffmpeg failed",
            extra={"cmd": "ffmpeg -i input.mp3"},
        )
        d = info.to_dict()
        assert d["task_id"] == "t-001"
        assert d["error_type"] == "ConversionError"
        assert d["message"] == "ffmpeg failed"
        assert d["extra"]["cmd"] == "ffmpeg -i input.mp3"
        assert d["traceback"] is None

    def test_error_info_empty(self):
        info = ErrorInfo()
        d = info.to_dict()
        assert d["task_id"] is None
        assert d["error_type"] == ""
        assert d["message"] == ""
        assert d["extra"] == {}

class TestFormatError:
    def test_format_error_simple(self):
        e = ValueError("bad value")
        result = format_error(e, task_id="t-001")
        assert result["task_id"] == "t-001"
        assert "ValueError" in result["error_type"]
        assert result["message"] == "bad value"
        assert result["extra"] == {}

    def test_format_error_with_extra(self):
        e = RuntimeError("disk full")
        result = format_error(e, task_id="t-002", extra={"path": "/data"})
        assert result["extra"]["path"] == "/data"

    def test_format_error_custom_exception(self):
        e = ConversionError("mp4 not supported")
        result = format_error(e, task_id="t-003")
        assert "ConversionError" in result["error_type"]

    def test_format_error_no_task_id(self):
        e = ValueError("no context")
        result = format_error(e)
        assert result["task_id"] is None

    def test_format_error_with_traceback(self):
        try:
            raise ValueError("traceback test")
        except ValueError as e:
            result = format_error(e, task_id="t-004")
            assert result["traceback"] is not None
            assert "ValueError" in result["traceback"]

    def test_format_error_no_traceback(self):
        e = ValueError("no traceback")
        result = format_error(e, task_id="t-005")
        assert result["traceback"] is None

class TestLogError:
    def test_log_error_returns_dict(self, caplog):
        with caplog.at_level("ERROR"):
            result = log_error(ValueError("log test"), task_id="t-001")
        assert isinstance(result, dict)
        assert result["message"] == "log test"

    def test_log_error_logs_message(self, caplog):
        with caplog.at_level("ERROR"):
            log_error(ValueError("logged"), task_id="t-001")
        assert "Error in task t-001" in caplog.text
        assert "ValueError" in caplog.text
