# T3.4.2 — Add Logging to Pipeline

## Цель
Добавить структурированное логирование в TranscriptionPipeline с correlation_id для отслеживания задач.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — ID задачи (используется как correlation_id); `step: str` — название шага; `message: str` — сообщение; `level: str` — уровень логирования (info/warning/error) |
| **Обработка** | 1. Настройка logger в TranscriptionPipeline. 2. Логирование каждого шага: start, convert, vad, transcribe, save, complete. 3. Добавление correlation_id (task_id) к каждому сообщению. 4. Логирование ошибок с traceback. |
| **Выход** | Лог-сообщения в формате: `%(asctime)s [%(correlation_id)s] %(levelname)s %(message)s` |

## Пошаговый план

1. **Добавить логирование в TranscriptionPipeline** (в `worker/pipeline.py`):
```python
import logging
import sys
from logging.config import dictConfig

# Structured logging config
LOGGING_CONFIG = {
    "version": 1,
    "disable_existing_loggers": False,
    "formatters": {
        "structured": {
            "format": "%(asctime)s [%(correlation_id)s] %(levelname)s %(name)s: %(message)s",
        },
    },
    "handlers": {
        "console": {
            "class": "logging.StreamHandler",
            "formatter": "structured",
            "stream": sys.stdout,
        },
    },
    "root": {
        "handlers": ["console"],
        "level": "INFO",
    },
}

def setup_logging(level: str = "INFO"):
    """Setup structured logging with correlation_id support."""
    dictConfig(LOGGING_CONFIG)
    logging.getLogger().setLevel(getattr(logging, level.upper()))

class CorrelationFilter(logging.Filter):
    """Add correlation_id to log records."""
    def filter(self, record):
        if not hasattr(record, 'correlation_id'):
            record.correlation_id = '-'
        return True

def get_logger(name: str) -> logging.Logger:
    """Get a logger with correlation_id filter."""
    logger = logging.getLogger(name)
    if not any(isinstance(f, CorrelationFilter) for f in logger.filters):
        logger.addFilter(CorrelationFilter())
    return logger
```

2. **Интегрировать в TranscriptionPipeline**:
```python
# В TranscriptionPipeline.__init__:
self.logger = get_logger(__name__)

# В TranscriptionPipeline.process:
self.logger.info("Pipeline started for task_id=%s audio=%s", task_id, audio_path)
# ...
self.logger.info("Audio converted for task_id=%s", task_id)
# ...
self.logger.info("Transcription completed for task_id=%s", task_id)
# ...
self.logger.error("Pipeline failed for task_id=%s: %s", task_id, e, exc_info=True)
# ...
self.logger.info("Temp files cleaned for task_id=%s", task_id)
```

# tests/unit/test_pipeline_logging.py
import pytest
import logging
from worker.pipeline import TranscriptionPipeline, PipelineConfig, get_logger, CorrelationFilter

class TestPipelineLogging:
    @pytest.fixture
    def config(self, tmp_path):
        return PipelineConfig(
            audio_dir=str(tmp_path / "audio"),
            output_dir=str(tmp_path / "output"),
            temp_dir=str(tmp_path / "temp"),
        )

    @pytest.fixture
    def pipeline(self, config):
        return TranscriptionPipeline(config)

    def test_pipeline_has_logger(self, pipeline):
        assert pipeline.logger is not None
        assert isinstance(pipeline.logger, logging.Logger)

    def test_logger_has_correlation_filter(self, pipeline):
        has_filter = any(
            isinstance(f, CorrelationFilter)
            for f in pipeline.logger.filters
        )
        assert has_filter is True

    def test_logger_info_message(self, pipeline, caplog, sample_audio, mocker):
        with caplog.at_level("INFO"):
            mocker.patch.object(pipeline, '_transcribe', return_value=mocker.MagicMock())
            pipeline.process("task-001", sample_audio)
        assert any("Pipeline started" in r.message for r in caplog.records)
        assert any("task_id=task-001" in r.message for r in caplog.records)

    def test_logger_error_on_failure(self, pipeline, caplog, sample_audio, mocker):
        with caplog.at_level("ERROR"):
            mocker.patch.object(
                pipeline, '_transcribe',
                side_effect=RuntimeError("test error")
            )
            pipeline.process("task-001", sample_audio)
        assert any("Pipeline failed" in r.message for r in caplog.records)
        assert any("test error" in r.message for r in caplog.records)

    def test_logger_creates_output_dir(self, pipeline, caplog):
        with caplog.at_level("INFO"):
            new_dir = str(pipeline.config.output_dir)
            pipeline.config.output_dir = new_dir
            # Accessing config triggers dir creation
        # Logger should be set up before dir creation
        assert pipeline.logger is not None

    def test_get_logger_returns_logger(self):
        logger = get_logger("test.module")
        assert isinstance(logger, logging.Logger)

    def test_get_logger_adds_filter_once(self):
        logger1 = get_logger("test.unique")
        logger2 = get_logger("test.unique")
        # Same logger, filter added only once
        filter_count = sum(1 for f in logger1.filters if isinstance(f, CorrelationFilter))
        assert filter_count == 1
