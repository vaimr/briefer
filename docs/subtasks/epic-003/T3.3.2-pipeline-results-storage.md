# T3.3.2 — Add Pipeline Results Storage

## Цель
Реализовать сохранение и загрузку результатов транскрипции в JSON-формате с полной структурой данных.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — уникальный ID задачи; `transcription_text: str` — текст транскрипции; `segments: list[dict]` — сегменты с полями id, start, end, text; `duration_ms: float` — длительность аудио |
| **Обработка** | 1. Сериализация данных в JSON. 2. Запись в `output_dir/{task_id}.json`. 3. Создание директории если не существует. 4. При загрузке: чтение JSON, десериализация, валидация. |
| **Выход** | `save_result()` → None (запись); `load_result(task_id)` → dict с полями task_id, text, segments, duration_ms, timestamp, status |

## Пошаговый план

1. **Добавить методы в `worker/pipeline.py`** (результаты уже интегрированы в TranscriptionPipeline, но вынесем в отдельный модуль для переиспользования):
```python
# worker/results.py
import json
import os
import logging
from dataclasses import asdict
from datetime import datetime, timezone
from pathlib import Path
from typing import Optional, Any

logger = logging.getLogger(__name__)

class ResultsStorage:
    """Store and load transcription results as JSON files."""

    def __init__(self, output_dir: str = "/tmp/briefer/output"):
        self.output_dir = output_dir
        Path(output_dir).mkdir(parents=True, exist_ok=True)

    def save_result(
        self,
        task_id: str,
        transcription_text: str,
        segments: list[dict] = None,
        duration_ms: float = 0.0,
        status: str = "success",
        error_msg: Optional[str] = None,
    ) -> str:
        """Save transcription result to JSON file.

        Args:
            task_id: Unique task identifier.
            transcription_text: Transcribed text.
            segments: List of segment dicts.
            duration_ms: Audio duration in milliseconds.
            status: "success" or "failed".
            error_msg: Error message if failed.

        Returns:
            Path to the saved JSON file.
        """
        if segments is None:
            segments = []

        data = {
            "task_id": task_id,
            "text": transcription_text,
            "segments": segments,
            "duration_ms": duration_ms,
            "status": status,
            "error_msg": error_msg,
            "timestamp": datetime.now(timezone.utc).isoformat(),
        }

        output_file = os.path.join(self.output_dir, f"{task_id}.json")
        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

        logger.info("Result saved: %s → %s", task_id, output_file)
        return output_file

    def load_result(self, task_id: str) -> Optional[dict[str, Any]]:
        """Load transcription result from JSON file.

        Args:
            task_id: Unique task identifier.

        Returns:
            Dict with result data, or None if not found.
        """
        output_file = os.path.join(self.output_dir, f"{task_id}.json")

        if not os.path.isfile(output_file):
            logger.warning("Result not found: %s", output_file)
            return None

        try:
            with open(output_file, 'r', encoding='utf-8') as f:
                data = json.load(f)

            # Validate required fields
            required = ["task_id", "text", "segments", "duration_ms", "status", "timestamp"]
            for field in required:
                if field not in data:
                    logger.error("Missing field '%s' in result for task_id=%s", field, task_id)
                    return None

            return data

        except json.JSONDecodeError as e:
            logger.error("Invalid JSON in result for task_id=%s: %s", task_id, e)
            return None

    def result_exists(self, task_id: str) -> bool:
        """Check if result exists for task_id."""
        output_file = os.path.join(self.output_dir, f"{task_id}.json")
        return os.path.isfile(output_file)

    def delete_result(self, task_id: str) -> bool:
        """Delete result file for task_id.

        Returns:
            True if file was deleted, False if not found.
        """
        output_file = os.path.join(self.output_dir, f"{task_id}.json")
        if os.path.isfile(output_file):
            os.remove(output_file)
            logger.info("Result deleted: %s", task_id)
            return True
        return False

    def list_results(self) -> list[str]:
        """List all task_ids with saved results."""
        results = []
        for filename in os.listdir(self.output_dir):
            if filename.endswith('.json'):
                task_id = filename[:-5]  # Remove .json
                results.append(task_id)
        return sorted(results)
```

```python
# tests/unit/test_results_storage.py
import pytest
import os
import json
from worker.results import ResultsStorage

class TestResultsStorage:
    @pytest.fixture
    def storage(self, tmp_path):
        return ResultsStorage(output_dir=str(tmp_path / "output"))

    def test_save_result_creates_json(self, storage, tmp_path):
        path = storage.save_result(
            task_id="task-001",
            transcription_text="Hello world",
            segments=[{"id": 0, "start": 0, "end": 1.0, "text": "Hello world"}],
            duration_ms=1000.0,
        )
        assert os.path.isfile(path)
        assert "task-001.json" in path

    def test_save_result_json_content(self, storage):
        storage.save_result(
            task_id="task-001",
            transcription_text="Привет мир",
            segments=[],
            duration_ms=500.0,
        )
        with open(storage.output_dir + "/task-001.json") as f:
            data = json.load(f)
        assert data["task_id"] == "task-001"
        assert data["text"] == "Привет мир"
        assert data["segments"] == []
        assert data["duration_ms"] == 500.0
        assert data["status"] == "success"
        assert "timestamp" in data

    def test_save_result_creates_output_dir(self, tmp_path):
        new_dir = str(tmp_path / "new_output")
        storage = ResultsStorage(output_dir=new_dir)
        storage.save_result("task-001", "text")
        assert os.path.isdir(new_dir)

    def test_save_result_empty_segments(self, storage):
        storage.save_result("task-001", "text", segments=None)
        with open(storage.output_dir + "/task-001.json") as f:
            data = json.load(f)
        assert data["segments"] == []

    def test_save_result_failed_status(self, storage):
        storage.save_result(
            task_id="task-001",
            transcription_text="",
            status="failed",
            error_msg="Whisper error",
        )
        with open(storage.output_dir + "/task-001.json") as f:
            data = json.load(f)
        assert data["status"] == "failed"
        assert data["error_msg"] == "Whisper error"

    def test_load_result_success(self, storage):
        storage.save_result("task-001", "Hello", duration_ms=1000)
        data = storage.load_result("task-001")
        assert data is not None
        assert data["text"] == "Hello"
        assert data["duration_ms"] == 1000

    def test_load_result_not_found(self, storage):
        data = storage.load_result("nonexistent")
        assert data is None

    def test_load_result_invalid_json(self, storage, tmp_path):
        bad_file = os.path.join(storage.output_dir, "task-001.json")
        with open(bad_file, 'w') as f:
            f.write("{invalid json}")
        data = storage.load_result("task-001")
        assert data is None

    def test_load_result_missing_field(self, storage, tmp_path):
        bad_file = os.path.join(storage.output_dir, "task-001.json")
        with open(bad_file, 'w') as f:
            json.dump({"task_id": "task-001", "text": "Hello"}, f)  # Missing segments, etc.
        data = storage.load_result("task-001")
        assert data is None

    def test_result_exists_true(self, storage):
        storage.save_result("task-001", "text")
        assert storage.result_exists("task-001") is True

    def test_result_exists_false(self, storage):
        assert storage.result_exists("nonexistent") is False

    def test_delete_result(self, storage):
        storage.save_result("task-001", "text")
        result = storage.delete_result("task-001")
        assert result is True
        assert not storage.result_exists("task-001")

    def test_delete_result_not_found(self, storage):
        result = storage.delete_result("nonexistent")
        assert result is False

    def test_list_results(self, storage):
        storage.save_result("task-002", "text2")
        storage.save_result("task-001", "text1")
        storage.save_result("task-003", "text3")
        results = storage.list_results()
        assert results == ["task-001", "task-002", "task-003"]

    def test_list_results_empty(self, storage):
        results = storage.list_results()
        assert results == []

    def test_save_result_utf8_encoding(self, storage):
        storage.save_result("task-001", "Привет мир! 你好世界! 🌍")
        with open(storage.output_dir + "/task-001.json", encoding='utf-8') as f:
            data = json.load(f)
        assert data["text"] == "Привет мир! 你好世界! 🌍"

    def test_timestamp_format(self, storage):
        storage.save_result("task-001", "text")
        with open(storage.output_dir + "/task-001.json") as f:
            data = json.load(f)
        # Should be ISO format
        from datetime import datetime
        datetime.fromisoformat(data["timestamp"])  # Raises if invalid
```

## Граничные случаи
- **Пустой segments list**: сохраняется как `[]`
- **None segments**: приводится к `[]`
- **UTF-8 текст**: корректно сохраняется/загружается (кириллица, эмодзи)
- **Неexistent файл**: `load_result()` возвращает `None`
- **Invalid JSON**: `load_result()` возвращает `None` (не exception)
- **Missing fields**: `load_result()` возвращает `None` (не exception)
- **Повторная запись**: перезаписывает существующий файл
- **Удаление несуществующего**: возвращает `False`
