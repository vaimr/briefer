# T3.2.1 — Create Whisper Transcription Module

## Цель
Реализовать транскрипцию аудио через faster-whisper.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| audio_path | Whisper transcribe | TranscriptionResult |
| audio_path, language | Transcribe with language | TranscriptionResult |
| audio_path, model_size | Load model → transcribe | TranscriptionResult |

## Пошаговый план

1. Создать `worker/transcribe.py`:

```python
"""Whisper transcription module."""

import logging
import os
from dataclasses import dataclass, field
from typing import List, Optional

logger = logging.getLogger(__name__)


@dataclass
class TranscriptionSegment:
    """A single transcription segment."""
    start: float
    end: float
    text: str
    speaker: Optional[str] = None
    confidence: float = 0.0


@dataclass
class TranscriptionResult:
    """Transcription result."""
    text: str
    segments: List[TranscriptionSegment] = field(default_factory=list)
    language: Optional[str] = None
    duration: float = 0.0
    model_size: str = "base"
    compute_type: str = "float32"


class TranscriptionError(Exception):
    """Transcription failed."""


class WhisperTranscriber:
    """Transcribe audio using faster-whisper."""

    _model_cache: dict = {}  # model_key → model_instance

    def __init__(
        self,
        model_size: str = "base",
        device: str = "cpu",
        compute_type: str = "float32",
        language: Optional[str] = None,
    ):
        """Initialize transcriber.

        Args:
            model_size: Whisper model size (tiny, base, small, medium, large).
            device: Device to use (cpu, cuda).
            compute_type: Compute type (int8, int8_float16, float16, float32).
            language: Source language code (None = auto-detect).
        """
        self.model_size = model_size
        self.device = device
        self.compute_type = compute_type
        self.language = language
        self._model = None

    @property
    def model_key(self) -> str:
        """Get model cache key."""
        return f"{self.model_size}:{self.device}:{self.compute_type}:{self.language or 'auto'}"

    def _get_model(self):
        """Get or create Whisper model (cached)."""
        if self._model is not None:
            return self._model

        if self.model_key in WhisperTranscriber._model_cache:
            self._model = WhisperTranscriber._model_cache[self.model_key]
            return self._model

        # Lazy import to avoid dependency if not needed
        try:
            from faster_whisper import WhisperModel
        except ImportError:
            raise TranscriptionError("faster-whisper not installed. Run: pip install faster-whisper")

        logger.info(f"Loading Whisper model: {self.model_size} on {self.device}")
        self._model = WhisperModel(
            self.model_size,
            device=self.device,
            compute_type=self.compute_type,
            download_root=None,
        )
        WhisperTranscriber._model_cache[self.model_key] = self._model
        logger.info(f"Model loaded successfully: {self.model_key}")
        return self._model

    def transcribe(
        self,
        audio_path: str,
        beam_size: int = 5,
        vad_filter: bool = True,
        language: Optional[str] = None,
    ) -> TranscriptionResult:
        """Transcribe audio file.

        Args:
            audio_path: Path to audio file.
            beam_size: Beam size for decoding.
            vad_filter: Use VAD to filter silence.
            language: Language code (overrides init).

        Returns:
            TranscriptionResult with text and segments.

        Raises:
            TranscriptionError: On transcription failure.
        """
        audio_path = str(audio_path)

        if not os.path.exists(audio_path):
            raise TranscriptionError(f"Audio file not found: {audio_path}")

        if os.path.getsize(audio_path) == 0:
            raise TranscriptionError(f"Audio file is empty: {audio_path}")

        model = self._get_model()
        lang = language or self.language

        logger.info(f"Transcribing {audio_path} (beam_size={beam_size}, vad={vad_filter})")

        try:
            segments, info = model.transcribe(
                audio_path,
                beam_size=beam_size,
                vad_filter=vad_filter,
                language=lang,
            )

            # Collect segments
            result_segments = []
            full_text = ""

            for seg in segments:
                segment = TranscriptionSegment(
                    start=seg.start,
                    end=seg.end,
                    text=seg.text.strip(),
                    confidence=seg.avg_logprob,
                )
                result_segments.append(segment)
                full_text += seg.text + " "

            result = TranscriptionResult(
                text=full_text.strip(),
                segments=result_segments,
                language=info.language,
                duration=info.duration,
                model_size=self.model_size,
                compute_type=self.compute_type,
            )

            logger.info(
                f"Transcription complete: {len(result_segments)} segments, "
                f"{result.duration:.1f}s, language={result.language}"
            )
            return result

        except Exception as e:
            raise TranscriptionError(f"Transcription failed: {e}") from e

    def get_segments(self, result: TranscriptionResult) -> List[TranscriptionSegment]:
        """Get segments from result.

        Args:
            result: TranscriptionResult.

        Returns:
            List of segments.
        """
        return result.segments

    def get_full_text(self, result: TranscriptionResult) -> str:
        """Get full text from result.

        Args:
            result: TranscriptionResult.

        Returns:
            Full transcribed text.
        """
        return result.text

    @classmethod
    def clear_model_cache(cls) -> None:
        """Clear the model cache."""
        cls._model_cache.clear()
        logger.info("Model cache cleared")
```

## Граничные случаи

- Файл не существует — TranscriptionError("not found")
- Пустой файл — TranscriptionError("empty")
- language = None — auto-detect
- language = "en" — English
- language = "ru" — Russian
- beam_size = 5 — default
- beam_size = 1 — fast mode
- vad_filter = True — VAD enabled
- vad_filter = False — VAD disabled
- model_cache: first load — loads model
- model_cache: second load — returns cached
- clear_model_cache: clears all
- transcribe with invalid audio — TranscriptionError
- segments with empty text — included (not filtered)
- single segment — OK
- 100 segments — OK
