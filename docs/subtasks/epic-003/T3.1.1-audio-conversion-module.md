# T3.1.1 — Create Audio Conversion Module

## Цель
Реализовать конвертацию аудио в WAV формат для Whisper через ffmpeg.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| input_path, output_path | ffmpeg convert → WAV | True/False |
| input_path, output_path, duration_limit | convert + check duration | True/False |
| invalid input_path | Check exists | ConversionError |

```python
"""Audio file conversion using ffmpeg."""

import logging
import os
import shutil
import subprocess
from pathlib import Path
from typing import Optional

logger = logging.getLogger(__name__)


class AudioConversionError(Exception):
    """Audio conversion failed."""


class AudioConverter:
    """Convert audio files using ffmpeg."""

    SUPPORTED_FORMATS = {".mp3", ".ogg", ".wav", ".webm", ".m4a", ".flac", ".aac", ".opus"}

    def __init__(
        self,
        ffmpeg_path: Optional[str] = None,
        sample_rate: int = 16000,
        channels: int = 1,
        duration_limit: int = 3600,
    ):
        """Initialize audio converter.

        Args:
            ffmpeg_path: Path to ffmpeg binary.
            sample_rate: Output sample rate in Hz.
            channels: Output number of channels (1 = mono).
            duration_limit: Maximum input duration in seconds.
        """
        self.ffmpeg_path = ffmpeg_path or shutil.which("ffmpeg")
        self.sample_rate = sample_rate
        self.channels = channels
        self.duration_limit = duration_limit

        if not self.ffmpeg_path:
            raise AudioConversionError("ffmpeg not found in PATH")

    def convert_to_wav(
        self,
        input_path: str,
        output_path: str,
        duration_limit: Optional[int] = None,
    ) -> str:
        """Convert audio file to WAV format.

        Args:
            input_path: Path to input audio file.
            output_path: Path to output WAV file.
            duration_limit: Maximum duration in seconds (overrides default).

        Returns:
            Path to output WAV file.

        Raises:
            AudioConversionError: On conversion failure.
        """
        input_path = str(input_path)
        output_path = str(output_path)

        # Check input file exists
        if not os.path.exists(input_path):
            raise AudioConversionError(f"Input file not found: {input_path}")

        # Check input file is not empty
        if os.path.getsize(input_path) == 0:
            raise AudioConversionError(f"Input file is empty: {input_path}")

        # Check input extension
        _, ext = os.path.splitext(input_path)
        ext = ext.lower()
        if ext not in self.SUPPORTED_FORMATS:
            raise AudioConversionError(f"Unsupported format: {ext}")

        # Check duration
        actual_duration = self.get_duration(input_path)
        limit = duration_limit if duration_limit is not None else self.duration_limit

        if actual_duration > limit:
            raise AudioConversionError(
                f"Input duration {actual_duration:.1f}s exceeds limit {limit}s"
            )

        # Build ffmpeg command
        cmd = [
            self.ffmpeg_path,
            "-i", input_path,
            "-ar", str(self.sample_rate),
            "-ac", str(self.channels),
            "-f", "wav",
            "-y",  # overwrite output
            output_path,
        ]

        logger.info(f"Converting {input_path} -> {output_path}")
        logger.debug(f"ffmpeg command: {' '.join(cmd)}")

        try:
            result = subprocess.run(
                cmd,
                capture_output=True,
                timeout=600,  # 10 minutes max
                check=True,
            )
            logger.info(f"Conversion successful: {output_path}")
            return output_path

        except subprocess.CalledProcessError as e:
            stderr = e.stderr.decode("utf-8", errors="replace") if e.stderr else ""
            raise AudioConversionError(f"ffmpeg failed: {stderr}") from e
        except subprocess.TimeoutExpired as e:
            raise AudioConversionError("Conversion timed out after 600s") from e

    def get_duration(self, file_path: str) -> float:
        """Get audio file duration in seconds.

        Args:
            file_path: Path to audio file.

        Returns:
            Duration in seconds.

        Raises:
            AudioConversionError: On failure.
        """
        file_path = str(file_path)

        if not os.path.exists(file_path):
            raise AudioConversionError(f"File not found: {file_path}")

        cmd = [
            self.ffmpeg_path,
            "-i", file_path,
            "-f", "null",
            "-",
        ]

        try:
            result = subprocess.run(
                cmd,
                capture_output=True,
                timeout=60,
                check=False,
            )
            stderr = result.stderr.decode("utf-8", errors="replace")

            # Parse "Duration: 00:01:23.45" from stderr
            import re
            match = re.search(r"Duration: (\d+):(\d+):(\d+\.\d+)", stderr)
            if match:
                h, m, s = match.groups()
                return int(h) * 3600 + int(m) * 60 + float(s)

            raise AudioConversionError("Could not parse duration from ffmpeg output")

        except subprocess.TimeoutExpired:
            raise AudioConversionError("Duration check timed out") from None

    def is_supported(self, file_path: str) -> bool:
        """Check if file format is supported.

        Args:
            file_path: Path to file.

        Returns:
            True if supported.
        """
        _, ext = os.path.splitext(file_path)
        return ext.lower() in self.SUPPORTED_FORMATS
```

## Граничные случаи

- Файл не существует — AudioConversionError("not found")
- Пустой файл (0 байт) — AudioConversionError("empty")
- Неподдерживаемый формат (.txt) — AudioConversionError("unsupported")
- duration_limit = 3600 — default limit
- duration_limit = 60 — custom limit
- input duration > limit — AudioConversionError
- input duration == limit — OK
- input duration < limit — OK
- ffmpeg не установлен — AudioConversionError("not found in PATH")
- ffmpeg exit code != 0 — AudioConversionError with stderr
- ffmpeg timeout > 600s — AudioConversionError("timed out")
- output_dir не существует — создаётся автоматически (os.makedirs)
- output_path с unicode — OK
- convert_to_wav с duration_limit overrides default
