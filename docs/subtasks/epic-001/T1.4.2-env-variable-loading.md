1. Создать `bot/env_loader.py`:

```python
"""Environment variable loader with validation."""

import os
from pathlib import Path
from typing import Optional

from dotenv import load_dotenv

from bot.config import BotConfig, WorkerConfig, SharedConfig


def load_env_file(path: Optional[str] = None) -> None:
    """Load .env file from specified path or default location.

    Args:
        path: Path to .env file. Defaults to project root .env.
    """
    if path:
        env_path = Path(path)
    else:
        env_path = Path(__file__).parent.parent / ".env"

    if env_path.exists():
        load_dotenv(env_path, override=True)
    else:
        example_path = Path(__file__).parent.parent / ".env.example"
        if example_path.exists():
            print("Warning: .env not found. Using .env.example as reference.")


def get_bot_config() -> BotConfig:
    """Load and return bot configuration."""
    return BotConfig()


def get_worker_config() -> WorkerConfig:
    """Load and return worker configuration."""
    return WorkerConfig()


def get_shared_config() -> SharedConfig:
    """Load and return shared configuration."""
    return SharedConfig()


def validate_required_env_vars(*vars: str) -> list[str]:
    """Validate that required environment variables are set.

    Args:
        *vars: Variable names to check.

    Returns:
        List of missing variable names.
    """
    missing = [v for v in vars if not os.environ.get(v)]
    return missing


def validate_matrix_config() -> bool:
    """Validate Matrix configuration is complete.

    Returns:
        True if all required Matrix fields are set.
    """
    missing = validate_required_env_vars(
        "MATRIX_USER",
        "MATRIX_PASSWORD",
        "MATRIX_HOMESERVER",
    )
    if missing:
        raise ValueError(f"Missing Matrix config: {', '.join(missing)}")
    return True


def validate_worker_config() -> bool:
    """Validate Worker configuration is complete.

    Returns:
        True if all required Worker fields are set.
    """
    missing = validate_required_env_vars(
        "REDIS_URL",
        "WHISPER_MODEL",
    )
    if missing:
        raise ValueError(f"Missing Worker config: {', '.join(missing)}")
    return True
```

2. Создать `tests/unit/test_env_loader.py` для проверки.

## Граничные случаи

- `.env` не существует — warning, но не ошибка
- `.env` пустой — все значения из defaults
- `.env` содержит пустые строки — игнорируются
- `.env` содержит пустые значения `KEY=` — пустая строка, не default
- `validate_matrix_config()` без `MATRIX_USER` — ValueError
- `validate_worker_config()` без `REDIS_URL` — ValueError
- Переменная установлена в пустую строку — считается "set"
- Дублирующиеся переменные в `.env` — последняя побеждает
