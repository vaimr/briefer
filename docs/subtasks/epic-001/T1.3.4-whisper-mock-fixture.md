## Цель
Создать fixture для mock-Whisper в тестах, имитирующего faster_whisper.

## Spec

| Input | Processing | Output |

| Вызов mock-Whisper методов | Возврат предопределённых Segments | Результаты транскрипции |

## Пошаговый план

1. Создать `tests/fixtures/whisper_mock.py`:

   ```python
   """Mock Whisper fixtures for testing without real Whisper model."""

   from dataclasses import dataclass, field
   from typing import Iterator, List, Optional
   from unittest.mock import MagicMock
   import pytest


   @dataclass
   class MockWord:
       """Mock word from Whisper transcription."""
       word: str
       start: float
       end: float
       probability: float = 0.9


   @dataclass
   class MockSegment:
       """Mock segment from Whisper transcription."""
       text: str
       start: float
       end: float
       words: List[MockWord] = field(default_factory=list)

       def __str__(self):
           return self.text


   @pytest.fixture
   def mock_whisper_segment():
       """Mock Whisper Segment with all fields."""
       return MockSegment(
           text="Hello, world!",
           start=0.0,
           end=1.5,
           words=[
               MockWord(word="Hello", start=0.0, end=0.5, probability=0.95),
               MockWord(word="world", start=0.6, end=1.2, probability=0.92),
           ],
       )


   @pytest.fixture
   def mock_whisper_segments():
       """Mock Whisper Segments iterator."""
       segments = [
           MockSegment(text="First sentence.", start=0.0, end=1.0),
           MockSegment(text="Second sentence.", start=1.5, end=2.5),
           MockSegment(text="Third sentence.", start=3.0, end=4.0),
       ]
       return iter(segments)


   @pytest.fixture
   def mock_whisper_model():
       """Mock WhisperModel class."""
       model = MagicMock()
       model.model_id = "large-v3"
       model.device = "cpu"
       model.compute_type = "int8"
       model.generate_vad_signal.return_value = True

       # Transcribe method
       mock_segments = [
           MockSegment(text="Sample transcription text.", start=0.0, end=2.0),
           MockSegment(text="This is a test of the whisper mock.", start=2.5, end=5.0),
       ]
       model.transcribe.return_value = (iter(mock_segments), {})

       # VAD methods
       model.get_vad_segments.return_value = []
       model.filter_vad_segments.return_value = mock_segments

       return model


   @pytest.fixture
   def whisper_transcription_result(mock_whisper_model):
       """Pre-built transcription result using mock model."""
       segments, info = mock_whisper_model.transcribe("fake_audio.wav")
       full_text = "".join(seg.text for seg in segments)
       return {
           "segments": list(segments),
           "text": full_text,
           "language": info.get("language", "en") if info else "en",
       }
   ```

2. Создать `tests/unit/test_whisper_mock.py` для проверки fixtures.

## Граничные случаи

- Пустая транскрипция — `transcribe()` возвращает пустой iterator
- Один segment — `len(segments) == 1`
- Segment без слов — `segment.words == []`
- Segment с нулевой длительностью — `end - start == 0`
- `generate_vad_signal()` возвращает `False` если VAD не найден
- `filter_vad_segments()` возвращает пустой список если нет VAD-сегментов
