# T1.3.5 — Create HTTP Mock Fixture

## Цель
Создать fixture для mock-HTTP-запросов в тестах, имитирующего LLM API и внешние сервисы.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| Вызов mock-HTTP методов | Возврат предопределённых ответов | Результаты HTTP-запросов |

## Пошаговый план

1. Создать `tests/fixtures/http_mock.py`:

   ```python
   """Mock HTTP fixtures for testing external API calls."""

   from unittest.mock import MagicMock, Mock
   from http import HTTPStatus
   import pytest


   @pytest.fixture
   def mock_requests_session():
       """Mock requests.Session with common methods."""
       session = MagicMock()

       # GET requests
       mock_response_get = MagicMock()
       mock_response_get.status_code = 200
       mock_response_get.json.return_value = {"key": "value"}
       mock_response_get.text = '{"key": "value"}'
       mock_response_get.content = b'{"key": "value"}'
       mock_response_get.headers = {"Content-Type": "application/json"}
       session.get.return_value = mock_response_get

       # POST requests
       mock_response_post = MagicMock()
       mock_response_post.status_code = 200
       mock_response_post.json.return_value = {"result": "success"}
       mock_response_post.text = '{"result": "success"}'
       mock_response_post.content = b'{"result": "success"}'
       session.post.return_value = mock_response_post

       # Error responses
       mock_response_error = MagicMock()
       mock_response_error.status_code = 500
       mock_response_error.raise_for_status.side_effect = Exception("500 Internal Server Error")
       session.get.side_effect = [mock_response_get, mock_response_error]

       return session


   @pytest.fixture
   def mock_llm_api():
       """Mock LLM API response."""
       mock_response = MagicMock()
       mock_response.status_code = 200
       mock_response.json.return_value = {
           "choices": [
               {
                   "message": {
                       "role": "assistant",
                       "content": "This is a generated summary of the text.",
                   }
               }
           ],
           "usage": {
               "prompt_tokens": 100,
               "completion_tokens": 50,
               "total_tokens": 150,
           },
       }
       return mock_response


   @pytest.fixture
   def mock_audio_download():
       """Mock audio file download."""
       mock_response = MagicMock()
       mock_response.status_code = 200
       mock_response.content = b"fake audio data for testing"
       mock_response.headers = {"Content-Type": "audio/wav"}
       return mock_response


   @pytest.fixture
   def mock_pdf_output():
       """Mock PDF generation output."""
       mock_pdf = MagicMock()
       mock_pdf.getvalue.return_value = b"%PDF-1.4 fake PDF content"
       return mock_pdf


   @pytest.fixture
   def mock_http_error_4xx():
       """Mock 4xx HTTP error response."""
       mock_response = MagicMock()
       mock_response.status_code = 400
       mock_response.raise_for_status.side_effect = Exception("400 Bad Request")
       return mock_response


   @pytest.fixture
   def mock_http_error_5xx():
       """Mock 5xx HTTP error response."""
       mock_response = MagicMock()
       mock_response.status_code = 503
       mock_response.raise_for_status.side_effect = Exception("503 Service Unavailable")
       return mock_response


   @pytest.fixture
   def mock_connection_error():
       """Mock connection error."""
       import requests

       return requests.exceptions.ConnectionError("Connection refused")


   @pytest.fixture
   def mock_timeout_error():
       """Mock timeout error."""
       import requests

       return requests.exceptions.Timeout("Request timed out")
   ```

2. Создать `tests/unit/test_http_mock.py` для проверки fixtures.

## Граничные случаи

- `status_code == 400` — `raise_for_status()` вызывает `HTTPError`
- `status_code == 500` — retryable ошибка
- `status_code == 503` — retryable ошибка
- `ConnectionError` — retryable ошибка
- `Timeout` — retryable ошибка
- `json()` вызывается на пустом response — вызывает `JSONDecodeError`
- `content` — пустые bytes `b""`
- `headers` — отсутствует `Content-Type`
