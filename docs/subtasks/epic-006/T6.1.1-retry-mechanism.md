# T6.1.1 — Retry Mechanism with Exponential Backoff

## Цель
Реализовать универсальный retry-механизм с exponential backoff, jitter и декоратором для обработки transient-ошибок.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `func: Callable` — функция для вызова; `max_retries: int` — макс. попыток; `base_delay: float` — базовая задержка; `retryable_exceptions: tuple` — типы ошибок для retry |
| **Обработка** | 1. Вызов func. 2. При ошибке из retryable_exceptions → sleep(base_delay * 2^attempt + jitter). 3. Повторить до max_retries. 4. При исчерпании → raise последнюю ошибку. |
| **Выход** | Возвращаемое значение func при успехе, или raise последнюю ошибку |

## Пошаговый план

1. **Создать `worker/retry.py`**:
```python
import asyncio
import functools
import logging
import random
import time
from typing import Callable, Type, TypeVar, Tuple, Optional

logger = logging.getLogger(__name__)

T = TypeVar("T")

class RetryableError(Exception):
    """Exception that indicates the operation can be retried."""
    pass

class MaxRetriesExhaustedError(Exception):
    """Exception raised when all retry attempts are exhausted."""
    def __init__(self, message: str, last_exception: Optional[Exception] = None):
        self.last_exception = last_exception
        super().__init__(message)

def retry_with_backoff(
    func: Callable[..., T],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    jitter: bool = True,
    retryable_exceptions: Tuple[Type[Exception], ...] = (
        ConnectionError,
        TimeoutError,
        OSError,
        RetryableError,
    ),
    **kwargs,
) -> T:
    """Execute func with exponential backoff retry.

    Args:
        func: Function to execute.
        max_retries: Maximum number of retry attempts.
        base_delay: Base delay in seconds before first retry.
        max_delay: Maximum delay cap in seconds.
        jitter: Whether to add random jitter to delay.
        retryable_exceptions: Tuple of exception types that trigger retry.

    Returns:
        Result of func.

    Raises:
        MaxRetriesExhaustedError: If all retries fail.
    """
    last_exception = None

    for attempt in range(max_retries + 1):
        try:
            return func(*kwargs.get("args", ()), **kwargs.get("kwargs", {}))
        except retryable_exceptions as e:
            last_exception = e
            if attempt < max_retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                if jitter:
                    delay *= (0.5 + random.random() * 0.5)
                logger.warning(
                    "Retry %d/%d for %s: %s (delay=%.2fs)",
                    attempt + 1, max_retries, func.__name__, e, delay,
                )
                time.sleep(delay)
            else:
                logger.error(
                    "All %d retries exhausted for %s: %s",
                    max_retries, func.__name__, e,
                )
                raise MaxRetriesExhaustedError(
                    f"All {max_retries} retries exhausted for {func.__name__}",
                    last_exception=e,
                ) from e

    raise MaxRetriesExhaustedError(
        f"Function {func.__name__} failed after all retries",
        last_exception=last_exception,
    )

async def async_retry_with_backoff(
    coro_func: Callable[..., T],
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    jitter: bool = True,
    retryable_exceptions: Tuple[Type[Exception], ...] = (
        ConnectionError,
        TimeoutError,
        OSError,
        RetryableError,
    ),
    **kwargs,
) -> T:
    """Execute async coro with exponential backoff retry."""
    last_exception = None

    for attempt in range(max_retries + 1):
        try:
            return await coro_func(*kwargs.get("args", ()), **kwargs.get("kwargs", {}))
        except retryable_exceptions as e:
            last_exception = e
            if attempt < max_retries:
                delay = min(base_delay * (2 ** attempt), max_delay)
                if jitter:
                    delay *= (0.5 + random.random() * 0.5)
                logger.warning(
                    "Async retry %d/%d for %s: %s (delay=%.2fs)",
                    attempt + 1, max_retries, coro_func.__name__, e, delay,
                )
                await asyncio.sleep(delay)
            else:
                logger.error(
                    "All %d async retries exhausted for %s: %s",
                    max_retries, coro_func.__name__, e,
                )
                raise MaxRetriesExhaustedError(
                    f"All {max_retries} retries exhausted for {coro_func.__name__}",
                    last_exception=e,
                ) from e

    raise MaxRetriesExhaustedError(
        f"Async function {coro_func.__name__} failed after all retries",
        last_exception=last_exception,
    )

def retry(
    max_retries: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    jitter: bool = True,
    retryable_exceptions: Tuple[Type[Exception], ...] = (
        ConnectionError,
        TimeoutError,
        OSError,
        RetryableError,
    ),
):
    """Decorator for retry with exponential backoff."""
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @functools.wraps(func)
        def wrapper(*args, **kwargs) -> T:
            return retry_with_backoff(
                func,
                max_retries=max_retries,
                base_delay=base_delay,
                max_delay=max_delay,
                jitter=jitter,
                retryable_exceptions=retryable_exceptions,
                args=args,
                kwargs=kwargs,
            )
        return wrapper
    return decorator
```

2. **Написать unit-тесты**:
```python
# tests/unit/test_retry.py
import pytest
import asyncio
import time
from worker.retry import (
    retry_with_backoff, async_retry_with_backoff,
    retry, RetryableError, MaxRetriesExhaustedError,
)

class TestRetryWithBackoff:
    def test_success_first_call(self):
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            return "ok"
        result = retry_with_backoff(func, max_retries=3)
        assert result == "ok"
        assert call_count == 1

    def test_retry_succeeds_on_second(self):
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise ConnectionError("temp")
            return "ok"
        result = retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert result == "ok"
        assert call_count == 2

    def test_retry_exhausted_raises(self):
        def func():
            raise ConnectionError("always fails")
        with pytest.raises(MaxRetriesExhaustedError) as exc_info:
            retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert exc_info.value.last_exception is not None
        assert isinstance(exc_info.value.last_exception, ConnectionError)

    def test_retryable_error_triggers_retry(self):
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise RetryableError("retryable")
            return "ok"
        result = retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert result == "ok"

    def test_non_retryable_error_no_retry(self):
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            raise ValueError("not retryable")
        with pytest.raises(ValueError):
            retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert call_count == 1  # Only 1 call, no retry

    def test_exponential_backoff_timing(self):
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            if call_count < 3:
                raise ConnectionError("fail")
            return "ok"
        start = time.monotonic()
        result = retry_with_backoff(func, max_retries=3, base_delay=0.05, jitter=False)
        elapsed = time.monotonic() - start
        assert result == "ok"
        assert elapsed >= 0.15  # 0.05 + 0.10 + 0.20 = 0.35 total delays

    def test_max_delay_cap(self):
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise ConnectionError("fail")
            return "ok"
        result = retry_with_backoff(func, max_retries=3, base_delay=100.0, max_delay=1.0, jitter=False)
        assert result == "ok"

    def test_no_jitter(self):
        """Jitter=False should still work (deterministic delay)."""
        call_count = 0
        def func():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise ConnectionError("fail")
            return "ok"
        result = retry_with_backoff(func, max_retries=3, base_delay=0.01, jitter=False)
        assert result == "ok"

    def test_retry_with_args_kwargs(self):
        def func(a, b, c=10):
            return a + b + c
        result = retry_with_backoff(
            func, max_retries=3, base_delay=0.01,
            args=(1, 2), kwargs={"c": 3},
        )
        assert result == 6

    def test_max_retries_zero(self):
        def func():
            raise ConnectionError("fail")
        with pytest.raises(MaxRetriesExhaustedError):
            retry_with_backoff(func, max_retries=0, base_delay=0.01)

    def test_max_retries_exhausted_error_message(self):
        def func():
            raise ConnectionError("fail")
        with pytest.raises(MaxRetriesExhaustedError) as exc_info:
            retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert "retries exhausted" in str(exc_info.value)

class TestAsyncRetryWithBackoff:
    @pytest.mark.asyncio
    async def test_async_success(self):
        async def func():
            return "async_ok"
        result = await async_retry_with_backoff(func, max_retries=3)
        assert result == "async_ok"

    @pytest.mark.asyncio
    async def test_async_retry_succeeds(self):
        call_count = 0
        async def func():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise ConnectionError("temp")
            return "async_ok"
        result = await async_retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert result == "async_ok"
        assert call_count == 2

    @pytest.mark.asyncio
    async def test_async_retry_exhausted(self):

            raise ConnectionError("always")
        with pytest.raises(MaxRetriesExhaustedError):
            await async_retry_with_backoff(func, max_retries=3, base_delay=0.01)

    @pytest.mark.asyncio
    async def test_async_non_retryable_no_retry(self):
        call_count = 0
        async def func():
            nonlocal call_count
            call_count += 1
            raise ValueError("no retry")
        with pytest.raises(ValueError):
            await async_retry_with_backoff(func, max_retries=3, base_delay=0.01)
        assert call_count == 1

class TestRetryDecorator:
    def test_decorator_success(self):
        call_count = 0
        @retry(max_retries=3, base_delay=0.01)
        def func():
            nonlocal call_count
            call_count += 1
            if call_count < 2:
                raise ConnectionError("temp")
            return "ok"
        result = func()
        assert result == "ok"

    def test_decorator_retry_exhausted(self):
        @retry(max_retries=3, base_delay=0.01)
        def func():
            raise ConnectionError("fail")
        with pytest.raises(MaxRetriesExhaustedError):
            func()

    def test_decorator_preserves_name(self):
        @retry(max_retries=1, base_delay=0.01)
        def my_special_function():
            return "ok"
        assert my_special_function.__name__ == "my_special_function"
