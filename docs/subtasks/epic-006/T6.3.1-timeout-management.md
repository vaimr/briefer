# T6.3.1 — Timeout Management for Bot and Worker

## Цель
Реализовать модуль таймаутов для операций бота и воркера. Каждый внешний вызов (Matrix API, Redis, LLM, ffmpeg) должен иметь конфигурируемый таймаут. При превышении — raise `TimeoutError` с логированием.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `coro` — coroutine для выполнения; `timeout: float | None` — таймаут в секундах; `operation_name: str` — имя операции для логирования |
| **Обработка** | 1. Если `timeout is None` → await coro без ограничения. 2. Если `timeout <= 0` → raise `ValueError`. 3. Иначе → `asyncio.wait_for(coro, timeout=timeout)`. 4. При `asyncio.TimeoutError` → raise `TimeoutError` с логированием. |
| **Выход** | Возвращаемое значение coro, или raise `TimeoutError` / `ValueError` |

## Пошаговый план

1. **Создать `bot/timeout.py`**:
```python
import asyncio
import logging
from typing import Any, Optional

logger = logging.getLogger(__name__)


class TimeoutConfig:
    """Timeout configuration for bot and worker operations."""

    def __init__(
        self,
        matrix_timeout: float = 30.0,
        redis_timeout: float = 10.0,
        whisper_timeout: float = 300.0,
        llm_timeout: float = 60.0,
        pdf_timeout: float = 30.0,
        ffmpeg_timeout: float = 120.0,
    ):
        self.matrix_timeout = matrix_timeout
        self.redis_timeout = redis_timeout
        self.whisper_timeout = whisper_timeout
        self.llm_timeout = llm_timeout
        self.pdf_timeout = pdf_timeout
        self.ffmpeg_timeout = ffmpeg_timeout

    @classmethod
    def from_dict(cls, data: dict) -> "TimeoutConfig":
        """Create TimeoutConfig from a dictionary."""
        return cls(
            matrix_timeout=data.get("matrix_timeout", 30.0),
            redis_timeout=data.get("redis_timeout", 10.0),
            whisper_timeout=data.get("whisper_timeout", 300.0),
            llm_timeout=data.get("llm_timeout", 60.0),
            pdf_timeout=data.get("pdf_timeout", 30.0),
            ffmpeg_timeout=data.get("ffmpeg_timeout", 120.0),
        )


async def with_timeout(
    coro,
    timeout: Optional[float] = None,
    operation_name: str = "operation",
) -> Any:
    """Execute a coroutine with a timeout.

    Args:
        coro: Coroutine to execute.
        timeout: Timeout in seconds. None means no timeout.
        operation_name: Name for logging.

    Returns:
        Result of the coroutine.

    Raises:
        TimeoutError: If the coroutine exceeds the timeout.
        ValueError: If timeout is negative or zero.
    """
    if timeout is not None:
        if timeout <= 0:
            raise ValueError(f"Timeout must be positive, got {timeout}")
        try:
            return await asyncio.wait_for(coro, timeout=timeout)
        except asyncio.TimeoutError:
            logger.error(
                "Operation '%s' timed out after %.1fs",
                operation_name, timeout,
            )
            raise TimeoutError(
                f"Operation '{operation_name}' timed out after {timeout}s"
            ) from None
    else:
        return await coro
```

2. **Написать unit-тесты**:
```python
# tests/unit/test_timeout.py
import pytest
import asyncio
from bot.timeout import with_timeout, TimeoutConfig


class TestWithTimeout:
    @pytest.mark.asyncio
    async def test_operation_completes_in_time(self):
        result = await with_timeout(asyncio.sleep(0.01), timeout=5.0)
        assert result is None

    @pytest.mark.asyncio
    async def test_operation_times_out(self):
        with pytest.raises(TimeoutError):
            await with_timeout(asyncio.sleep(10.0), timeout=0.1, operation_name="slow_op")

    @pytest.mark.asyncio
    async def test_no_timeout(self):
        result = await with_timeout(asyncio.sleep(0.01), timeout=None)
        assert result is None

    @pytest.mark.asyncio
    async def test_timeout_zero_raises_value_error(self):
        with pytest.raises(ValueError):
            await with_timeout(asyncio.sleep(0.01), timeout=0.0)

    @pytest.mark.asyncio
    async def test_timeout_negative_raises_value_error(self):
        with pytest.raises(ValueError):
            await with_timeout(asyncio.sleep(0.01), timeout=-1.0)

    @pytest.mark.asyncio
    async def test_timeout_returns_coro_result(self):
        async def coro():
            return 42
        result = await with_timeout(coro(), timeout=5.0)
        assert result == 42

    @pytest.mark.asyncio
    async def test_timeout_error_message(self):
        with pytest.raises(TimeoutError) as exc_info:
            await with_timeout(asyncio.sleep(10.0), timeout=0.1, operation_name="test_op")
        assert "test_op" in str(exc_info.value)
        assert "0.1s" in str(exc_info.value)

    @pytest.mark.asyncio
    async def test_timeout_with_value_error_negative(self):
        with pytest.raises(ValueError):
            await with_timeout(asyncio.sleep(0.01), timeout=-5.0)


class TestTimeoutConfig:
    def test_defaults(self):
        config = TimeoutConfig()
        assert config.matrix_timeout == 30.0
        assert config.redis_timeout == 10.0
        assert config.whisper_timeout == 300.0
        assert config.llm_timeout == 60.0
        assert config.pdf_timeout == 30.0
        assert config.ffmpeg_timeout == 120.0

    def test_custom_values(self):
        config = TimeoutConfig(
            matrix_timeout=60.0,
            redis_timeout=5.0,
        )
        assert config.matrix_timeout == 60.0
        assert config.redis_timeout == 5.0
        assert config.whisper_timeout == 300.0  # default

    def test_from_dict(self):
        config = TimeoutConfig.from_dict({
            "matrix_timeout": 45.0,
            "llm_timeout": 120.0,
        })
        assert config.matrix_timeout == 45.0
        assert config.llm_timeout == 120.0
        assert config.redis_timeout == 10.0  # default

    def test_from_dict_empty(self):
        config = TimeoutConfig.from_dict({})
        assert config.matrix_timeout == 30.0  # all defaults

    def test_from_dict_invalid_type(self):
        with pytest.raises(TypeError):
            TimeoutConfig.from_dict("not a dict")
```

## Граничные случаи
- `timeout=0` → raise `ValueError`
- `timeout < 0` → raise `ValueError`
- `timeout=None` → без ограничения (default)
- Операция завершается в пределах таймаута → no exception
- Операция never completes → `TimeoutError`
- `asyncio.TimeoutError` перехватывается и переупаковывается в `TimeoutError`
