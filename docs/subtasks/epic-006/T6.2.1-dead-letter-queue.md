# T6.2.1 — Dead Letter Queue (DLQ)

## Цель
Реализовать Dead Letter Queue для задач, которые не удалось обработать после всех retry-попыток. DLQ хранит metadata о неудачных задачах для последующего анализа и повторной обработки.

## Spec

| Поле | Описание |
|------|----------|
| **Вход** | `task_id: str` — ID задачи; `error: str` — описание ошибки; `metadata: dict` — доп. данные (room_id, file_path, timestamp); `queue_name: str` — имя Redis-списка |
| **Обработка** | 1. Создание Redis-списка `dlq:{queue_name}`. 2. Сериализация записи в JSON. 3. Добавление через `rpush`. 4. Индексация через hash `dlq:index:{task_id}`. |
| **Выход** | `str` — task_id добавленной записи; `list[dict]` — все записи из DLQ |

## Пошаговый план

1. **Создать `worker/dlq.py`**:
```python
import json
import logging
import time
from typing import Optional, Any

logger = logging.getLogger(__name__)

class DeadLetterQueue:
    """Dead Letter Queue backed by Redis.

    Stores failed tasks with metadata for later analysis and reprocessing.
    Uses Redis list (rpush/lrange) for the DLQ and hash for indexing.
    """

    def __init__(self, redis_client, queue_name: str = "transcription_dlq"):
        self.redis = redis_client
        self.queue_name = queue_name
        self.dlq_key = f"dlq:{queue_name}"
        self.index_key = f"dlq:index:{queue_name}"

    def add(
        self,
        task_id: str,
        error: str,
        metadata: Optional[dict] = None,
    ) -> str:
        """Add a failed task to the DLQ.

        Args:
            task_id: Unique task identifier.
            error: Error message describing the failure.
            metadata: Additional context (room_id, file_path, etc.).

        Returns:
            task_id of the added record.
        """
        record = {
            "task_id": task_id,
            "error": error,
            "metadata": metadata or {},
            "timestamp": time.time(),
            "retried": True,
        }

        # Add to DLQ list
        self.redis.rpush(self.dlq_key, json.dumps(record))

        # Index by task_id for fast lookup
        self.redis.hset(
            self.index_key,
            task_id,
            json.dumps({"task_id": task_id, "timestamp": record["timestamp"]}),
        )

        logger.info(
            "Added task %s to DLQ '%s': %s",
            task_id, self.queue_name, error,
        )
        return task_id

    def get(self, task_id: str) -> Optional[dict]:
        """Get a DLQ record by task_id.

        Returns:
            Record dict or None if not found.
        """
        index_entry = self.redis.hget(self.index_key, task_id)
        if not index_entry:
            return None

        # Get full record from list (scan for task_id)
        all_records = self.get_all()
        for record in all_records:
            if record["task_id"] == task_id:
                return record
        return None

    def get_all(self, limit: Optional[int] = None) -> list[dict]:
        """Get all records from the DLQ.

        Args:
            limit: Maximum number of records to return.

        Returns:
            List of record dicts.
        """
        raw = self.redis.lrange(self.dlq_key, 0, limit - 1 if limit else -1)
        records = []
        for item in raw:
            if isinstance(item, bytes):
                item = item.decode("utf-8")
            records.append(json.loads(item))
        return records

    def get_recent(self, count: int = 10) -> list[dict]:
        """Get the most recent N records from the DLQ."""
        return self.get_all(limit=count)

    def clear(self) -> int:
        """Clear all records from the DLQ.

        Returns:
            Number of records removed.
        """
        count = self.redis.llen(self.dlq_key)
        self.redis.delete(self.dlq_key)
        self.redis.delete(self.index_key)
        logger.info("Cleared %d records from DLQ '%s'", count, self.queue_name)
        return count

    def size(self) -> int:
        """Get the number of records in the DLQ."""
        return self.redis.llen(self.dlq_key)

    def remove(self, task_id: str) -> bool:
        """Remove a specific record from the DLQ.

        Returns:
            True if removed, False if not found.
        """
        records = self.get_all()
        for i, record in enumerate(records):
            if record["task_id"] == task_id:
                # Remove from list by index
                self.redis.lset(self.dlq_key, i, "__DELETED__")
                self.redis.ltrim(self.dlq_key, 0, -2)
                # Remove from index
                self.redis.hdel(self.index_key, task_id)
                logger.info("Removed task %s from DLQ", task_id)
                return True
        return False

    def purge_old(self, max_age_hours: int = 168) -> int:
        """Remove records older than max_age_hours.

        Args:
            max_age_hours: Maximum age in hours.

        Returns:
            Number of records removed.
        """
        cutoff = time.time() - (max_age_hours * 3600)
        records = self.get_all()
        removed = 0
        for record in records:
            if record["timestamp"] < cutoff:
                if self.remove(record["task_id"]):
                    removed += 1
        return removed
```

import json
import time

        mock_redis.lrange.return_value = []
        mock_redis.llen.return_value = 0
        mock_redis.delete.return_value = 1
        mock_redis.hget.return_value = None
        mock_redis.hdel.return_value = 1
        return DeadLetterQueue(mock_redis, "test_dlq")

    def test_add_records_to_list(self, dlq, mock_redis):
        dlq.add("t-001", "ConnectionError: timeout")
        mock_redis.rpush.assert_called_once()
        call_args = mock_redis.rpush.call_args
        assert call_args[0][0] == "dlq:test_dlq"
        record = json.loads(call_args[0][1])
        assert record["task_id"] == "t-001"
        assert record["error"] == "ConnectionError: timeout"
        assert "timestamp" in record
        assert record["retried"] is True

    def test_add_indexes_by_task_id(self, dlq, mock_redis):
        dlq.add("t-001", "timeout error")
        mock_redis.hset.assert_called_once()
        call_args = mock_redis.hset.call_args
        assert call_args[0][0] == "dlq:index:test_dlq"
        assert call_args[0][1] == "t-001"

    def test_add_with_metadata(self, dlq, mock_redis):
        dlq.add("t-001", "error", {"room_id": "!room:server", "file": "/audio.wav"})
        call_args = mock_redis.rpush.call_args
        record = json.loads(call_args[0][1])
        assert record["metadata"]["room_id"] == "!room:server"

    def test_get_existing_record(self, dlq, mock_redis):
        record = {"task_id": "t-001", "error": "fail", "metadata": {}, "timestamp": 123, "retried": True}
        mock_redis.hget.return_value = json.dumps({"task_id": "t-001", "timestamp": 123})
        mock_redis.lrange.return_value = [json.dumps(record).encode()]
        result = dlq.get("t-001")
        assert result is not None
        assert result["task_id"] == "t-001"
        assert result["error"] == "fail"

    def test_get_nonexistent_record(self, dlq, mock_redis):
        mock_redis.hget.return_value = None
        result = dlq.get("t-999")
        assert result is None

    def test_get_all_empty(self, dlq, mock_redis):
        mock_redis.lrange.return_value = []
        result = dlq.get_all()
        assert result == []

    def test_get_all_with_records(self, dlq, mock_redis):
        records = [
            {"task_id": "t-001", "error": "e1", "metadata": {}, "timestamp": 100, "retried": True},
            {"task_id": "t-002", "error": "e2", "metadata": {}, "timestamp": 200, "retried": True},
        ]
        mock_redis.lrange.return_value = [json.dumps(r).encode() for r in records]
        result = dlq.get_all()
        assert len(result) == 2
        assert result[0]["task_id"] == "t-001"
        assert result[1]["task_id"] == "t-002"

    def test_get_recent(self, dlq, mock_redis):
        records = [{"task_id": f"t-{i}", "error": f"e{i}", "metadata": {}, "timestamp": i, "retried": True} for i in range(5)]
        mock_redis.lrange.return_value = [json.dumps(r).encode() for r in records]
        result = dlq.get_recent(count=3)
        assert len(result) == 3

    def test_get_all_with_limit(self, dlq, mock_redis):
        mock_redis.lrange.return_value = []
        dlq.get_all(limit=5)
        mock_redis.lrange.assert_called_once()
        call_args = mock_redis.lrange.call_args
        assert call_args[0][1] == 0
        assert call_args[0][2] == 4  # limit - 1

    def test_clear(self, dlq, mock_redis):
        mock_redis.llen.return_value = 5
        count = dlq.clear()
        assert count == 5
        mock_redis.delete.assert_any_call("dlq:test_dlq")
        mock_redis.delete.assert_any_call("dlq:index:test_dlq")

    def test_size(self, dlq, mock_redis):
        mock_redis.llen.return_value = 42
        assert dlq.size() == 42

    def test_remove_existing(self, dlq, mock_redis):
        records = [{"task_id": "t-001", "error": "e", "metadata": {}, "timestamp": 100, "retried": True}]
        mock_redis.lrange.return_value = [json.dumps(r).encode()]
        result = dlq.remove("t-001")
        assert result is True
        mock_redis.hdel.assert_called_once()

    def test_remove_nonexistent(self, dlq, mock_redis):
        records = [{"task_id": "t-001", "error": "e", "metadata": {}, "timestamp": 100, "retried": True}]
        mock_redis.lrange.return_value = [json.dumps(r).encode()]
        result = dlq.remove("t-999")
        assert result is False

    def test_purge_old_records(self, dlq, mock_redis):
        old_record = {"task_id": "t-old", "error": "old", "metadata": {}, "timestamp": 0, "retried": True}
        new_record = {"task_id": "t-new", "error": "new", "metadata": {}, "timestamp": 9999999999, "retried": True}
        mock_redis.lrange.return_value = [json.dumps(old_record).encode(), json.dumps(new_record).encode()]
        result = dlq.purge_old(max_age_hours=1)
        assert result == 1

    def test_purge_no_old_records(self, dlq, mock_redis):
        new_record = {"task_id": "t-new", "error": "new", "metadata": {}, "timestamp": 9999999999, "retried": True}

        result = dlq.purge_old(max_age_hours=1)
        assert result == 0

        dlq = DeadLetterQueue(mock_redis)
        assert dlq.queue_name == "transcription_dlq"
        assert dlq.dlq_key == "dlq:transcription_dlq"
