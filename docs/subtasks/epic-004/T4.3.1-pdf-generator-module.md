# T4.3.1 — Create PDF Generator Module

## Цель
Создать модуль генерации PDF из markdown с использованием WeasyPrint.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| Markdown текст | Конвертация HTML → PDF | PDF bytes |
| Markdown текст + путь | Конвертация → сохранение | Файл на диске |

## Пошаговый план

1. Создать `worker/pdf_generator.py`:

```python
"""PDF generator from markdown using WeasyPrint."""

import logging
from io import BytesIO
from typing import Optional

from weasyprint import HTML, CSS

logger = logging.getLogger(__name__)


class PDFGeneratorError(Exception):
    """Base exception for PDF generation errors."""


class PDFGenerator:
    """Generate PDF from markdown content using WeasyPrint."""

    DEFAULT_STYLESHEET = b"""
        @page {
            size: A4;
            margin: 2cm;
        }
        body {
            font-family: Arial, Helvetica, sans-serif;
            font-size: 11pt;
            line-height: 1.5;
            color: #333;
        }
        h1 {
            font-size: 24pt;
            color: #1a1a1a;
            border-bottom: 2px solid #1a1a1a;
            padding-bottom: 8px;
        }
        h2 {
            font-size: 18pt;
            color: #333;
            margin-top: 24px;
        }
        h3 {
            font-size: 14pt;
            color: #555;
        }
        p {
            margin-bottom: 12px;
        }
        ul, ol {
            margin-bottom: 12px;
        }
        li {
            margin-bottom: 4px;
        }
        hr {
            border: none;
            border-top: 1px solid #ccc;
            margin: 16px 0;
        }
        blockquote {
            border-left: 4px solid #ccc;
            margin-left: 0;
            padding-left: 12px;
            color: #666;
            font-style: italic;
        }
        code {
            font-family: monospace;
            background: #f4f4f4;
            padding: 2px 4px;
            border-radius: 3px;
        }
        pre {
            background: #f4f4f4;
            padding: 12px;
            border-radius: 4px;
            overflow-x: auto;
        }
        strong {
            font-weight: bold;
        }
        em {
            font-style: italic;
        }
    """

    def __init__(self, custom_css: Optional[bytes] = None):
        """Initialize PDF generator.

        Args:
            custom_css: Optional custom CSS bytes.
        """
        self.stylesheet = custom_css or self.DEFAULT_STYLESHEET

    def generate_pdf(self, markdown_text: str) -> bytes:
        """Generate PDF from markdown text.

        Args:
            markdown_text: Markdown content to convert.

        Returns:
            PDF bytes.

        Raises:
            PDFGeneratorError: On generation failure.
        """
        if not markdown_text or not markdown_text.strip():
            raise PDFGeneratorError("Empty markdown text")

        html_content = self._markdown_to_html(markdown_text)
        full_html = self._wrap_html(html_content)

        try:
            pdf_bytes = HTML(string=full_html).write_pdf(
                stylesheets=[CSS(data=self.stylesheet)]
            )
            return pdf_bytes
        except Exception as e:
            raise PDFGeneratorError(f"PDF generation failed: {e}") from e

    def save_pdf(self, markdown_text: str, output_path: str) -> str:
        """Generate PDF and save to file.

        Args:
            markdown_text: Markdown content.
            output_path: Path to save PDF file.

        Returns:
            Path to saved PDF file.

        Raises:
            PDFGeneratorError: On generation failure.
        """
        pdf_bytes = self.generate_pdf(markdown_text)
        try:
            with open(output_path, "wb") as f:
                f.write(pdf_bytes)
            logger.info(f"PDF saved to {output_path}")
            return output_path
        except IOError as e:
            raise PDFGeneratorError(f"Failed to save PDF: {e}") from e

    def generate_pdf_io(self, markdown_text: str) -> BytesIO:
        """Generate PDF and return as BytesIO.

        Args:
            markdown_text: Markdown content.

        Returns:
            BytesIO object containing PDF data.

        Raises:
            PDFGeneratorError: On generation failure.
        """
        pdf_bytes = self.generate_pdf(markdown_text)
        return BytesIO(pdf_bytes)

    @staticmethod
    def _markdown_to_html(text: str) -> str:
        """Convert markdown to HTML (basic conversion).

        Args:
            text: Markdown text.

        Returns:
            HTML string.
        """
        import re

        html = text

        # Headers
        html = re.sub(r"^### (.+)$", r"<h3>\1</h3>", html, flags=re.MULTILINE)
        html = re.sub(r"^## (.+)$", r"<h2>\1</h2>", html, flags=re.MULTILINE)
        html = re.sub(r"^# (.+)$", r"<h1>\1</h1>", html, flags=re.MULTILINE)

        # Bold and italic
        html = re.sub(r"\*\*(.+?)\*\*", r"<strong>\1</strong>", html)
        html = re.sub(r"\*(.+?)\*", r"<em>\1</em>", html)

        # Code blocks
        html = re.sub(
            r"```(.+?)```",
            r"<pre><code>\1</code></pre>",
            html,
            flags=re.DOTALL,
        )

        # Inline code
        html = re.sub(r"`(.+?)`", r"<code>\1</code>", html)

        # Unordered lists
        html = re.sub(r"^[-*] (.+)$", r"<li>\1</li>", html, flags=re.MULTILINE)

        # Paragraphs
        html = re.sub(r"\n\n", "</p>\n\n<p>", html)
        html = f"<p>{html}</p>"

        # Horizontal rule
        html = re.sub(r"^---$", "<hr>", html, flags=re.MULTILINE)

        # Clean up empty paragraphs
        html = re.sub(r"<p>\s*</p>", "", html)

        return html

    @staticmethod
    def _wrap_html(content: str) -> str:
        """Wrap HTML content in full document.

        Args:
            content: HTML body content.

        Returns:
            Full HTML document string.
        """
        return f"""<!DOCTYPE html>
<html>
<head>
    <meta charset="utf-8">
    <title>Summary</title>
</head>
<body>
{content}
</body>
</html>"""
```

## Граничные случаи

- Пустой текст → PDFGeneratorError
- Текст только из пробелов → PDFGeneratorError
- Markdown без special chars → HTML рендерится
- Markdown с заголовками → `<h1>`, `<h2>`, `<h3>`
- Markdown с **bold** → `<strong>`
- Markdown с `code` → `<code>`
- Markdown с ```code block``` → `<pre><code>`
- Markdown с `---` → `<hr>`
- Markdown с `- item` → `<li>`
- Невалидный markdown → HTML рендерится как есть
- WeasyPrint недоступен → ImportError → PDFGeneratorError
