## Цель

## Spec

| Input | Processing | Output |
|-------|-----------|--------|

## Пошаговый план

1. Создать `worker/chunking.py`:

```python
"""Text chunking for LLM context limits."""

import re
from typing import List


def chunk_text(text: str, chunk_size: int = 3000) -> List[str]:
    """Split text into chunks respecting sentence boundaries.

    Chunks are split on sentence boundaries (.!? followed by whitespace)
    and filtered to exclude micro-chunks < 200 characters.

    Args:
        text: Input text to chunk.
        chunk_size: Maximum chunk size in characters. Default: 3000.

    Returns:
        List of text chunks, each <= chunk_size characters.
        Returns [text] if len(text) <= chunk_size.
        Returns [] for empty/whitespace-only input.
    """
    if not text or not text.strip():
        return []

    if len(text) <= chunk_size:
        return [text]

    # Split on sentence boundaries
    sentences = re.split(r'(?<=[.!?])\s+', text.strip())

    chunks: List[str] = []
    current_chunk: List[str] = []
    current_length = 0

    for sentence in sentences:
        sentence_length = len(sentence)

        # If single sentence > chunk_size, force split
        if sentence_length > chunk_size:
            # Flush current chunk
            if current_chunk:
                chunk_text_str = " ".join(current_chunk)
                if len(chunk_text_str) >= 200:
                    chunks.append(chunk_text_str)
                current_chunk = []
                current_length = 0

            # Force split long sentence
            words = sentence.split()
            force_chunk: List[str] = []
            force_length = 0
            for word in words:
                if force_length + len(word) + 1 > chunk_size:
                    force_chunk_str = " ".join(force_chunk)
                    if len(force_chunk_str) >= 200:
                        chunks.append(force_chunk_str)
                    force_chunk = []
                    force_length = 0
                force_chunk.append(word)
                force_length += len(word) + 1

            if force_chunk:
                force_chunk_str = " ".join(force_chunk)
                if len(force_chunk_str) >= 200:
                    chunks.append(force_chunk_str)

        elif current_length + sentence_length + (1 if current_chunk else 0) <= chunk_size:
            # Add to current chunk
            if current_chunk:
                current_chunk.append(sentence)
                current_length += sentence_length + 1
            else:
                current_chunk.append(sentence)
                current_length = sentence_length
        else:
            # Flush current chunk
            chunk_text_str = " ".join(current_chunk)
            if len(chunk_text_str) >= 200:
                chunks.append(chunk_text_str)
            current_chunk = [sentence]
            current_length = sentence_length

    # Flush remaining
    if current_chunk:
        chunk_text_str = " ".join(current_chunk)
        if len(chunk_text_str) >= 200:
            chunks.append(chunk_text_str)

    return chunks


def merge_chunks(chunks: List[str], merge_threshold: int = 500) -> List[str]:
    """Merge adjacent chunks that are below merge_threshold.

    Args:
        chunks: List of text chunks.
        merge_threshold: Merge chunks if combined size < threshold.

    Returns:
        Merged chunks list.
    """
    if not chunks:
        return []

    merged: List[str] = []
    current = chunks[0]

    for chunk in chunks[1:]:
        combined = current + " " + chunk
        if len(combined) < merge_threshold:
            current = combined
        else:
            merged.append(current)
            current = chunk

    merged.append(current)
    return merged
```

## Граничные случаи

- Пустой текст `""` → `[]`
- Текст только из пробелов `"   "` → `[]`
- Текст короче chunk_size → `[text]` (без фильтрации)
- Один длинный чанк (> chunk_size) → force split
- Микро-чанк < 200 символов → фильтруется
- Предложение без точки в конце → остаётся в чанке
- Многоточие `...` → не считется как конец предложения
- `merge_chunks([])` → `[]`
- `merge_chunks([chunk])` → `[chunk]` (без изменений)
- `merge_threshold < len(chunk)` → чанк не объединяется
