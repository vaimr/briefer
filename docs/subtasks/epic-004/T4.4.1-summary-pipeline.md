# T4.4.1 — Build Summary Pipeline

## Цель
Создать конвейер обработки: транскрипция → чанки → LLM саммари → markdown → PDF.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| Raw transcriptions | Full pipeline | PDF bytes |
| Transcription dict | Full pipeline | PDF bytes + metadata |

## Пошаговый план

1. Создать `worker/summary_pipeline.py`:

```python
"""Summary generation pipeline: transcription → LLM summary → PDF."""

import logging
from datetime import datetime
from typing import Dict, List, Optional

from worker.chunking import chunk_text
from worker.llm_client import LLMClient, LLMClientError
from worker.llm_response import LLMResponse, parse_llm_response
from worker.pdf_generator import PDFGenerator
from worker.pdf_styles import PDFStyleConfig
from worker.templates import SummaryTemplate

logger = logging.getLogger(__name__)


class SummaryPipelineError(Exception):
    """Base exception for pipeline errors."""


class SummaryPipeline:
    """End-to-end summary generation pipeline."""

    def __init__(
        self,
        llm_client: LLMClient,
        pdf_generator: Optional[PDFGenerator] = None,
        style_config: Optional[PDFStyleConfig] = None,
        chunk_size: int = 3000,
    ):
        """Initialize pipeline.

        Args:
            llm_client: LLM client for summarization.
            pdf_generator: PDF generator. Defaults to default config.
            style_config: PDF style config.
            chunk_size: Chunk size for text splitting.
        """
        self.llm_client = llm_client
        self.pdf_generator = pdf_generator or PDFGenerator(
            custom_css=style_config.generate_css() if style_config else None
        )
        self.chunk_size = chunk_size

    def generate_summary(
        self,
        transcription_text: str,
        date: Optional[str] = None,
        duration: str = "N/A",
    ) -> str:
        """Generate summary from transcription text.

        Args:
            transcription_text: Raw transcription text.
            date: Date string.
            duration: Duration string.

        Returns:
            Summary text.

        Raises:
            SummaryPipelineError: On pipeline failure.
        """
        if not transcription_text or not transcription_text.strip():
            return "Нет данных для саммари"

        try:
            # Chunk the transcription
            chunks = chunk_text(transcription_text, self.chunk_size)
            if not chunks:
                return "Нет данных для саммари"

            # Summarize each chunk
            summaries = []
            for i, chunk in enumerate(chunks):
                logger.info(f"Summarizing chunk {i + 1}/{len(chunks)}")
                try:
                    chunk_summary = self.llm_client.summarize(chunk)
                    summaries.append(chunk_summary)
                except LLMClientError as e:
                    logger.error(f"Failed to summarize chunk {i + 1}: {e}")
                    summaries.append(f"[Ошибка при саммари чанка {i + 1}]")

            # Combine summaries
            combined = "\n\n".join(summaries)
            return combined[: self.llm_client.MAX_SUMMARY_LENGTH]

        except Exception as e:
            raise SummaryPipelineError(f"Pipeline failure: {e}") from e

    def generate_pdf(
        self,
        transcription_text: str,
        date: Optional[str] = None,
        duration: str = "N/A",
        key_points: Optional[List[str]] = None,
    ) -> bytes:
        """Generate PDF from transcription text.

        Args:
            transcription_text: Raw transcription text.
            date: Date string.
            duration: Duration string.
            key_points: Optional key points.

        Returns:
            PDF bytes.

        Raises:
            SummaryPipelineError: On pipeline failure.
        """
        # Generate summary
        summary = self.generate_summary(transcription_text, date, duration)

        # Generate markdown
        markdown = SummaryTemplate.render_summary(
            summary=summary,
            date=date,
            duration=duration,
            key_points=key_points,
        )

        # Generate PDF
        return self.pdf_generator.generate_pdf(markdown)

    def generate_full_report(
        self,
        transcription_text: str,
        date: Optional[str] = None,
        duration: str = "N/A",
        key_points: Optional[List[str]] = None,
    ) -> Dict[str, object]:
        """Generate full report with summary and PDF.

        Args:
            transcription_text: Raw transcription text.
            date: Date string.
            duration: Duration string.
            key_points: Optional key points.

        Returns:
            Dict with summary, markdown, and PDF bytes.
        """
        if date is None:
            date = datetime.now().strftime("%Y-%m-%d %H:%M")

        summary = self.generate_summary(transcription_text, date, duration)
        markdown = SummaryTemplate.render_summary(
            summary=summary,
            date=date,
            duration=duration,
            key_points=key_points,
        )
        pdf_bytes = self.pdf_generator.generate_pdf(markdown)

        return {
            "summary": summary,
            "markdown": markdown,
            "pdf": pdf_bytes,
            "date": date,
            "duration": duration,
        }
```

## Граничные случаи

- Пустая транскрипция → "Нет данных для саммари"
- Провал LLM на одном чанке → "[Ошибка при саммари чанка N]"
- Все чанки провалили → все "[Ошибка...]"
- Очень длинная транскрипция (>10000 символов) → много чанков
- chunk_size = 100 → много маленьких чанков
- chunk_size = 10000 → 1 большой чанок
- date = None → datetime.now()
- key_points = None → "*Нет ключевых моментов*"
- key_points = [] → "*Нет ключевых моментов*"
- PDF генерация провалилась → SummaryPipelineError
