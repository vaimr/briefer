## Spec

| Input | Processing | Output |

1. Создать `worker/pdf_styles.py`:

```python
"""PDF styling configuration and CSS generation."""

from typing import Dict, Optional


class PDFStyleConfig:
    """Configuration for PDF styling."""

    def __init__(
        self,
        font_family: str = "Arial, Helvetica, sans-serif",
        font_size: str = "11pt",
        line_height: float = 1.5,
        color: str = "#333333",
        margin: str = "2cm",
        page_size: str = "A4",
        heading_color: str = "#1a1a1a",
        link_color: str = "#0066cc",
    ):
        """Initialize PDF style configuration.

        Args:
            font_family: Font family.
            font_size: Base font size.
            line_height: Line height multiplier.
            color: Base text color.
            margin: Page margin.
            page_size: Page size (A4, Letter, etc.).
            heading_color: Heading color.
            link_color: Link color.
        """
        self.font_family = font_family
        self.font_size = font_size
        self.line_height = line_height
        self.color = color
        self.margin = margin
        self.page_size = page_size
        self.heading_color = heading_color
        self.link_color = link_color

    def generate_css(self) -> bytes:
        """Generate CSS from configuration.

        Returns:
            CSS bytes for WeasyPrint.
        """
        return f"""
        @page {{
            size: {self.page_size};
            margin: {self.margin};
        }}
        body {{
            font-family: {self.font_family};
            font-size: {self.font_size};
            line-height: {self.line_height};
            color: {self.color};
        }}
        h1 {{
            font-size: 24pt;
            color: {self.heading_color};
            border-bottom: 2px solid {self.heading_color};
            padding-bottom: 8px;
        }}
        h2 {{
            font-size: 18pt;
            color: {self.heading_color};
            margin-top: 24px;
        }}
        h3 {{
            font-size: 14pt;
            color: {self.heading_color};
        }}
        p {{
            margin-bottom: 12px;
        }}
        a {{
            color: {self.link_color};
            text-decoration: none;
        }}
        a:hover {{
            text-decoration: underline;
        }}
        ul, ol {{
            margin-bottom: 12px;
        }}
        li {{
            margin-bottom: 4px;
        }}
        hr {{
            border: none;
            border-top: 1px solid #ccc;
            margin: 16px 0;
        }}
        blockquote {{
            border-left: 4px solid #ccc;
            margin-left: 0;
            padding-left: 12px;
            color: #666;
            font-style: italic;
        }}
        code {{
            font-family: monospace;
            background: #f4f4f4;
            padding: 2px 4px;
            border-radius: 3px;
        }}
        pre {{
            background: #f4f4f4;
            padding: 12px;
            border-radius: 4px;
            overflow-x: auto;
        }}
        strong {{
            font-weight: bold;
        }}
        em {{
            font-style: italic;
        }}
        """.strip().encode("utf-8")

    @classmethod
    def from_dict(cls, config: Dict[str, str]) -> "PDFStyleConfig":
        """Create config from dictionary.

        Args:
            config: Dict of style parameters.

        Returns:
            PDFStyleConfig instance.
        """
        return cls(
            font_family=config.get("font_family", "Arial, Helvetica, sans-serif"),
            font_size=config.get("font_size", "11pt"),
            line_height=float(config.get("line_height", "1.5")),
            color=config.get("color", "#333333"),
            margin=config.get("margin", "2cm"),
            page_size=config.get("page_size", "A4"),
            heading_color=config.get("heading_color", "#1a1a1a"),
            link_color=config.get("link_color", "#0066cc"),
        )

    def to_dict(self) -> Dict[str, str]:
        """Convert config to dictionary.

        Returns:
            Dict of style parameters.
        """
        return {
            "font_family": self.font_family,
            "font_size": self.font_size,
            "line_height": str(self.line_height),
            "color": self.color,
            "margin": self.margin,
            "page_size": self.page_size,
            "heading_color": self.heading_color,
            "link_color": self.link_color,
        }

    def merge(self, other: "PDFStyleConfig") -> "PDFStyleConfig":
        """Merge two configs, other takes precedence.

        Args:
            other: Config to merge with.

        Returns:
            Merged config.
        """
        merged = self.to_dict()
        other_dict = other.to_dict()
        merged.update(other_dict)
        return self.from_dict(merged)
```

2. Создать `tests/unit/test_pdf_styles.py` для проверки.

## Граничные случаи

- Все значения по умолчанию — CSS генерируется корректно
- Кастомные значения — CSS использует кастомные
- `from_dict({})` — все значения по умолчанию
- `from_dict({"line_height": "invalid"})` — ValueError при float()
- `from_dict({"line_height": "2.0"})` — line_height = 2.0
- `to_dict()` — все поля в словаре
- `merge()` — other переопределяет self
- `merge()` — self значения сохраняются если other не задан
- CSS содержит все селекторы: @page, body, h1-h3, p, a, ul, ol, li, hr, blockquote, code, pre, strong, em
- CSS кодировка UTF-8
