# T4.1.1 — Create LLM API Client

## Цель
Создать модуль `worker/llm_client.py` для вызова LLM API (OpenAI-compatible) с retry logic.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| Текст транскрипции | Вызов LLM API → саммари | Текст саммари ≤ 2000 символов |
| Текст транскрипции | Вызов LLM API → stream | Потоковый текст саммари |

## Пошаговый план

1. Создать `worker/llm_client.py`:

```python
"""LLM API client for OpenAI-compatible endpoints."""

import logging
from dataclasses import dataclass, field
from typing import Iterator, Optional

import requests

logger = logging.getLogger(__name__)


@dataclass
class LLMResponse:
    """Response from LLM API."""
    text: str = ""
    tokens_used: int = 0
    finish_reason: str = ""
    model: str = ""


class LLMClientError(Exception):
    """Base exception for LLM client errors."""


class RateLimitError(LLMClientError):
    """Rate limit exceeded."""


class ServerError(LLMClientError):
    """Server error from LLM API."""


class TimeoutError(LLMClientError):
    """Request timeout."""


class LLMClient:
    """Client for OpenAI-compatible LLM API."""

    MAX_SUMMARY_LENGTH = 2000
    MAX_TRANSCRIPTION_LENGTH = 4000
    MAX_RETRIES = 3
    RETRY_DELAY = 2  # seconds
    REQUEST_TIMEOUT = 60  # seconds

    def __init__(self, api_url: str, api_key: str, model: str = "gpt-4"):
        self.api_url = api_url.rstrip("/")
        self.api_key = api_key
        self.model = model
        self.headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        }

    def summarize(self, transcription_text: str) -> str:
        """Generate summary of transcription text.

        Args:
            transcription_text: Raw transcription text.

        Returns:
            Summary text, truncated to MAX_SUMMARY_LENGTH.

        Raises:
            LLMClientError: On API failure after retries.
        """
        if not transcription_text or not transcription_text.strip():
            return "Нет данных для саммари"

        # Truncate if too long
        if len(transcription_text) > self.MAX_TRANSCRIPTION_LENGTH:
            transcription_text = transcription_text[:self.MAX_TRANSCRIPTION_LENGTH]

        payload = {
            "model": self.model,
            "messages": [
                {
                    "role": "system",
                    "content": "Ты — ассистент для краткого изложения аудио-транскрипций. "
                               "Делай саммари до 2000 символов.",
                },
                {
                    "role": "user",
                    "content": f"Сделай краткое саммари этого текста:\n\n{transcription_text}",
                },
            ],
            "max_tokens": 1500,
        }

        for attempt in range(self.MAX_RETRIES):
            try:
                response = requests.post(
                    f"{self.api_url}/chat/completions",
                    json=payload,
                    headers=self.headers,
                    timeout=self.REQUEST_TIMEOUT,
                )
                response.raise_for_status()
                data = response.json()
                summary = data["choices"][0]["message"]["content"]
                return summary[: self.MAX_SUMMARY_LENGTH]

            except requests.exceptions.HTTPError as e:
                status = getattr(e.response, "status_code", None)
                if status == 429:
                    raise RateLimitError(f"Rate limited: {e}") from e
                if status and status >= 500:
                    if attempt == self.MAX_RETRIES - 1:
                        raise ServerError(f"Server error: {e}") from e
                else:
                    raise LLMClientError(f"LLM API error: {e}") from e

            except requests.exceptions.ConnectionError:
                if attempt == self.MAX_RETRIES - 1:
                    raise LLMClientError("Connection failed after retries") from None
                logger.warning(f"Connection error, retrying... (attempt {attempt + 1})")

            except requests.exceptions.Timeout:
                if attempt == self.MAX_RETRIES - 1:
                    raise TimeoutError("Request timed out") from None
                logger.warning(f"Timeout, retrying... (attempt {attempt + 1})")

            import time

            time.sleep(self.RETRY_DELAY)

        raise LLMClientError("Max retries exceeded")
```

2. Создать `tests/unit/test_llm_client.py` для проверки.

## Граничные случаи

- Пустой текст — возвращает "Нет данных для саммари"
- Текст > 4000 символов — truncates до 4000
- API 429 — raises RateLimitError (не retry)
- API 500 — retry 3 раза, затем ServerError
- API 4xx (не 429) — raises LLMClientError (не retry)
- ConnectionError — retry 3 раза, затем LLMClientError
- Timeout — retry 3 раза, затем TimeoutError
- Пустой response — KeyError при доступе к choices
