# T4.4.2 — Add LLM Prompt Engineering

## Цель
Создать систему управления промптами для LLM с поддержкой кастомизации.

## Spec

| Input | Processing | Output |
|-------|-----------|--------|
| System prompt + user text | Формирование payload | Dict для LLM API |
| Кастомный промпт | Валидация → форматирование | Валидный промпт |

## Пошаговый план

1. Создать `worker/prompts.py`:

```python
"""LLM prompt management and engineering."""

from dataclasses import dataclass, field
from typing import Dict, List, Optional


@dataclass
class PromptTemplate:
    """Prompt template with variables."""

    system: str = ""
    user: str = ""
    variables: List[str] = field(default_factory=list)

    def format(self, **kwargs) -> "PromptTemplate":
        """Format template with variables.

        Args:
            **kwargs: Variable values.

        Returns:
            New PromptTemplate with formatted strings.

        Raises:
            ValueError: If required variables are missing.
        """
        for var in self.variables:
            if var not in kwargs:
                raise ValueError(f"Missing required variable: {var}")

        return PromptTemplate(
            system=self.system.format(**kwargs),
            user=self.user.format(**kwargs),
            variables=self.variables,
        )


@dataclass
class PromptPayload:
    """Payload for LLM API call."""

    model: str = "gpt-4"
    messages: List[Dict[str, str]] = field(default_factory=list)
    max_tokens: int = 1500
    temperature: float = 0.7

    def to_dict(self) -> Dict:
        """Convert to API-compatible dict.

        Returns:
            Dict suitable for LLM API.
        """
        payload = {
            "model": self.model,
            "messages": self.messages,
            "max_tokens": self.max_tokens,
        }
        if self.temperature != 0.7:
            payload["temperature"] = self.temperature
        return payload


class PromptEngineer:
    """Engineer prompts for LLM summarization."""

    DEFAULT_SYSTEM_PROMPT = (
        "Ты — ассистент для краткого изложения аудио-транскрипций. "
        "Делай саммари до {max_length} символов. "
        "Используй русский язык. "
        "Выделяй ключевые моменты. "
        "Будь точным и лаконичным."
    )

    DEFAULT_USER_PROMPT = "Сделай краткое саммари этого текста:\n\n{text}"

    SUMMARY_PROMPT = PromptTemplate(
        system=DEFAULT_SYSTEM_PROMPT,
        user=DEFAULT_USER_PROMPT,
        variables=["max_length", "text"],
    )

    KEY_POINTS_PROMPT = PromptTemplate(
        system=(
            "Ты — ассистент для выделения ключевых моментов из текста. "
            "Перечисли до {max_points} ключевых моментов в виде маркированного списка."
        ),
        user="Извлеки ключевые моменты из этого текста:\n\n{text}",
        variables=["max_points", "text"],
    )

    def __init__(
        self,
        system_prompt: Optional[str] = None,
        user_prompt: Optional[str] = None,
        max_tokens: int = 1500,
        temperature: float = 0.7,
    ):
        self.system_prompt = system_prompt or self.DEFAULT_SYSTEM_PROMPT
        self.user_prompt = user_prompt or self.DEFAULT_USER_PROMPT
        self.max_tokens = max_tokens
        self.temperature = temperature

    def create_summary_prompt(
        self,
        text: str,
        max_length: int = 2000,
    ) -> PromptPayload:
        """Create summary prompt payload.

        Args:
            text: Text to summarize.
            max_length: Max summary length.

        Returns:
            PromptPayload for LLM API.
        """
        formatted = self.SUMMARY_PROMPT.format(max_length=max_length, text=text)
        messages = [
            {"role": "system", "content": formatted.system},
            {"role": "user", "content": formatted.user},
        ]
        return PromptPayload(
            messages=messages,
            max_tokens=self.max_tokens,
            temperature=self.temperature,
        )

    def create_key_points_prompt(
        self,
        text: str,
        max_points: int = 10,
    ) -> PromptPayload:
        """Create key points extraction prompt.

        Args:
            text: Text to extract key points from.
            max_points: Maximum number of key points.

        Returns:
            PromptPayload for LLM API.
        """
        formatted = self.KEY_POINTS_PROMPT.format(max_points=max_points, text=text)
        messages = [
            {"role": "system", "content": formatted.system},
            {"role": "user", "content": formatted.user},
        ]
        return PromptPayload(
            messages=messages,
            max_tokens=self.max_tokens,
            temperature=self.temperature,
        )

    def create_custom_prompt(
        self,
        system: str,
        user: str,
        variables: Optional[List[str]] = None,
    ) -> PromptTemplate:
        """Create custom prompt template.

        Args:
            system: System prompt template.
            user: User prompt template.
            variables: List of variable names.

        Returns:
            PromptTemplate instance.
        """
        vars_list = variables or []
        return PromptTemplate(
            system=system,
            user=user,
            variables=vars_list,
        )

    def validate_prompt(self, payload: PromptPayload) -> bool:
        """Validate prompt payload.

        Args:
            payload: PromptPayload to validate.

        Returns:
            True if valid.

        Raises:
            ValueError: If validation fails.
        """
        if not payload.messages:
            raise ValueError("No messages in payload")

        for msg in payload.messages:
            if "role" not in msg:
                raise ValueError("Message missing 'role' field")
            if "content" not in msg:
                raise ValueError("Message missing 'content' field")
            if msg["role"] not in ("system", "user", "assistant"):
                raise ValueError(f"Invalid role: {msg['role']}")
            if not msg["content"].strip():
                raise ValueError("Message content is empty")

        if payload.max_tokens <= 0:
            raise ValueError(f"Invalid max_tokens: {payload.max_tokens}")

        if not (0.0 <= payload.temperature <= 2.0):
            raise ValueError(f"Invalid temperature: {payload.temperature}")

        return True
```

## Граничные случаи

- `PromptTemplate.format()` без обязательных переменных — ValueError
- `PromptTemplate.format()` с лишними переменными — игнорируются
- `PromptPayload.to_dict()` с default temperature — не включает temperature
- `PromptPayload.to_dict()` с кастомной temperature — включает temperature
- `validate_prompt([])` — ValueError "No messages"
- `validate_prompt([{"role": "system"}])` — ValueError "missing content"
- `validate_prompt([{"role": "invalid"}])` — ValueError "Invalid role"
- `validate_prompt([{"content": ""}])` — ValueError "empty content"
- `validate_prompt([{"role": "system", "content": "   "}])` — ValueError "empty content"
- `validate_prompt(max_tokens=-1)` — ValueError
- `validate_prompt(temperature=3.0)` — ValueError
- `validate_prompt(temperature=0.0)` — OK
- `validate_prompt(temperature=2.0)` — OK
- `create_summary_prompt("")` — пустой текст в user message
- `create_key_points_prompt("", 0)` — max_points = 0
